require utils, utils_steps;

int girderrange;

void girder_restrictor::Init()
{
  girderrange = 300;
}

void girder_restrictor::InitGraphic()
{
}

override void CWorm::Message(CObject* sender, EMType Type, int msize, CMessageData* mdata)
{                          
    local cw = GetCurrentWorm();

	 if(Type == M_FRAME && girderrange != 0)
     {     
          
          
          if(cw != NullObj)
          {
               if(cw->CurWeapon->CheckName("Girder") ||
			      cw->CurWeapon->CheckName("Girder Starter-Pack")  ||
			      cw->CurWeapon->CheckName("Double Girder")  ||
			      cw->CurWeapon->CheckName("3 Girder Pack") ||
				  cw->CurWeapon->CheckName("Super Girder"))
            		{
                    local dx = cw->PosX - GS->landSelectX;
                    local dy = cw->PosY - GS->landSelectY;
                                                                                
                    local distance = sqrt(dx * dx + dy * dy);                    
                    
                    if (distance > girderrange)
                       {
                       local angle = atan2(dx, dy);                       
                       GS->landSelectX = cw->PosX - girderrange * sin(angle);
                       GS->landSelectY = cw->PosY - girderrange * cos(angle);
                       }                                                  
               }    
          }
     }
     
     
     
     super;
}
