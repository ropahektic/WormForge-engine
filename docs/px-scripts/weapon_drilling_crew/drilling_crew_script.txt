require utils, utils_pyroman, p_sprite_builder, utils_curtis, utils_steps, sprite_curtis_249;

CSprite* driller_aiming;
CMissile* dmis_driller;
bool driller;

void drilling_crew_script::InitGraphic()
{
    CFile *f;
    f = GetAttachment("driller.png");
    driller_aiming = LoadSprite(f, 8, 0);
}

//---------------------------------DRILLING CREW------------------------------//

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) {
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Drilling Crew";
     sprite = driller_aiming->Index;
     Params->hand_type = 1;
     Params->hand_scale = 1.2;
     Params->hand_radius = -13.2;
     Params->hand_rotation = -0.0;
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;
     Params->weap_rotation = 0.5;
     Params->weap_radius = -11.0;
     Params->animate = true;
     Params->anim_speed = 1.10;
     Params->dont_rotate_with_angle = true;
     Params->hand_ang_if_static = 0.4;
     Params->ang_if_static = 0.0;
     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}

//---------------------------------DRILLING CREW------------------------------//

override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata)
{
    super;
    Sdigging = false;
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_FRAME)
    {
        if (weap->CheckName("Drilling Crew"))
        {
          if (driller)
          {
            dmis_driller = this; dmis_driller->Sdigging = true; driller = true;
          }
        }

        if (dmis_driller == this && Sdigging)
        {
            GG->land->MakeHole(7,dmis_driller->PosX,dmis_driller->PosY-7.0);
        }

    }

    super;
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
    if (Weap->CheckName("Drilling Crew"))
    {
        driller = true;
    }

    super;
}
