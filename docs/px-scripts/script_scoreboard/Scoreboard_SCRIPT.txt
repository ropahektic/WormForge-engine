void Scoreboard_SCRIPT::InitGraphic()
{ 
     SB_Background  = GetAttachment("scoreboard background.png");
     GUI_Scoreboard = LoadImage(SB_Background);
}

void Scoreboard_SCRIPT::FirstFrame()
{
     new CScoreboardHandler();
     Scoreboard_PosX = 0;
     Scoreboard_PosY = 0;
     
     for(local i = 0; i <= 5; i++)
     {
          if(Root->GetObject(TI_Team, (i + 1)) != NullObj)
          {
               sb_teams_playing[i] = true;
          }
          else
          {
               sb_teams_playing[i] = false;
          }
     }
     
     keySync->AddKey(72); // H
     sb_delay_timer = 0;
}

CScoreboardHandler::CScoreboardHandler()
{
     super(Root, GS);  
     
     GUI_Scoreboard_n = GUI_Scoreboard;
}

void CScoreboardHandler::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{
     local cw = GetCurrentWorm();
   
     if(cw != NullObj)
     {
          if(keySync->KeyPressed(72) && !GameChatOpened)
          {
               if(Type == M_FRAME)
               {
                    ListAndDraw();
               }
               
               if(Type == M_DRAWQUEUE)
               {
                    Scoreboard_PosX = 0;
                    Scoreboard_PosY = 0;
         
                    AddInterfaceBitmap(15, Scoreboard_PosX, Scoreboard_PosY, GUI_Scoreboard_n, 509, 423, 1);
               }
          }
     }
     
     if(game_over == true)
     {
          if(Type == M_FRAME)
          {
               sb_delay_timer += 20;
               
               if(sb_delay_timer > 1000)
               {
                    ListAndDraw();
               }
          }
          
          if(Type == M_DRAWQUEUE)
          {     
               if(sb_delay_timer > 1000)
               {
                    Scoreboard_PosX = 0;
                    Scoreboard_PosY = 0;
         
                    AddInterfaceBitmap(15, Scoreboard_PosX, Scoreboard_PosY, GUI_Scoreboard_n, 509, 423, 1);
               }
          }
     } 
}
