void CMissile::BumperCollision(float Speed)
{
	CGObject *bumperob;

	for(int i = 0; i < Env->Objs.Count; i += 1)
	{
		bumperob = CGObject(Env->Objs.Objs[i]);
		if(bumperob == this) continue;
		if(bumperob == NullObj) continue;
		if(GetCurrentWorm() != NullObj)
		                    {
		                    if(bumperob == GetCurrentWorm()) continue;
		                    }


		if(bumperob->ClType == OC_Worm)
		{
			if(bumperb_COLMASK->Check(PosX, PosY, bumperob->ColMask, bumperob->PosX, bumperob->PosY))
			{
				float ForceX = bumperob->PosX - PosX;
				float ForceY = bumperob->PosY - PosY;
				float dist = sqrt(ForceX*ForceX + ForceY*ForceY);

				if(dist > 0)
				{
					ForceX = 0.7*sqrt(Speed) * ForceX / dist;
					ForceY = 0.7*sqrt(Speed) * ForceY / dist;

					CMessageData msg;
					msg.params[0]	= 0;
					msg.fparams[1]	= PosX;
					msg.fparams[2]	= PosY;
					msg.fparams[3]	= ForceX;
					msg.fparams[4]	= ForceY;
					msg.params[5]	= 1;
					msg.params[6]	= 0;

					bumperob->Message(this, M_GUNEXP, 1032, &msg);
				}
			}
		}
	}
}
