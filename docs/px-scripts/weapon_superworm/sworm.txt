CMissile* swmis;
CColMask* sworm_COLMASK;
bool superworm;

void sworm::Init()
{
}

void sworm::FirstFrame()
{
sworm_COLMASK = new CColMask(10,10,MakeRectMask(20,20));
}

override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata)
{
    super;
    SType = false;
    Sdigging = false;
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_FRAME)
    {
        if (superworm) {swmis = this; swmis->Sdigging = true; superworm = false;}
        
        if (swmis == this && Sdigging)
        {
            if (IsFlying)
            {
                Dodig = true;
                GG->land->MakeHole(8,PosX,PosY);
                
                if (modulef(SpX) > 8 || modulef(SpY) > 8) Sdigging = false;
            }
        }
    }
    
    if(SType == true)
   	{
    	if(Type == M_FRAME)
        {
            SWormCollision(15.0);
        }
		/*
		if(Type == M_DRAWQUEUE)
        {
            if(Sinking)
			{
			SetColorMod(RGB(90, 90, 240), 6);
			}
        }
		*/
   	}
    
    super;
}



override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc) 
{
super;  
if(Weap->CheckName("Super Worm"))
	{
	superworm = true;
	CGObject *swormo;
    CMissile *swormm;
	swormo = CGObject(Env->Objs.Objs[Env->Objs.Count-1]);
	if(swormo->ClType == OC_Missile)
    	{
        swormm = swormo;
        swormm->SType = true;
    	}
 	}
}

void CMissile::SWormCollision(float Speed)
{
	CGObject *swormob;
	
	for(int i = 0; i < Env->Objs.Count; i += 1)
	{
		swormob = CGObject(Env->Objs.Objs[i]);
		if(swormob == this) continue;
		if(swormob == NullObj) continue;
		if(GetCurrentWorm() != NullObj)
		                    {
		                    if(swormob == GetCurrentWorm()) continue;
		                    }
		
		
		if(swormob->ClType == OC_Worm)
		{
			if(sworm_COLMASK->Check(PosX, PosY, swormob->ColMask, swormob->PosX, swormob->PosY))
			{
				float ForceX = swormob->PosX - PosX;
				float ForceY = swormob->PosY - PosY;
				float dist = sqrt(ForceX*ForceX + ForceY*ForceY);
				
				if(dist > 0)
				{
					ForceX = 0.7*sqrt(Speed) * ForceX / dist;
					ForceY = 0.7*sqrt(Speed) * ForceY / dist;
					
					CMessageData msg;
					msg.params[0]	= 0;
					msg.fparams[1]	= PosX;
					msg.fparams[2]	= PosY;
					msg.fparams[3]	= ForceX;
					msg.fparams[4]	= ForceY;
					msg.params[5]	= 1;
					msg.params[6]	= 0;
					
					swormob->Message(this, M_GUNEXP, 1032, &msg);
				}
			}
		}
	}
}
