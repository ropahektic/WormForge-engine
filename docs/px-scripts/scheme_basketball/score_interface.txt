int RedScoreNumber;
int BlueScoreNumber;

void score_interface::FirstFrame()
   {
   local RedScore = new CRedScoreInterface(Root);
   RedScore->KCx = int(GS->LevelSX / 2) - 102;
   RedScore->KCy = int(GS->LevelSY / 2) - 322;
   RedScoreNumber = 0;

   local BlueScore = new CBlueScoreInterface(Root);
   BlueScore->KCx = int(GS->LevelSX / 2) + 102;
   BlueScore->KCy = int(GS->LevelSY / 2) - 322;
   BlueScoreNumber = 0;
   
   new CInterfaceHandler(Root);
   }

CRedScoreInterface::CRedScoreInterface(CObject* parent)
   {
   super(parent,GS);
   CardRed = MakeBitmap(118,70);
   KCx = 0;
   KCy = 0;
   }

void CRedScoreInterface::DrawBackground()
   {
   CardRed->Rect(0,0,118,70,ARGB(255,0,0,0));
   }

void CRedScoreInterface::DrawText()
   {
   local a = 1;
   local b = 1;

   GS->Disp->DrawText(23,CardRed, 52, 21, ItoA(RedScoreNumber), &a, &b);
   }

void CRedScoreInterface::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
   if (Type == M_FRAME)
       {
       DrawBackground();
       DrawText();
       }
   if (Type == M_DRAWQUEUE)
       {
       local Ox = (KCx - GS->ActCameraX);
       local Oy = (KCy - GS->ActCameraY);

       AddInterfaceBitmap(0,Ox ,Oy,CardRed,118,70,0);
       }
}

void score_interface::Init()
{
    for (local i = 0; i < 6; i++)
    	{
    	BBWormColors[i] = 8;
        BBWormNames[i] = "Not playing";
        BBPlayer[i] = NullObj;
    	}
}

override CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
{
    super;
    BBNameReader = true;
    1pointers = 0;
    2pointers = 0;
    3pointers = 0;
    personalscore = 0;
}

void CInterfaceHandler::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
if (Type == M_FRAME)
	{
    ListNDraw();
    }

if (Type == M_DRAWQUEUE)
    {
    AddInterfaceBitmap(0, (-GS->Disp->xRes/2) + (GUIn->SX+60) / 2, (-GS->Disp->yRes/2)+(GUIn->SY+60) / 2, GUIn,266,112,0);
    }
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
if (Type == M_FRAME)
    {
    local w = GetCurrentWorm();
    if (w != NullObj) BBPlayer = CWorm(w);
        
    if (gframe > 1)
		{
		if (BBNameReader)
        	{
    		for (local i = 0; i < 6; i++)
	    		{
    	    	if (BBWormColors[i] == 8)
        	    	{
            		BBWormNames[i] = GetName();
	            	BBWormColors[i] = GetWormColor(this);
    	        	BBPlayer[i] = CWorm(this);
        	    	break;
            		}
        		}
        	BBNameReader = false;
        	}
    	}
	}
super;
}

void CBasketBall::CalculatePoints()
{
    local w = GetCurrentWorm();
    if (w != NullObj)
		{
		BBPlayer = CWorm(w);
    	if(ScoreCoefficient == 1) {BBPlayer->1pointers += 1; BBPlayer->personalscore += 1;}
		if(ScoreCoefficient == 2) {BBPlayer->2pointers += 1; BBPlayer->personalscore += 2;}
		if(ScoreCoefficient == 3) {BBPlayer->3pointers += 1; BBPlayer->personalscore += 3;}
		}
}
