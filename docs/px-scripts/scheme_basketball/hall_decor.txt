void CBasketBall::ShowSprite()
{
        	bb_message_params->scale = 0.0;
            bb_message_params->blend_type = 2;
            bb_message_params->duration = 3;   // seconds, fade duration
            bb_message_params->rot = 178.5;
            bb_message_params->turns = 0;
            bb_message_params->alpha = 100;
            bb_message_params->ZPlane = 3;
            bb_message_params->fade = false;

			if(sprite_roll == 0) bb_message_params->sprite = out->Index;
			else if(sprite_roll == 1)
			    {
			    if	   (ScoreCoefficient == 1) bb_message_params->sprite = b1p->Index; 
			    else if(ScoreCoefficient == 2) bb_message_params->sprite = b2p->Index;
			    else if(ScoreCoefficient == 3)
					{
					bb_message_params->sprite = b3p->Index;
					local i = RandomInt(1, 3);
    				if      (i == 1) crowd1->Play(1.0, 0.0, false);
    				else if (i == 2) crowd2->Play(1.0, 0.0, false);
    				else if (i == 3) crowd3->Play(1.0, 0.0, false);
    				}
			    }
            else if(sprite_roll == 2)
			    {
			    if	   (ScoreCoefficient == 1) bb_message_params->sprite = r1p->Index;
			    else if(ScoreCoefficient == 2) bb_message_params->sprite = r2p->Index;
			    else if(ScoreCoefficient == 3)
					{
					bb_message_params->sprite = r3p->Index;
					local i = RandomInt(1, 3);
    				if      (i == 1) crowd1->Play(1.0, 0.0, false);
    				else if (i == 2) crowd2->Play(1.0, 0.0, false);
    				else if (i == 3) crowd3->Play(1.0, 0.0, false);
    				}
			    }
            else bb_message_params->sprite = 0;

			BBMessage = new CAnimation(Root, GS->LevelSX / 2, 200, NullObj, bb_message_params);
            animate = true;
            stopFrame = gframe;
            return;
}
