//List of sprites

CSprite* supersuicide;

void supersuicide_SCRIPT::InitGraphic()
{
   //Loading picture data
   CFile *f;

   f = GetAttachment("supersuicide.png");
   supersuicide = LoadSprite(f, 12, 0);

   //Let's override the sprites!

   SetSpriteOverride(375, supersuicide);
   SetSpriteOverride(376, supersuicide);
   SetSpriteOverride(377, supersuicide);
}
