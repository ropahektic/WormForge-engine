int GenerateWeaponID(int RNGn, int RNGnA, int RNGnB)
{
local NTurn = StartTurnCounter;
if(NTurn > 100){NTurn = 100;}
if(RNGnB <= ( NTurn * 0.5 ) )
	{
    return EndGameBonus[RNGnA % 3];
	}

if(RNGn <= 7)
	{
    return DropVeryRarely[(RNGnA % 11)];
    }
else if(RNGn <= 18)
	{
    return DropLess[RNGnA % 5];
    }
else if(RNGn <= 43)
	{
    return DropMore[RNGnA % 4];
    }
else{
    return DropAverage[RNGnA % 25];
    }
}

void CLemming::LookForFire()
{
	local Obje = Env->CheckCollision( this, PosX, PosY );
    if(Obje != NullObj)
		{
        if(Obje->ClType == OC_SeaBubble || Obje->ClType == OC_Smoke || Obje->ClType == OC_Gas)
			{
            OnFire = true;
            if(Fr < 9000)
				{
				OnFireFrame = Fr;
				}
			else
				{
                Fr = 0;
                OnFireFrame = 0;
                }
        	}
    	}
}

void CLemming::IsNotMoving()
{
    if(PosX == PosHistoryX && PosY == PosHistoryY)
		{
        StandingStill = true;
        }
    PosHistoryX = PosX;
    PosHistoryY = PosY;
}

bool CLemming::WaitSome(int amount, int stateat)
{
    if(stateat != StateM){WaitTimer = 0; WaitInEffect = false; return true;}
    WaitTimer++;

    if(WaitTimer > amount)
		{
        WaitTimer = 0; WaitInEffect = false;
        return true;
    	}
	else
		{
        WaitInEffect = true;
        return false;
    	}
}

void CLemming::PickRandomDir()
{
    if(RandomHundo > 50)
		{
        WalkingDir = 0;
    	}
	else
		{
        WalkingDir = 1;
    	}
}

bool CLemming::CheckCollision()
{
    local Cdir = 0;
    if(WalkingDir == 0){ Cdir = -1; }else{ Cdir = 1; }
    local Obje = Env->CheckCollision( this, PosX+Cdir, PosY-1 );
    if(Obje != NullObj)
		{
        if(Obje->ClType == OC_Landscape)
			{
            StateM = 2; //Collision
            return false;
        	}
    	}
  	return true;
}

void CLemming::WalkThere(){
    local Cdir = 0;
    if(WalkingDir == 0){ Cdir = -1; }else{ Cdir = 1; }
    if(InMidAir == 0){

        SpX = SpX+(Cdir* 0.5);
        SpY = SpY-3.0;

    }else{
        if(Fr % 100 == 0){
            if (RandomHundo > 60){
                SpY = SpY-5.0;
                SpX = SpX+(Cdir * 2.0);
            }
         }
    }
}
void CLemming::TurnAround(){
    if(WalkingDir == 0){
        WalkingDir = 1;
    }else{
        WalkingDir = 0;
    }
}
void CLemming::NeedsToGrow(){
    if(WantSize != AtSize){
        StartGrowingNow = true;
    }
}

void CLemming::NearbyWorm(){
local WormCloseby = false;
if(Root->CurTeam == 0 && Root->CurWorm == 0){return;}
                     for (j = 6; j >= 1; j-=1)
                      {
                        for (i = 8; i >= 1; i-=1)
                        {

                            local Worm = CWorm(Root->GetObject(2, 16 * j + i));
                            if (Worm)
                            {
                                if(     (abs(Worm->PosX - PosX) < 55)   && (abs(Worm->PosY - PosY) < 55)){
                                    WormCloseby = true;
                                    if (owningTeam == Worm->TeamSide){
                                        //TeamSide = 0; // 1 left, 2 right
                                        EnemyIsClose = false;
                                    }else{
                                        EnemyIsClose = true;
                                    }
                                    //GG->WriteToChat(9, "worms close", false);
                                    break;
                                }

                            };

                        };
                      };
WormsIsClose = WormCloseby;

if(WormsIsClose == false){
    EnemyIsClose = false;
}

return;
}

int CLemming::CheckCollisionGrow(){

    local Obje = Env->CheckCollision( this, PosX, PosY-2 );
    if(Obje != NullObj){
            if(Obje->ClType == OC_Landscape){
            return 0;
        }
    }
    local Obje = Env->CheckCollision( this, PosX, PosY+2 );
    if(Obje != NullObj){
            if(Obje->ClType == OC_Landscape){
            return 1;
        }
    }
    local Obje = Env->CheckCollision( this, PosX-2, PosY );
    if(Obje != NullObj){
            if(Obje->ClType == OC_Landscape){
            return 2;
        }
    }
    local Obje = Env->CheckCollision( this, PosX+2, PosY );
    if(Obje != NullObj){
            if(Obje->ClType == OC_Landscape){
            return 3;
        }
    }

  return 4;
}
void CLemming::MoveDirection(int dir){ //collision 0up1down 2left3right
   if(dir == 0){ //hit head, walk somewhere else
       StateM = 0;
   }
   if(dir == 1){
        SpY = SpY -3.0;
   }
   if(dir == 2){
        SpY = SpY -2.0;
        SpX = SpX +2.0;
   }
   if(dir == 3){
        SpY = SpY -2.0;
        SpX = SpX -2.0;
   }
}
void CLemming::DoGrow(){

    if(AtSize > 100){StateM = 1; return; }
    if(Fr % 4 == 0){
        if(AtSize < WantSize){
            AtSize = AtSize + 1;
            ColMask = new CColMask(AtSize,AtSize,MakeCircleMask(AtSize));
        }
        if(AtSize > WantSize){
            AtSize = AtSize + 1;
            ColMask = new CColMask(AtSize,AtSize,MakeCircleMask(AtSize));
        }
        if(AtSize == WantSize){
            StateM = 1; return;
        }
    }
}

        }

        if(Fr % 42){if(OnFire == false){ LookForFire(); }}

        if (Sinking == 1) Free(true);

        if(Fr % 500 == 0){NeedsToGrow();}

        if(Fr % 25 == 0){IsNotMoving();}else{if(PosX != PosHistoryX && PosY != PosHistoryY){ StandingStill = false; } }
        if(Fr % 102 == 0 ){ NearbyWorm();
        if(EnemyIsClose == true ){

                DoExplosion (  PosX,  PosY, 50, 20, 0, 0 );
            }else if(AtSize < 12){

            Free(true);
            }

        if(Fr % 28 == 0){ CheckAllObj(); }


        if(Fr % 200 == 0 && WaitInEffect == false){
            GetNextState = true;

        if(GetNextState == true){

        if(StateM == 0){ //standing / sleeping
                if(WaitSome(500, 0)){
