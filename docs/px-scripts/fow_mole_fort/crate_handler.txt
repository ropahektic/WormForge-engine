int MapEmptySpaceR[500];

CCrate* NewlyCrate;

int SMlastdistL;
int SMlastdistR;

void crate_handler::Init()
{
    RefreshWeapon = -1;
    CrCustomSpawnActive = false;
    CrMapGenerated = false;
    SpawnJetCrate = 1;
    mapsizeX = 0;
    mapsizeY = 0;
}

{
    super(parent, GS);
    col = 8; //maybe needed?

}

{
    PositionSet = false;
    NewlyCrate = this;

    super;

    PosX = -500;
    PosY = -500;
    X = -500;
    Y = -500;
}

bool GenerateCrSpawnMap(int side, int mSX, int mSY){

    if(NewlyCrate == NullObj){return false;}

    local CrSpawnUnderY = 305;
    local CrSpawnOverY = 1088; //not used?

    CrSearchGridSize = 50;
    local extraX = 0;
    local side2X = 0;
    if(mSX == 2200 && mSY == 1504)
		{
        CrSpawnUnderY = 305;
        CrSpawnOverY = 1088;
        ax = int(880 / CrSearchGridSize);
        ay = int(790 / CrSearchGridSize);
        side2X = 1325;
    	}
    if(mSX == 2200 && mSY == 2000)
		{
        CrSpawnUnderY = 713;
        CrSpawnOverY = 1500;
        ax = int(730 / CrSearchGridSize);
        ay = int(790 / CrSearchGridSize);
        extraX = 80;
        side2X = 1377;
    	}

    local startx = 0;
    local starty = CrSpawnUnderY;

    local dOut = 0;
    local DFbmp = DigFoam;

    SMlastdistL = 0;
    SMlastdistR = 0;
    for(iy = 0; iy < ay; iy++ )            	//int
		{
        for(ix = 0; ix < ax; ix++ )         //int
			{
            local cy = (iy * CrSearchGridSize) + starty;
            local cx = 0;
            if(side == 0){cx = (ix * CrSearchGridSize) + extraX;}
			else {cx = (ix * CrSearchGridSize) + side2X ;}

            local Obje = Env->CheckCollision( NewlyCrate, cx, cy );
            if(Obje != NullObj)
				{
                if(Obje->ClType == OC_Landscape)
					{
                    if(side == 0){MapEmptySpaceL[(iy * ax) + ix] = 0;} //not enough room
					else {MapEmptySpaceR[(iy * ax) + ix] = 0;}
                    continue;
                    }
				else if(Obje->ClType == OC_Crate)
					{
                    if(side == 0){MapEmptySpaceL[(iy * ax) + ix] = 0;} //spawning on another crate
                    else {MapEmptySpaceR[(iy * ax) + ix] = 0;}
                    continue;
                    }
                }

            local hitx1 = 0;
            local hity1 = 0;
            local hit1 = TraceLine(NewlyCrate, cx, cy, cx, CurrWaterLvl + 10, -1, &hitx1, &hity1);

            if (hit1 != NullObj)
                {
                local disty = 0;
                disty = abs(cy - hity1);
                if(disty > 100){disty = 100;}
                if(disty == 0){disty = 1;}

            	if(side == 0)
					{
                    dOut = SMlastdistL + disty;
                    SMlastdistL = dOut;
                    MapEmptySpaceL[(iy * ax) + ix] = dOut;
                    }
				else{
                    dOut = SMlastdistR + disty;
                    SMlastdistR = dOut;
                    MapEmptySpaceR[(iy * ax) + ix] = dOut;
                    }
                continue;
                }

			else
				{
                if(side == 0){MapEmptySpaceL[(iy * ax) + ix] = 0;} //water
                else {MapEmptySpaceR[(iy * ax) + ix] = 0;}
                continue;
                }
        }
    }

    CrMapGenerated = true;
    return true;
}

void GetNewCrateSpawn(int mSX, int mSY, int rngseedA)
{
    CrateSpawnSuccess = false;
    local RndSide = RandomInt(0,1);
    if (GenerateCrSpawnMap(RndSide, mSX, mSY) == false){CrateSpawnSuccess = false; return;}

    //pick a value between 1 and SMlastdist
    local Rtarget = 0;
    if(RndSide == 0){if (SMlastdistL!=0)Rtarget = rngseedA % SMlastdistL;}
    else{if (SMlastdistR!=0)Rtarget = rngseedA % SMlastdistR;}

    local lastIndex = 0;

    //Find closest value to target in array
    //Optimize ?
    for(int i = 0; i < (ax * ay); i++ )
		{
        local MapInt = 0;
        if(RndSide == 0){MapInt = MapEmptySpaceL[i];}
		else{MapInt = MapEmptySpaceR[i];}

        if(MapInt != 0)
			{
            if(MapInt < Rtarget)
				{
                lastIndex = i;
                CrateSpawnSuccess = true;
                continue;
				}
			else {break;}
            }
        }

   	if(mSX == 2200 & mSY == 1504)
	   	{
       	if(RndSide == 0 & ax!=0)
			{
        	NewCrX = ((lastIndex % ax) * CrSearchGridSize);
        	NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 305;
       		}
		else
			{
        	NewCrX = ((lastIndex % ax) * CrSearchGridSize) + 1325;
        	NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 305;
       		}
    	}

	if(mSX == 2200 & mSY == 2000)
		{
       	if(RndSide == 0 & ax!=0)
			{
        	NewCrX = ((lastIndex % ax) * CrSearchGridSize) + 80;
        	NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 713;
       		}
		else
			{
        	NewCrX = ((lastIndex % ax) * CrSearchGridSize) + 1377;
        	NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 713;
       		}
    	}
}

void CCrateHandler::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
	if (Type == M_FRAME)
    {
    if(mapsizeX < 100 || mapsizeY < 100)
		{
        mapsizeX = GS->LevelSX;
        mapsizeY = GS->LevelSY;
        }

	local RNGt = GS->GetRandom();
    local mzX = mapsizeX;
    local mzY = mapsizeY;
    if(NewlyCrate != NullObj && CrCustomSpawnActive == true)
		{
        if(NewlyCrate->FixedWeapon == 0)
			{
            if(NewlyCrate->PositionSet == false)
				{
                GetNewCrateSpawn(mzX, mzY, RNGt);
                if(CrateSpawnSuccess == true)
					{
                    NewlyCrate->PosX = NewCrX;
                    NewlyCrate->PosY = NewCrY;
                    NewlyCrate->X = NewCrX;
                    NewlyCrate->Y = NewCrY;
                    NewlyCrate->PositionSet = true;
                    NewlyCrate->LookAtMe(NewCrX, NewCrY, 5);
                    }
				else
					{
                    //could not spawn crate
                    NewlyCrate->PositionSet = true;
                    NewlyCrate->Freeing = true;
                    NewlyCrate->Free(true);
                    }
                }
            }
        }

        DelayRoutine = ( DelayRoutine + 1 ) % DelayRoutineSet;
        if(DelayRoutine == 0)
			{
            for(int n = 0; n < 66; n++)
				{
                if(n != 0 && // None
				   n != 3 && // Torch
				   n != 10 && // Earthquake
				   n != 18 && // Kamikaze
				   n != 26 && // Mole Bomb
				   n != 29 && // Far Sight
				   n != 32 && // Blowtorch
				   n != 33 && // Drill
				   n != 34 && // Girder
				   n != 37 && // Ninja Rope
				   n != 38 && // Bungee
				   n != 39 && // Parachute
				   n != 57 && // Skip Go
				   n != 58 && // Surrender
				   n != 59 && // Select Worm
				   n != 64) // Fast Walk
					{
                    twep = GS->GetWeaponByIndex(1, n);
                    
                    if(twep != NullObj)
						{
                        local invertTmz = false;

						//Bug from hell. This fixes it but i cant explain why. Weapons now go to right team.
                        if(GetTeamColor(1) == GetTeamColor(2))
							{
                            for(int m = 2; m <= 6; m++)
								{
                                if( GetTeamColor(m) != GetTeamColor(m + 1))
									{
                                    if(GetTeamColor(m) > GetTeamColor(m + 1)){invertTmz = true;}
                                    break;
                                    }
                                }
                            }

                        if(invertTmz)
							{
                            GS->Info.SetWeaponEx( 2, n, twep->GetPageIndex(), ( WepAmountL[n]) );
                            GS->Info.SetWeaponEx( 1, n, twep->GetPageIndex(), ( WepAmountR[n]) );
                        	}
						else
							{
                            GS->Info.SetWeaponEx( 1, n, twep->GetPageIndex(), ( WepAmountL[n]) );
                            GS->Info.SetWeaponEx( 2, n, twep->GetPageIndex(), ( WepAmountR[n]) );
                            }
                        }
                    }
                }
        	}
    }
}
