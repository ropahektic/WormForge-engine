void ListRootObjs()
{
     for (local i = 0; i < Root->Childs.Count; i++)
     {
           GG->WriteToChat( 2, StrCon5(classtoString(Root->Childs.Objs[i]->ClType), " storeIndex: ", itoa(Root->Childs.Objs[i]->StoreIndex), " storeType: ", itoa(Root->Childs.Objs[i]->StoreType)), false);
     }
}

void ListEnvObjs()
{
     for (local i = 0; i < Env->Objs.Count; i++)
     {
           GG->WriteToChat( 3, classtoString(Env->Objs.Objs[i]->ClType), false);
           
     }
}

void messagedatatoString(CMessageData * data)
{
     //local clr = RandomInt(1,8);
     for(int i = 0; i<16 ;i++)
     {
         local param = data->params[i];
         local fparam = data->fparams[i];
         if(param >= 65536*10)  param = param / 65536;
         if(fparam >= 65536*10)  fparam = fparam / 65536;
         
         GG->WriteToChat(7, StrCon5( StrCon("params[", itoa(i)), StrCon("] = ",itoa(param)), "   ",  StrCon("          fparams[", itoa(i)), StrCon("] = ",itoa(fparam))), false);
     }
     GG->WriteToChat( 1, "", false);
}
