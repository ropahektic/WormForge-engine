  isFalling = false;
  adjustCooldown = 0;   
  targetXP = planeTargetX;
  targetYP = planeTargetY;  
  super; 
  PlayLocalSound(52, 5, 1.0, 1.0);

bool hitPL;

void planeCrash::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{            
    if(Type == M_FRAME)
    { 
         if (hitPL == true){ SpY = SpY*1.56;  } //idk how to modify GravityFactor
         
         if(PosY >= Env->Water)
            {
                ExplodeAt(PosX, PosY);
                Free(true); 
            }          
        float dumbDistance = abs(PosX - targetXP);
        if (dumbDistance < 25) 
            {
                offsetPLANE = 0.33;
                GravityFactor = GravityFactor*1.2;    //why are u pointing at the air?
                adjustCooldown = 25;
                hit = true;
            }  
        // Anim  
        if (planeFly_anim) {                                             
            local sprite = WhiteplaneFlyS->Index;                    
            WeapSpriteParams* Params = new WeapSpriteParams;
            new WeapSprite(this, sprite, Params);       
            planeFly_anim = false;
        }                  
        if (!isFalling) 
        {
            float xDistanceP = abs(PosX - targetXP);
            if (xDistanceP < 500) 
            {
                isFalling = true;
                offsetPLANE = 0;
            } 
            else if (xDistanceP>500 && xDistanceP<1000 && !hitPL)
            {
            SpY = SpY*0.99;
            }    
            else if (xDistanceP>1000 && xDistanceP<1800 && !hitPL)
            {
            SpY = SpY*0.75;
            }  
            else if (xDistanceP>1800 && xDistanceP<2500 && !hitPL)
            {
            SpY = SpY*0.4;  
            }
            else if (xDistanceP>2500 && !hitPL)
            SpY = SpY*0.3;  
        }    
        
        // HOMING BEHAVIOR
        if (isFalling && adjustCooldown == 0)
        {
            float dirX = targetXP - PosX;
            float dirY = targetYP - PosY;
            float distanceToTarget = sqrt(dirX * dirX + dirY * dirY);
            
            if (distanceToTarget > 0)  // Avoid division by zero
            {
                dirX = dirX / distanceToTarget;
                dirY = dirY / distanceToTarget;
            }

            float missileSpeed = 18.5;
            float homingStrength = 0.35 - offsetPLANE;
            
            // 2. Calculate the ideal velocity (direction * speed)
            float requiredSpX = dirX * missileSpeed;
            float requiredSpY = dirY * missileSpeed;

            // 3. Gently steer the current velocity towards the ideal velocity
            SpX += (requiredSpX - SpX) * homingStrength;
            SpY += (requiredSpY - SpY) * homingStrength;

            adjustCooldown = 3;
        }
        else if (isFalling)
        {
            adjustCooldown--;
        }        
    } 
    super(sender, Type, MSize, MData);   
}
