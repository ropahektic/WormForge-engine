override void CWorm::FireFinal(CWeapon* Weap, CShootDesc* Desc)
{
   if (Weap->CheckName("Flame Turret"))
   {
     local sg = new CCustomTurret(GetObject(25, 0), PosX + 15 * TurnSide, PosY - 50);
     sg->turretTeam = WormTeam;
     sg->ammoPerTurn = 18;
     sg->weaptype = 13;
//   sg->clearMines = true;
//     sg->FNum = 3;
     sg->admg = 25;
//   sg->ExplTimer = 1000;
//     sg->health = 60;
//   sg->minimalRadius = 50;
//   sg->sensorRadius = 400;
//   sg->lostRadius = sg->sensorRadius + 100;
     sg->ExpDamage = 45;
//     sg->maxHealth = sg->health;
     sg->sprNum = 126;
//   sg->breakable = true;
//   sg->ShootSY = FireAngle;
//   sg->ShootSX = FirePower * TurnSide;
     sg->towerSprite = customturretTowerF;
     sg->towerOffset = 19;
     vol = CalculateSoundVolume(PosX,PosY);
     pan = CalculateSoundPan(PosX,PosY);
     turretsetup->Play(vol, pan, false);
   } else {
     super;
   };
};
