
WormSprite* homingClusterWS;
CSprite* homing_cluster_aiming;                                                                                                
                                                                                 
void homing_cluster_anim::InitGraphic() {
    CFile* f;                                                                    
    f = GetAttachment("homing_cluster_bullet.png");                             
    homing_cluster_aiming = LoadSprite(f, 1, 0);                                                            
}                                                                                
                                                                                 
override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) { 
 super;                                                                          
  if (Type == M_FRAME) {                                                         
    if (gframe == 1) {                                                           
     local weapName; local sprite;

     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Homing Fire Cluster";                                               
     sprite = homing_cluster_aiming->Index;                                     
     Params->hand_radius = 17.0;                                                         
     Params->weap_rotation =  -0.5;
     Params->weap_radius = 9.0;
     Params->hand_type = 1;
     Params->hand_scale = 0.8;               ///// I recommend don`t change it. //////   //                                                         
                                                    
     homingClusterWS = new WormSprite(weapName, sprite, Params);
    }                                                                            
  }                                                                              
}                                                                                

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;                                                                           
  if (Type == M_FRAME){                                                          
     if (weap->CheckName("Homing Fire Cluster")) {                                
         if (homing_cluster_anim2) {                                              
            local sprite = homing_cluster_aiming->Index;                      
            WeapSpriteParams* Params = new WeapSpriteParams;
            new WeapSprite(this, sprite, Params);         
            weap->launch.anim.spriteIndex = 0;
            homing_cluster_anim2 = false;                                         
         } 
     }                                                                                                                                                                                                                          
  }                                                                              
}                                                                                
override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;                                                                           
  homing_cluster_anim2 = true;                                                    
}
