     ldata->launch.explodeOnSpace = true;

override void CMissile::ExplodeAt(fixed x,fixed y)
{
    super;
    if (weap->CheckName("Chilli Bomb"))
    {
        ShotClustlets(x, y, 8000, 3);
    }
    if (weap->CheckName("Fire Cluster Bomb"))
    {
        ShotClustlets(x, y, 1000, 1);
    }
    if (weap->CheckName("Digging Cluster Bomb"))
    {
        local bah = this;
        if (bah->imCheckMis != this) return;
        local w = GetCurrentWorm();
        if (w != NullObj)
        {
        DigClustlets(bah->PosX, bah->PosY);
        }
        else if (bah != NullObj) DigClustlets(bah->PosX, bah->PosY);
    }
    if (weap->CheckName("Homing Fire Cluster"))
    {
        if (imCheckMis->launchdata.explosion.damage > 100) imCheckMis->launchdata.explosion.damage = 50;
        if (imCheckMis != this) {
           DoExplosion( x, y, 50, 35, 0, ClWT);
           return;
        }
        ShotClustlets(x, y, 1000, 2);
    }
    if (weap->CheckName("Homing Digger"))
    {
        local HDC = this;
        if (imCheckMis != this) {
           DoExplosion( x, y, 50, 35, 0, ClWT);
           return;
        }
        local w = GetCurrentWorm();
        if (w != NullObj)
        {
          if (HDC != NullObj && HDC->PosX != w->PosX && HDC->PosY != w->PosY)
          {
            HomeDiggingClustlets(x, y, HDC->SpX, HDC->SpY);
          }
        }
        else if (HDC != NullObj)
        {
            HomeDiggingClustlets(x, y, HDC->SpX, HDC->SpY);
        }
    }
}




override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_FRAME)
    {
        if (weap->CheckName("Fire Cluster Bomb") ||
			weap->CheckName("Homing Fire Cluster") ||
			weap->CheckName("Homing Digger") ||
			weap->CheckName("Chilli Bomb") ||
			weap->CheckName("Digging Cluster"))
        {
            imCheckMis = this;
            if (not_too_much_particles) do_fire_trail_effects();
        }
        
        if (weap->CheckName("Digging Cluster Bomb"))
        {
            imCheckMis = this;
            if (not_too_much_particles) do_sprite_trail_effects(159, PosX, PosY);
        }
        if (weap->CheckName("Homing Digger"))
        {
            imCheckMis = this;
            if (not_too_much_particles) do_sprite_trail_effects(159, PosX, PosY);
        }
                         
    }                                    
    super;
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
    if (Weap->CheckName("Chilli Bomb"))
    {
        ClWT = WormTeam;
        ClWN = WormNumber;
        CheckParticlesNumber();
    }
    if (Weap->CheckName("Fire Cluster Bomb"))
    {
        ClWT = WormTeam;
        ClWN = WormNumber;
        CheckParticlesNumber();
    }
    if (Weap->CheckName("Digging Cluster Bomb"))
    {
        CheckParticlesNumber();
        Desc->Y -= 10;
        ClWT = WormTeam;
        ClWN = WormNumber;
    }
    if (Weap->CheckName("Homing Fire Cluster"))
    {
        ClWT = WormTeam;
        ClWN = WormNumber;
        CheckParticlesNumber();
    }
    if (Weap->CheckName("Homing Digger"))
    {
        ClWT = WormTeam;
        ClWN = WormNumber;
        CheckParticlesNumber();
    }
    super;
}
