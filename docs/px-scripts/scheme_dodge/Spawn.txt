$editor
 $object WormsSP
  $origin -13 -15
  $image "edspawn.png"
  $func SpawnPoint
  $radius 17
 $object_end 
$end


void SpawnPoint(CEditorObject* obj)
{
  x = obj->PosX;
  y = obj->PosY;

  for (j = 1; j < 7; j++) //team
  {
    for (i = 1; i < 9; i++)     //worm
    {
        local Worm = CWorm(Root->GetObject(2, 16 * j + i));
        if (Worm != NullObj)
        {
          Worm->PosX = x;
          Worm->PosY = y;
          x = x + 5;
        };
    };
   };
};
