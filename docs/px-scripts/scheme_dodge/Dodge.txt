override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
super;
      if(Type == M_FRAME)
      {
              if(GetCurrentWorm() != NullObj)
              {
                  if(GetCurrentWorm() == this && DodgeStarted == true && GotDamage != 0)
                  {
                  local str = MakeNumberString("Wave: ", CurWave,"");
                  local str2 = MakeNumberString(" ,Overall Frames: ", Timer+((CurWave-1)*600),"");
                  GG->WriteToChat( 1, StrCon(str, str2), false);
                  PosX = 865;              
                  PosY = 525;
                  DodgeStarted = false;
                  WormWave = CurWave;
                  if(WormWave > 23){ WormWave = 23;}
                  CurWave = 0;
                  Timer = 0;
                  AddTimer = 0;
                  RareTimer = 0;
                  mtp = 100;
                  swch = 0;
                  swchdir = 1; //-1 up +1 down
                  Boos = false;
                  LastWave = false;
                  BossKilled = false;
                  }
                  
                  if(JetFuel < 100){JetFuel = 255;}
              }
      }
}

void ShootZook(float px, float py, float sp, string wepname)
{

  zero(&sdesc);

  sdesc.SpX = sp;         
  sdesc.SpY = 0;  
  local ldata;
  ldata = GetWeaponByName(wepname); 
  local msl = new CMissile(Root->GetObject(25, 0), &ldata->launch, &sdesc);
  msl->GravityFactor = 0;
                     if(StrCmp(wepname, "Sheep"))
                     {
                     msl->Side = -1;
                               if(CurWave == 8 || CurWave == 18)
                               {
                               msl->SuperSheep = true;
                               }
                     }
                     
                     if(StrCmp(wepname, "SuperSheep"))
                     {
                     msl->SpY = -2;
                     return;
                     }
                     
                     if(StrCmp(wepname, "Mortar"))
                     {
                     msl->Random = true;
                     }
                     
                     if(StrCmp(wepname, "Bullet"))
                     {
                     msl->Special = true;
                     }
                     
                     if(StrCmp(wepname, "Plane"))
                     {
                     msl->Plane = true;
                     }
                     
                     if(StrCmp(wepname, "SRocket"))
                     {
                     msl->SpY = 2;
                     return;
                     }
                     
                     if(StrCmp(wepname, "Bazooka") && LastWave == true)
                     {
                     msl->BossKilling = true;
                     }
  msl->SpY = 0;       
}
