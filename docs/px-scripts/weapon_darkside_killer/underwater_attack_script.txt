{
    if (Type == M_FRAME)
    {
        if (CurWeapon->CheckName("Underwater Attack") && ObjState == WS_AIMING)
        {
                 if (FireAngle < 0.875) FireAngle = 0.875;
                 if (FireAngle > 1.000) FireAngle = 1.000;
        }

        if (worm != NullObj)
        {
        if (DarksideKillerUA)
        {
            local hutick2 = gframe-hutickUA;
            DoItUA();
            if (DoingIt == false)
            {
                 Missile_HTargetUA->PosX = UAtargetX0 + RandomInt(-hutick2, hutick2)*1.5;
                 Missile_HTargetUA->PosY = UAtargetY0 - RandomInt(0, hutick2) - 100;
            }
            if (UAmX >= 1){
                 if (hutick2 >= 250)
                 {
                     Missile_HTargetUA->PosX = UAtargetX0;
                     Missile_HTargetUA->PosY = UAtargetY0;
                 } else
                 if (hutick2 >= 200)
                 {
                     Missile_HTargetUA->PosX = UAtargetX0;
                     Missile_HTargetUA->PosY = GS->env->Water + 100;
                 } else
                 if (hutick2 >= 135)
                 {
                     Missile_HTargetUA->PosX = GS->LevelSX + 100;
                     Missile_HTargetUA->PosY = GS->env->Water + 50;
                 } else
                 if (hutick2 >= 40)
                 {
                     Missile_HTargetUA->PosX = GS->LevelSX + 250;
                     Missile_HTargetUA->PosY = -250;
                 }
            } else {
                 if (hutick2 >= 250)
                 {
                     Missile_HTargetUA->PosX = UAtargetX0;
                     Missile_HTargetUA->PosY = UAtargetY0;
                 } else
                 if (hutick2 >= 200)
                 {
                     Missile_HTargetUA->PosX = UAtargetX0;
                     Missile_HTargetUA->PosY = GS->env->Water + 100;
                 } else
                 if (hutick2 >= 135)
                 {
                     Missile_HTargetUA->PosX = -100;
                     Missile_HTargetUA->PosY = GS->env->Water + 50;
                 } else
                 if (hutick2 >= 40)
                 {
                     Missile_HTargetUA->PosX = -250;
                     Missile_HTargetUA->PosY = -250;
                 }
            }
        }
        if (Root->Timer < 500 && DarksideKillerUA)
        {
            Missile_HTargetUA->PosX = RandomInt(0, GS->LevelSX);
            Missile_HTargetUA->PosY = GS->LevelSY;
            DarksideKillerUA = false;
        }
        } else {
            DoingItUA = false;
            DarksideKillerUA = false;
        }
    }
super;
}
