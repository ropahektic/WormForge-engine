override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_FRAME)
    {
        if ((CurWeapon->CheckName("Darkside Killer") || CurWeapon->CheckName("Underwater Attack")) && ObjState == WS_AIMING)
        {
                 if (FireAngle < 0.875) FireAngle = 0.875;
                 if (FireAngle > 1.000) FireAngle = 1.000;
        }
        local worm = GetCurrentWorm();
        if (worm != NullObj)
        {
            if (WDarksideKiller)
            {
                DoIt();
                WDarksideKiller = false;
            }
            if (WDarksideKillerUA)
            {
                DoIt();
                WDarksideKillerUA = false;
            }
            if (makeShake)
            {
                GS->ShakeX = DKshake;
                GS->ShakeY = DKshake;
                DKshake = DKshake * 0.85;
                if (DKshake < 0.1)
                {
                    DKshake = 0;
                    GS->ShakeX = 0;
                    GS->ShakeY = 0;
                    makeShake = false;
                }
            }
        } else if (DKshake > 0){
            GS->ShakeX = 0;
            GS->ShakeY = 0;
        }
   }
super;
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
    if (Weap->CheckName("Darkside Killer"))
    {
        PlaySound(59, 10, 0.7, 0.7);
        PlaySound(59, 10, 0.7, 0.7);
        PlaySound(59, 10, 0.7, 0.7);
        DKtargetX0 = Desc->AddX;
        DKtargetY0 = Desc->AddY;
        WDarksideKiller = true;
        DoingIt = true;
        DKhutick = gframe;
        makeShake = true;
        DKshake = 20;
        return;
    }
    if (Weap->CheckName("Underwater Attack"))
    {
        PlaySound(69, 10, 0.7, 0.7);
        PlaySound(69, 10, 0.7, 0.7);
        UAtargetX0 = Desc->AddX;
        UAtargetY0 = Desc->AddY;
        WDarksideKillerUA = true;
        DoingIt = true;
        UAhutick = gframe;
        makeShake = true;
        DKshake = 20;
        return;
    } 
super;
}
