require utils, p_sprite_builder, pxparticles, sprite_curtis_250, sprite_curtis_251;

CSprite* graved_aiming;
CSprite* sgraved_aiming;
CSprite* sgravedc;
CSprite* sgravedcda;
CSprite* sgravedcdb;
CSprite* sgravedcdc;

CMissile* dmis_graved;
bool graved;

void gravedigger_script::InitGraphic()
{
    CFile *f;
    f = GetAttachment("graved.png");
    graved_aiming = LoadSprite(f, 12, 0);
    f = GetAttachment("sgraved.png");
    sgraved_aiming = LoadSprite(f, 12, 0);
    f = GetAttachment("sgravedc.png");
    sgravedc = LoadSprite(f, 12, 0);
    f = GetAttachment("sgravedcda.png");
    sgravedcda = LoadSprite(f, 1, 0);
    f = GetAttachment("sgravedcdb.png");
    sgravedcdb = LoadSprite(f, 1, 0);
    f = GetAttachment("sgravedcdc.png");
    sgravedcdc = LoadSprite(f, 1, 0);
    
    SetSpriteOverride(255, sgravedc);
    SetSpriteOverride(256, sgravedcda);
    SetSpriteOverride(257, sgravedcdb);
    SetSpriteOverride(258, sgravedcdc);
}

//---------------------------------GRAVEDIGGER (DIG)--------------------------//

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) {
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Gravedigger";
     sprite = graved_aiming->Index;
     Params->hand_type = 1;
     Params->hand_scale = 1.3;
     Params->hand_radius = -22.0;
     Params->hand_rotation = -0.0;
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;
     Params->weap_rotation = 0.5;
     Params->weap_radius = -14.5;
     Params->animate = true;
     Params->anim_speed = 1.10;
     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;
  if (Type == M_FRAME){
     if (weap->CheckName("Gravedigger")) {
         if (gravedigger_anim) {
              local sprite = graved_aiming->Index;
              WeapSpriteParams* Params = new WeapSpriteParams;
              Params->animate = true;
              Params->anim_speed = 1.10;
              new WeapSprite(this, sprite, Params);
              weap->launch.anim.spriteIndex = 0;
              gravedigger_anim = false;
         }
     }
  }
}
override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;
  gravedigger_anim = true;
}

//---------------PARTICLE FX---------------//

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    super;
    if (Type == M_FRAME) {
        if (CurWeapon->CheckName("Gravedigger")) {
            if (ObjState == WS_AIMING || ObjState == WS_SETPOWER)
            DoAnimParticleEffects(

            249,     //   SPRITE
            1,       //   RADIUS
            -28,     //   RELX
            -20,     //   RELY
            70,      //   LIFETIME
            0.02,    //   SPEED
            1.0,     //   VELO.MIN
            4.0,     //   VELO.MAX
            255,     //   START.ALPHA
            255,     //   END.ALPHA
            0,       //   WIND
            0.50,    //   START.SIZE
            0.01,    //   END.SIZE
            20.00,   //   ANGLE.MIN
            20.00,   //   ANGLE.MAX
            0.0,     //   RANDOMNESS
            1.5);    //   GRAVITY
        }
    }
}

//-----------------------------------DIGGING----------------------------------//

override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata)
{
    super;
    Sdigging = false;
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_FRAME)
    {
        if (weap->CheckName("Gravedigger") || weap->CheckName("Super Gravedigger"))
        {
          if (graved)
          {
            dmis_graved = this; dmis_graved->Sdigging = true; graved = true;
          }
        }

        if (dmis_graved == this && Sdigging)
        {
            GG->land->MakeHole(13,dmis_graved->PosX,dmis_graved->PosY-0.0);
        }

    }

    super;
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
    if (Weap->CheckName("Gravedigger") || Weap->CheckName("Super Gravedigger"))
    {
        graved = true;
    }

    super;
}
