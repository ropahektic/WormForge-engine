void bl_script::FirstFrame()
{               
        keySync->AddKey(49); // 1 button
        keySync->AddKey(50); // 2 button
        keySync->AddKey(51); // 3 button
        keySync->AddKey(52); // 4 button
        keySync->AddKey(53); // 5 button
        
        f = GetAttachment("tazsnd.wav");	 tazsnd = new CSoundFile(f);                
}

override void CWorm::Message(CObject *sender, EMType Type, int mSize, CMessageData *mData)
{  
   if(Type == M_TURNBEGIN || Type == M_TURNEND)
   {
   barrelselector = true;
   barreltypebl = 0;
   }
      
   if(Type == M_FRAME)
   {
      if(CurWeapon->CheckName("Barrel Launcher") && barrelselector != false)
      {         
         local Team = CTeam(GetObject(21, WormTeam));
         Team->AllowSelectBounce = false;
                  
         if(keySync->KeyPressedNow(49))
         {         
         barreltypebl = 0;
         ShowMessage("Napalm Barrel");
         PlayLocalSound(59, 1.0, 1.0, 1.0);
         }          
         else if(keySync->KeyPressedNow(50))
         {                  
         barreltypebl = 1;
         ShowMessage("Mines Barrel");
         PlayLocalSound(88, 1.0, 1.0, 1.0);
         }                  
         else if(keySync->KeyPressedNow(51))
         {         
         barreltypebl = 2;
         ShowMessage("Electric Barrel");
         tazsnd->Play(1.0, 0.0, false);
         }
         else if(keySync->KeyPressedNow(52))
         {         
         barreltypebl = 3;
         ShowMessage("Ice Barrel");
         PlayLocalSound(24, 1.0, 1.0, 1.0);
         }
         else if(keySync->KeyPressedNow(53))
         {                  
         barreltypebl = 4;
         ShowMessage("Teleport Barrel");
         PlayLocalSound(67, 1.0, 1.0, 1.0);         
         }          
      }
   }                           
super;
}

void MinesDrum::Explosion()
{
CMineParams MParams;
            CShootDesc SDesc;
            int nmines = 5;
            for(local i=0; i<nmines; i+=1)  // default 10
            { 
              zero(&MParams);
              zero(&SDesc);
                          
              MParams.Radius = 48;
              MParams.Prefuse = 3000;
              MParams.Fuse = 3000;
              MParams.Flags = fflags; // default 6
              MParams.Bias = 0;
              MParams.Damage = 40;  // default 48
              MParams.BlastPower = 100;
              
              SDesc.Team = 0;
              SDesc.X = PosX;
              SDesc.Y = PosY;
              SDesc.SpX = RandomFloat(-7.0, 7.0);
              SDesc.SpY = RandomFloat(-7.0, -4.0);
        
              local Mine = new CMine(Root->GetObject(25, 0), &MParams, &SDesc, false, 0);                 
            }
            DoExplosion(PosX, PosY, 100, 10, 0, OwnerTeam);
}

{

           for(int i=0; i<400;i++)

               p = new PxParticle(611, PosX, PosY);
               
               p->SetLifeTime(70);
               p->SetMotionRandomness(0.4);
               p->SetAirResistance(0.02);

               p->SetStartSize(s, s);

               p->SetEndAlpha(175); //0

}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
if (Weap->CheckName("Barrel Launcher"))
    {
    if (barrelselector == true)
        {
        if (barreltypebl == 0){new NapalmDrum(GetObject(25, 0), Desc, this); return;}
        else if (barreltypebl == 1){new MinesDrum(GetObject(25, 0), Desc, this); return;}
        else if (barreltypebl == 2){new ElectroDrum(GetObject(25, 0), Desc, this); return;}
        else if (barreltypebl == 3){new IceDrum(GetObject(25, 0), Desc, this); return;}
        else if (barreltypebl == 4){new TeleportDrum(GetObject(25, 0), Desc, this); return;}
        barrelselector = false;                                                                                                                                
        }
	}
        
else if (Weap->CheckName("Normal Barrel"))
    {
    new MovingOilDrum(Root->GetObject(25, 0), Desc, this); return;
    }
        
else
	{
	super;
	}
}

CTazerShot::CTazerShot(CObject *parent, ElectroDrum *drum)
{
	super(parent, GS);
	ClType = OC_AirStrike;
	
	ClearIgnoreList();
	
    ang = MATH_HALF_PI;
	
	LifeTime = TazerLifeTime;
	
	Removed = false;
	Dead = false;
	
	DieFraction = 0.0;
	DieRate = 0.05;
	
	DirX = sin(ang);
	DirY = -cos(ang);
	PosX = drum->PosX + 4 * DirX;
	PosY = drum->PosY + 4 * DirY;
	
	LastX = drum->PosX;
	LastY = drum->PosY;
	
	NextProgress = 0;
	StopProgress = false;
	
	NumRemoveOnNextLink = 0;
	
	ClearPoints();
	AddPoint(PosX, PosY);
	
	SoundObj = tazerLoopSound->Play(1.0, 0.0, false);
}
