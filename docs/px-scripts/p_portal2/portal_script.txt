CPortal::CPortal(int type, int x, int y, int sizeX, int sizeY, int TurnSide, int spy, bool ExitOnly, bool LandTele, string linkedTo, string name, bool trap, int ang_type)
{
  super(Root, GS);
  
  p_mask = portal_mask_50;
  
  teleMissiles = true;
  teleWorms = true;
  teleMines = true;
  teleCrates = true;
  teleDrums = true;
  teleBowl = true;
  teleArrow = true;
  teleCross = true;
  teleMagnets = true;
  
  timer_to_deactivate = 150;
  
  memX = 0;
  memY = 0;

  isBlue = false;
  isOrange = false;
  if (type == 1) isOrange = true;
  if (type == 2) isBlue = true;
  
  Disabled = false;
  Px = x;
  Py = y;
  pr_x1 = x - sizeX/2;
  pr_x2 = x + sizeX/2;
  pr_y1 = y - sizeY/2;
  pr_y2 = y + sizeY/2;
  
  xSize = sizeX;
  ySize = sizeY;
  
  spx1 = 0;
  spy1 = 0;
  
////////////////////////////////////  
  portal_angle_type = ang_type;
  
  isSide = false;
  isRight = false;
  isLeft = false;
  isBottom = false;
  isTopRight = false;
  isTopLeft = false;
  isBottomRight = false;
  isBottomLeft = false;
  isAngled = false;
  isTop = false;
  isUP = false;
  if (portal_angle_type == 3) {             // was : 4 3 2 1 5 6 7 8
      isRight = true; isSide = true;
      spx1 = 1; spy1 = 0;
  }
  else if (portal_angle_type == 7){
      isLeft = true; isSide = true;
      spx1 = -1; spy1 = 0;
  }
  else if (portal_angle_type == 5) { 
      isBottom = true; isUP = true;
      spx1 = 0; spy1 = -1;
  }
  else if (portal_angle_type == 1) { 
      isTop = true; isUP = true;
      spx1 = 0; spy1 = 1;
  }
  else if (portal_angle_type == 2) { 
      isTopRight = true; isAngled = true;
      spx1 = -1; spy1 = 1;
  }
  else if (portal_angle_type == 6) { 
      isBottomLeft = true; isAngled = true;
      spx1 = 1; spy1 = -1;
  }
  else if (portal_angle_type == 8) { 
      isTopLeft = true; isAngled = true;
      spx1 = 1; spy1 = 1;
  }
  else if (portal_angle_type == 4) { 
      isBottomRight = true; isAngled = true;
      spx1 = -1; spy1 = -1;
  }
  else 
  {
    isUp = true;                                        // alternative rough reading, but script will never get this place :D
    if (TurnSide > 0) {
      isRight = true; isLeft = false;
    }
    else {
      isRight = false; isLeft = true;
    }
    if (spy > 15) {
      isUP = true; isBottom = true; isTop = false;
    }
    else if (spy < -15) {
      isUP = true; isBottom = false; isTop = true;
    }
    else {
      isSide = true; 
      isUP = false;
    }
  }   
////////////////////////////////////
    
  isExit = ExitOnly;
  tpMissiles = true; 
  trapTeleport = trap;  
  landTeleport = LandTele;
  // 1 - top
  // 2 - bottom
  // 3 - left
  // 4 - right 
  
  if (isExit) landTeleport = false;
                                                           // Setting point of teleportation (to be out of wall)
  if (isSide)
  {
    if (isRight) tp_x = int(pr_x1);
    if (isLeft) tp_x = int(pr_x2);
  } else tp_x = int(pr_x1 + (pr_x2 - pr_x1) / 2);
  
  if (isUP)
  { 
    if (isBottom) tp_y = int(pr_y1);  // 5 - let`s call it "gravity speed factor". re really need it to be more top :P
    if (isTop) tp_y = int(pr_y2);
  } else tp_y = int(pr_y1 + (pr_y2 - pr_y1) / 2);
  
  if (isAngled)
  {
    if (isTopRight)   // x: min, y: 2/3
    {
      tp_x = int(pr_x1 + 5);
      tp_y = int(pr_y1 + (pr_y2 - pr_y1) / 3 * 2);
    }
    if (isTopLeft)   // x: max, y: 2/3
    {
      tp_x = int(pr_x1 + (pr_x2 - pr_x1) - 5);
      tp_y = int(pr_y1 + (pr_y2 - pr_y1) / 3 * 2);
    }
    if (isBottomRight)   // x: min, y: 1/3
    {
      tp_x = int(pr_x1 + 5);
      tp_y = int(pr_y1 + (pr_y2 - pr_y1) / 3);
    }
    if (isBottomLeft)   // x: max, y: 1/3
    {
      tp_x = int(pr_x1 + (pr_x2 - pr_x1) - 5);
      tp_y = int(pr_y1 + (pr_y2 - pr_y1) / 3);
    }
  } else {
      tp_x = int(pr_x1 + (pr_x2 - pr_x1) / 2);
      tp_y = int(pr_y1 + (pr_y2 - pr_y1) / 2);
  } 
  timer = 0;
  waiting = false;
  active = true;
  start_be_inactive = false;
  active_timer = 0;
  LinkedTo = linkedTo;
  t_name = StrSave(name);
  linked = CPortal(NullObj);
  
  timer2 = 0;
  anim_fps = 500;
  
  if (landTeleport)
  {
    touchSensor = new CTouchSensor(this, pr_x1, pr_y1, pr_x2 - pr_x1, pr_y2 - pr_y1);
  }
};

void CPortal::TeleportObject(CGObject* obj)
{ 
if (Disabled == false && active)
{              
 if (obj->ColMask->Check(linked->tp_x, linked->tp_y, CGObject(Env->Objs.Objs[0])->ColMask, 0, 0))
 {
    GG->land->MakeHole( 15, linked->tp_x, linked->tp_y);  // this can happen in veeeery limited situations 
//    GG->land->ApplyMask( 15, linked->tp_x, linked->tp_y);
 }
 if (obj != NullObj)
 {
 if (obj->ClType == OC_Worm) 
 {
  waiting = true;
  linked->waiting = true;
        
  local worm = CWorm(obj);
  local prepare_state;
  
  if (worm->ObjState == WS_WALKING) worm->SetState(WS_IDLE);    
  if (worm->ObjState == WS_ROPING) worm->SetState(WS_AFTERROPE);
//  if (worm->ObjState == WS_BLOWTORCH) {worm->SetState(WS_IDLE); prepare_state = 1;}    // seems glitchy.
//  if (worm->ObjState == WS_DRILLING) {worm->SetState(WS_IDLE); prepare_state = 2;}
        
  worm->PosX = linked->tp_x;
  worm->PosY = linked->tp_y;
  memX = worm->SpX;
  memY = worm->SpY;
  
  worm->portal_just_teleported = true;
  
  if (prepare_state != 0)
  {
    if (prepare_state == 1) 
    {
      worm->SetState(WS_BLOWTORCH);
    }
    else if (prepare_state == 2) 
    {
      worm->SetState(WS_DRILLING);
    }
    prepare_state = 0;
  }
///////////////////////////////////////////////////////////////////
////////    Here is where Magic begins ;D    //////////////////////
///////////////////////////////////////////////////////////////////  
  if (isSide)
  {
    if (isRight)                                                                //  Right, worm
    {
        if (isRight == linked->isRight)
        {
            worm->SpX = -memX;
            if (worm->SpX > 2) worm->SpX = -worm->SpX;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = -2;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isRight == linked->isLeft)
        {
            worm->SpX = memX;
            if (worm->SpX < 2) worm->SpX = -worm->SpX;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isRight == linked->isBottom)
        {
            worm->SpX = memY;
            worm->SpY = memX;
            if(worm->SpY > 2) worm->SpY = -worm->SpY;
            else if (worm->SpY > -2) worm->SpX = -2;
        }
        else if (isRight == linked->isTop)
        {
            worm->SpX = memY;
            worm->SpY = memX;
            if(worm->SpY < -2) worm->SpY = -worm->SpY;
            else if(worm->SpY < 2) worm->SpY = 2;
        }
        else if (isRight == linked->isTopRight)
        {
            worm->SpX = -memX / 2;
            worm->SpY = -memY + memX / 2;
            if(worm->SpX > 0) worm->SpX = -worm->SpX;
            if(worm->SpY < 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isRight == linked->isTopLeft)
        {
            worm->SpX = memX / 2;
            worm->SpY = memY + memX / 2;
            if(worm->SpX < 0) worm->SpX = -worm->SpX;
            if(worm->SpY < 0) worm->SpY = -worm->SpY;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isRight == linked->isBottomRight)
        {
            worm->SpX = -memX / 2 + memY;
            worm->SpY = -memX;
            if(worm->SpX > 0) worm->SpX = -worm->SpX;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isRight == linked->isBottomLeft)
        {
            worm->SpX = -memX / 2;
            worm->SpY = -memY - memX / 2;
            if(worm->SpX < 0) worm->SpX = -worm->SpX;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
    }
    else if (isLeft)                                                               // Left, worm
    {
        if (isLeft == linked->isLeft) 
        {   
            worm->SpX = memX;
            if (worm->SpX < 0) worm->SpX = -worm->SpX;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isLeft == linked->isRight) 
        {   
            worm->SpX = memX;
            if (worm->SpX > 2) worm->SpX = -worm->SpX;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = -2;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isLeft == linked->isBottom)
        {
            worm->SpX = memY;
            worm->SpY = memX;
            if (worm->SpY > 0) worm->SpY = -worm->SpY;
        }
        else if (isLeft == linked->isTop)
        {
            worm->SpX = memY;
            worm->SpY = memX;
            if(worm->SpY < 0) worm->SpY = -worm->SpY;
        }
        else if (isLeft == linked->isTopRight)
        {
            worm->SpX = memX / 2;
            worm->SpY = memY + memX / 2;
            if(worm->SpX > 0) worm->SpX = -worm->SpX;
            if(worm->SpY < 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isLeft == linked->isTopLeft)
        {
            worm->SpX = -memX / 2;
            worm->SpY = memY - memX / 2;
            if(worm->SpX < 0) worm->SpX = -worm->SpX;
            if(worm->SpY < 0) worm->SpY = -worm->SpY;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isLeft == linked->isBottomRight)
        {
            worm->SpX = memX / 2;
            worm->SpY = -memY + memX / 2;
            if(worm->SpX > 0) worm->SpX = -worm->SpX;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isLeft == linked->isBottomLeft)
        {
            worm->SpX = memX / 2;
            worm->SpY = -memY - memX / 2;
            if(worm->SpX < 0) worm->SpX = -worm->SpX;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
    }
  }
  else if (isUP)
  {
    if (isBottom)                                                                   // Bottom, worm
    {
        if (isBottom == linked->isBottom)
        {
            worm->SpY = -memY;
            if (worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->SpY > -2 && worm->SpY < 0) worm->SpY = -2;
        }
        else if (isBottom == linked->isTop)
        {
            if (worm->SpY < 0) worm->SpY = -worm->SpY;
            
        }
        else if (isBottom == linked->isRight)
        {
            worm->SpX = -memY;
            worm->SpY = -memX;
            if (worm->SpX > 0) worm->SpX = -worm->SpX;
            if (worm->SpX > -2) worm->SpX = -2;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = -2;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isBottom == linked->isLeft)
        {
            worm->SpX = -memY;
            worm->SpY = -memX;
            if (worm->SpX < 0) worm->SpX = -worm->SpX;
            if (worm->SpX < 2 && worm->SpX > 0) worm->SpX = 2;
            if (worm->SpY < 2 && worm->SpY > 0) worm->SpY = 2;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isBottom == linked->isTopRight)
        {
            worm->SpX = -memX + memY / 2;
            worm->SpY = memY / 2;
            if (worm->SpX > 0) worm->SpX = -worm->SpX;
            if (worm->SpX > -2) worm->SpX = -2;
            if (worm->SpY < 0) worm->SpY = -worm->SpY;
            if (worm->SpY > 0 && worm->SpY < 2) worm->SpY = 2;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isBottom == linked->isTopLeft)
        {
            worm->SpX = -memX - memY / 2;
            worm->SpY = -memY / 2;
            if (worm->SpX < 0) worm->SpX = -worm->SpX;
            if (worm->SpX < 2) worm->SpX = 2;
            if (worm->SpY < 0) worm->SpY = -worm->SpY;
            if (worm->SpY < 2 && worm->SpY > 0) worm->SpY = 2;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isBottom == linked->isBottomRight)
        {
            worm->SpX = -memX + memY / 2;
            worm->SpY = memY / 2;
            if (worm->SpX > 0) worm->SpX = -worm->SpX;
            if (worm->SpX > -2) worm->SpX = -2;
            if (worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->SpY > -2 && worm->SpY < 2) worm->SpY = -2;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }
        else if (isBottom == linked->isBottomLeft)
        {
            if (memX < 0) memX = -memX;
            worm->SpX = memY / 2;
            worm->SpY = memY / 2 + memX;
            if (worm->SpX < 0) worm->SpX = -worm->SpX;
            if (worm->SpX < 2) worm->SpX = 2;
            if (worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->SpY > -2 && worm->SpY < 2) worm->SpY = -2;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = 2;
            if (worm->TurnSide < 0) worm->TurnSide = -worm->TurnSide;
        }
    }
    else if (isTop)                                                                   // Top, worm
    {
        if (isTop == linked->isTop)
        {
            worm->SpY = -memY;
            if (worm->SpY < 0) worm->SpY = -worm->SpY;
            if (worm->SpY < 2) worm->SpY = 2;
        }
        else if (isTop == linked->isBottom)
        {
            if (worm->SpY > 0) worm->SpY = -worm->SpY;
        }
        else if (isTop == linked->isRight)
        {
            worm->SpX = memY;
            worm->SpY = -memX;
            if (worm->SpX > 0) worm->SpX = -worm->SpX;
            if (worm->SpX > -2 && worm->SpX < 2) worm->SpX = -2;
            if(worm->SpY > 0) worm->SpY = -worm->SpY;
            if (worm->TurnSide > 0) worm->TurnSide = -worm->TurnSide;
        }

void CPortal::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if (Type == M_SKIPTURN) return;
  if (Type == M_TURNEND) start_be_inactive = true;
  if (Type == M_TURNBEGIN) 
  {
      start_be_inactive = false;       // Deactivate timer if turn started earlier then timer deactivated portal.
      inactive_timer = 0;              // Usefull if some1 is afk. Becouse otherwise timer would disable portal`s activity on some1`s turn start :D
      active = true;
  }
  
  if (Type == M_TOUCHSENSOR_COLLIDE)
  {
    if (linked != NullObj && Disabled == false)
    {
       local ts = CTouchSensor(sender);
       local activator = ts->activator;
       if (activator->ClType != OC_Worm)
       {
         if (!tpMissiles || (activator->ClType != OC_Missile)) return;
       } else {
         if (!trapTeleport && (activator != GetCurrentWorm())) return;
       }         
       if (timer <= 0)
       {
          MData->params[0] = 1; //disable collision
       }
    }
    return;
  }
  if (Type == M_TOUCHSENSOR)
  {
     if (linked == NullObj || Disabled) return;
     
     if (timer > 0)
     {
       timer = 2;
       return;
     }
     
     local ts = CTouchSensor(sender);
     local activator = ts->activator;
     
     if (activator->ClType != OC_Worm)
     {
       if (!tpMissiles || (activator->ClType != OC_Missile)) return;
     }  else {
       if (!trapTeleport && (activator != GetCurrentWorm())) return;
     }         
            
     TeleportObject(activator);
     
     if (activator->ClType == OC_Worm)
     {
       ts->noCollide = true;
     } else {
       ts->noCollide = false;
     }
     
     timer = 2;
     linked->timer = 2;
     return;
  }

  
  if (Type != M_FRAME) return;

// M_FRAME type starts here.
  

  if (linked == NullObj)
  {
    if (StrCmp(LinkedTo, "")) return;
    for (local i = 0; i < Root->Childs.Count; i += 1)
    {
      local obj = CPortal(Root->Childs.Objs[i]);
      if (obj is CPortal)
      {
        if (StrCmp(obj->t_name, LinkedTo))
        {
          linked = obj;
          linked->linked = this;
          break;
        }
      }
    }
    return;
  }
  if (linked == NullObj) Disabled = true;
//  else Disabled = false;
  if (Disabled == false && active)
  {
  
  if (timer2 % anim_fps) DoPortalEffects();         // particles animation
  timer2++; 
   
  if (landTeleport) 
  {
    if (timer > 0) timer--;
    return;
  }
  
  if (isExit) 
  {
    waiting = false;
    return;
  }
  
      CheckAndTeleport();

/*      for (local i = 1; i < Env->Objs.Count; i++)
      {
        local m = CLaserRay(Env->Objs.Objs[i]);
        if (m == NullObj) continue;
//        if (m->ClType != OC_AirStrike) continue;
        if (m is CLaserRay)
        {
          CLaserRay* ray = CLaserRay(m);                                   // imitation of Gas teleportation
//          local power = gas->Power;
        }
      }*/
  }
  
  if(start_be_inactive)
  {
      inactive_timer++;
      if (inactive_timer > timer_to_deactivate)  // 150 = 3 secs
      {
         active = false;         
         start_be_inactive = false;
         inactive_timer = 0;
      }
  }

  waiting = false;
}



void CPortal::CheckAndTeleport()
{

    if (teleMissiles)
    {
     for (local i = 1; i < Env->Objs.Count; i++)
     {
        local m = CMissile(Env->Objs.Objs[i]);
        if (m == NullObj) continue;
        if (m->ClType != OC_Missile) continue;         
        if (m->weap->CheckName("Portal Gun")) continue;
        if (m->portal_readed) continue; 
        if (pr_x1 < m->PosX && pr_x2 > m->PosX && pr_y1 < m->PosY && pr_y2 > m->PosY)
        {
          if (m->tLastTeleport != this)
          {
            m->tLastTeleport = linked;  
            TeleportObject(m);
            DoTeleportationEffects();
          }
        } else {
          if (m->tLastTeleport == this) m->tLastTeleport = NullObj;
        }
     }
    }

void CPortal::TeleportFire(CFire *Fire, fixed x, fixed y)
{
//    GG->WriteToChat( 3, "teleporting fire", false);
    local tx = (linked->pr_x1 + linked->pr_x2)/2;
    local ty = (linked->pr_y1 + linked->pr_y2)/2;
    Fire = CFire(NullObj);
    CFireDesc desc;
    zero(&desc);
    desc.X = tx;
    desc.Y = ty;
    desc.SpX = linked->spx1 * 7;
    desc.SpY = linked->spy1 * 7;
    desc.Count = 0;
    desc.Duration = 1500;
    desc.UpPower = 1;
    desc.Damage = 2;
    desc.Unk = 0;
    desc.UnkB = 5;
    desc.UnkD = 0;
    desc.OwnerTeam = 0;
    local f = new CFire(Root->GetObject(25, 0), &desc, 0);
}

override CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
{
    super;
    portal_just_teleported = false;
    portal_deactive_tele_bool = 40;
    portal_deactive_tele_counter = 40;
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    super;
    if (Type == M_FRAME)
    {
        if (portal_just_teleported)
        {
            if (portal_deactive_tele_counter != 0)
            {
                portal_deactive_tele_counter--;
            }
            else
            {
                portal_deactive_tele_counter = portal_deactive_tele_bool;
                portal_just_teleported = false;
            }
        }
    }
}
