//-----------------------------------------------------------------------------
// Abstract data types
//-----------------------------------------------------------------------------
// Nodes
//-----------------------------------------------------------------------------
// IntLinkedNode
//-----------------------------------------------------------------------------
CIntLinkedNode : CObject;
//-----------------------------------------------------------------------------
CIntLinkedNode::CIntLinkedNode(CObject* Parent) {
  super(Parent, GS);
  Data = 0;
} 
//-----------------------------------------------------------------------------                                         
void CIntLinkedNode::NewNext(int data) {
  Next = new CIntLinkedNode(this);
  Next->Data = data;
  Next->Next = NullObj;
  Next->Prev = this;
} 
//-----------------------------------------------------------------------------                                        
void CIntLinkedNode::NewPrev(int data) {
  Prev = new CIntLinkedNode(this);
  Prev->Data = data;
  Prev->Next = this;
  Prev->Prev = NullObj;
} 
//-----------------------------------------------------------------------------


//-----------------------------------------------------------------------------
// Queues
//-----------------------------------------------------------------------------
// IntQueue
//-----------------------------------------------------------------------------
CIntQueue : CObject;
//-----------------------------------------------------------------------------
CIntQueue::CIntQueue(CObject* Parent) {
  super(Parent, GS);
  Head = new CIntLinkedNode(Parent);
  Tail = Head;
  Size = 1;
}                                
//-----------------------------------------------------------------------------
void CIntQueue::Push(int data) {
  Tail->NewNext(data);
  Tail = Tail->Next;
  Size++;
}                                   
//-----------------------------------------------------------------------------
int CIntQueue::Peek() {
  return Head->Data;                 
}                          
//-----------------------------------------------------------------------------
int CIntQueue::Pop() {
  local retInt = Head->Data;
  Head = Head->Next;
  Size--;
  return retInt;                 
}
