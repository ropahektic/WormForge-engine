require utils;

override CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
{
    super;
    anti_FD = false;
    remote_fd_switch = false; // this made to activate additional no_loss_control option remotely (from other scripts) 
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
local w = GetCurrentWorm();
  if (w != NullObj)
  {
    local rem;
    if (anti_FD && (w->SpY > 8.0 || w->SpX > 8.0 || w->SpX < -8.0 || w->SpY < -8.0) && (w->ObjState == WS_JUMP || w->ObjState == WS_FLYING2 || w->ObjState == WS_SLIDING || w->ObjState == WS_FLYING || w->ObjState == WS_AFTERROPE || w->ObjState == WS_POWERFLY)) 
    {
       if (rem == 1) w->SetState(WS_JUMP);
       else if (rem == 2) w->SetState(WS_FLYING2);
       else if (rem == 3) w->SetState(WS_AFTERROPE);
       else if (rem == 4) w->SetState(WS_SLIDING);
       else w->SetState(WS_IDLE);
    }          
    else if (anti_FD && (w->SpY < 8.0 || w->SpX < 8.0 || w->SpX > -8.0 || w->SpY > -8.0))
    {
       if (w->ObjState == WS_JUMP) rem = 1;
       else if (w->ObjState == WS_FLYING2) rem = 2;
       else if (w->ObjState == WS_AFTERROPE) rem = 3;
       else if (w->ObjState == WS_SLIDING) rem = 4;
       else 
       {
          rem = 5;
          if (remote_fd_switch) w->SetState(WS_IDLE);
       }
    }
  }
  if (Type == M_TURNEND)
  {
     anti_FD = false;
  }
super;
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
    if (Weap->CheckName("Anti Fall Damage"))
    {
        if(anti_FD == false)
        {
           local w = GetCurrentWorm();
        
           w->anti_FD = true;
           ShowMessage("No Fall Damage: ON");
        }
        else if(anti_FD == true)
        {
           local w = GetCurrentWorm();
        
           w->anti_FD = false;
           ShowMessage("No Fall Damage: OFF");
        }
        super;    
    }
    else super;
}
