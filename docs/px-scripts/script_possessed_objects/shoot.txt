void ShootZook(float px, float py, float spx, float spy, string wepname)
{
  CShootDesc sdesc;
  zero(&sdesc);
  sdesc.X = px;    
  sdesc.Y = py;  
  sdesc.SpX = spx;         
  sdesc.SpY = spy;  
  local ldata;
  ldata = GetWeaponByName(wepname); 
  local msl = new CMissile(Root->GetObject(25, 0), &ldata->launch, &sdesc);    
}
