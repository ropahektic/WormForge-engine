void CMine::Possess()
{
   PossessType = RandomInt(1,11);
   //PossessType = 1;
   if(PossessType == 8 || PossessType == 10){PossessType = PossessType + 1;}    //Dont forget to remove this once I add new types.
   if(PossessType == 1)
   {
   Armor = 10;
   }
   
   if(PossessType == 7 || PossessType == 4 || PossessType == 3)
   {
   Armor = 1;
   }
}

override void CMine::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
         if(Type == M_FRAME)
         {
            Possess_Frame();
         }
         
         if(Type == M_GUNEXP)
         {
            if(PossessType == 1 && Armor > 0)
            {
                 Armor = Armor - 1;
                 return;
            }
            if(PossessType == 7 && Armor > 0)
            {
                 Armor = Armor - 1;
                 Terraform();
            }
            if(PossessType == 4 && Armor > 0)
            {
                 Armor = Armor - 1;
                 Shoot();
            }
            if(PossessType == 3 && Armor > 0)
            {
                 Armor = Armor - 1;
                 Swap();
                 return;
            }
         }
         
         if(Type == M_EXPLOSION)
         {
         fixed x = MData->fparams[1];
         fixed y = MData->fparams[2];
                 if(x <= PosX + 50 && x >= PosX - 50)
                 {
                         if(y <= PosY + 50 && y >= PosY - 50)
                         {
                              if(PossessType == 1 && Armor > 0)
                              {
                                   Armor = Armor - 1;
                                   return;
                              }
                              if(PossessType == 7 && Armor > 0)
                              {
                                   Armor = Armor - 1;
                                   Terraform();
                              }
                              if(PossessType == 4 && Armor > 0)
                              {
                                   Armor = Armor - 1;
                                   Shoot();
                              }
                              if(PossessType == 3 && Armor > 0)
                              {
                                   Armor = Armor - 1;
                                   Swap();
                                   return;
                              }
                         }
                 }
         }
super;
}
