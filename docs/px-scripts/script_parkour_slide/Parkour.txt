//PARKOUR MOD

//Additional abilities of the worms:

//Wall-Jump (ENTER)
//If the wall is close enough, the worm will make an additional jump in the air.
//Basic math: 
//SpY = SpY - WallJumpMultiplier.
//SpX = (WallJumpMultiplier-1);

//Bunny-Hop (BACKSPACE)
//If the floor is close, the worm will make another jump with additional X speed. Can be used while sliding.
//Basic math: 
//SpX = SpX + BhopMultiplier*TurnSide.
//SpY = SpY - 4

//Slide (SHIFT)
//Worm getting "Slide" state, allowing him to increase and keep SpX for some time.
//If the worm used Slide, while sliding, he will get Flying state, that allows him to stop upon connecting with ground.
//Basic math: 
//SpX = SpX + SlideMultiplier*TurnSide.
//SetState(WS_SLIDING)

void Parkour::InitGraphic()     //func that will run on GraphicInit. Usually used to load sprites.
{
}

void Parkour::Init()     //func that will run on init. Usually used to load non-sprite objects, like sounds.
{
keySync->AddKey(38);  // up
keySync->AddKey(40);  // down
keySync->AddKey(39);  //right
keySync->AddKey(37);  //left
keySync->AddKey(16);  // shift
}

void Parkour::FirstFrame()
{	
}



override CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
{
super;
WallJumpMultiplier = 5;
BhopMultiplier = 3;
SlideMultiplier = 3;
WallJumpDelay = 0;
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
super;
   if(GetCurrentWorm() != NullObj)
   {
      if(GetCurrentWorm() == this)
      {
         if(Type == M_FRAME)
         {
            if(SpY > 6){SpY = 6;}    //weeeeeeeeeeeeeeeee...
            if(SpY < -16){SpY = -16;}  
            if(WallJumpDelay >= 1) {WallJumpDelay = WallJumpDelay - 1;}
         }
      
         if(Type == M_ENTER)
         {
            if(WallJumpDelay < 1)
               {
               WallJump();
               WallJumpDelay = 4;
               }
         }
      
         if(Type == M_BACKSPACE)
         {
         BhopJump();
         }
      
         if(keySync->KeyPressedNow(16) == true)
         {
         Slide();
         }

    
         if(keySync->KeyPressed(38) == true || keySync->KeyPressed(39) == true || keySync->KeyPressed(40) == true || keySync->KeyPressed(37) == true)
         {
         Land();
         } 
      }
   }  
}


void CWorm::WallJump()
{
     if(ObjState == WS_JUMP || ObjState == WS_SLIDING || ObjState == WS_FLYING2)
     {
       //wall to the right
       int hit2x;
       int hit2y;
       CGObject *hit2 = TraceLine(this, PosX, PosY, PosX+20, PosY, 2, &hit2x, &hit2y);   
              
       //wall to the left
       int hit3x;
       int hit3y;
       CGObject *hit3 = TraceLine(this, PosX, PosY, PosX-20, PosY, 2, &hit3x, &hit3y); 
       
                if(hit2 != NullObj && hit3 != NullObj)
		{   
		   local x2 = sqrt((hit2x - PosX)*(hit2x - PosX));
		   local x3 = sqrt((hit3x - PosX)*(hit3x - PosX));
		   local side = 0;
		   if(x2 < x3){ side = -1;}
		   else{ side = 1;}
		   
		   
                       if(SpY > 0){SpY = 0;}         			
                       SpY = SpY - WallJumpMultiplier;
                       SpX = (WallJumpMultiplier-1) * side;
                       TurnSide = side;
                       if(ObjState != WS_JUMP){ SetState(WS_JUMP);}
		}
		else
		{
		    if(hit2 != NullObj)
		    {
		       if(SpY > 0){SpY = 0;}
                       SpY = SpY - WallJumpMultiplier;
                       SpX = -1*(WallJumpMultiplier-1);
                       TurnSide = -1;
                       if(ObjState != WS_JUMP){ SetState(WS_JUMP);}
		    }
		    
		    if(hit3 != NullObj)
		    {
		       if(SpY > 0){SpY = 0;}
                       SpY = SpY - WallJumpMultiplier;
                       SpX = (WallJumpMultiplier-1);
                       TurnSide = 1;
                       if(ObjState != WS_JUMP){ SetState(WS_JUMP);}
		    }
		}
     }
}



void CWorm::BhopJump()
{
     if(ObjState == WS_JUMP || ObjState == WS_FLYING2 || ObjState == WS_FLYING || ObjState == WS_POWERFLY)
     {
       int hitx1;
       int hity1;
       CGObject *hit1 = TraceLine(this, PosX, PosY, PosX, PosY+30, 2, &hitx1, &hity1);
                if(hit1 != NullObj)
		{     
                       if(SpY > 0){SpY = 0;}       			
                       SpY = SpY - BhopMultiplier;
                       SpX = SpX + (BhopMultiplier-1)*TurnSide;
                       if(ObjState != WS_JUMP){ SetState(WS_JUMP);}
		}
     }
}

void CWorm::Slide()
{             
   if(ObjState == WS_JUMP)
   {
   SetState(WS_FLYING2);
   SpX = SpX + SlideMultiplier*TurnSide;                
   }
} 
     
void CWorm::Land()
{
   if(ObjState == WS_FLYING2)
   {
   SetState(WS_SLIDING);
   }      
}
