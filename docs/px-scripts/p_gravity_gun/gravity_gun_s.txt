bool CWorm::GG_CanFire()
{
    if (gg_canfire_timer_now >= gg_canfire_timer)
    {
        return true;
    } else
    {
        gg_canfire_timer_now++;
        return false;
    }
}

void CWorm::GG_ResetFireCounter()
{
    if (gg_canfire_timer_now == gg_canfire_timer) gg_canfire_timer_now = 0;
}

void CWorm::CheckObjExist(CGObject* obj)
{
    if (obj == NullObj && obj == holded_obj)
    {
        DisableGGun();
        holded_obj = NullObj;
        gg_draw_ghost_ggun = false;
        gg_used_gg = false;
    }
    for (local i = 1; i < Env->Objs.Count; i++)
    {
        local ob = CGObject(Env->Objs.Objs[i]);
        if (ob == obj) 
        {
            if (!ob->IsValid())               
            {
                obj = NullObj;      
               if (obj == holded_obj) 
               {
                   DisableGGun();
                   holded_obj = NullObj;
                   gg_draw_ghost_ggun = false; 
                   gg_used_gg = false;               
               }
            }
            return;
        }
    }
    if (holded_obj != NullObj)
    {
        DisableGGun();
        gg_draw_ghost_ggun = false; 
        gg_used_gg = false;      
    }
}

void CWorm::GG_calculate_angle()
{
    if (CurWeapon->CheckName("Gravity Gun"))
    {
        float ang = (1 - FireAngle) * MATH_PI;
	if (TurnSide < 0) ang = MATH_TWO_PI - ang;       
	GG_ang_x = sin(ang);
	GG_ang_y = -cos(ang);
        if (!gg_draw_ghost_ggun)
	{
		gg_last_ang = GG_WS->aimAng*TurnSide;
		if (TurnSide < 0) gg_last_ang = MATH_PI - gg_last_ang;
	}
	else if (abs(FireAngle) >= 0.8)
	{
//		if (ObjState == WS_AIMING) 
                gg_last_ang = GG_WS->aimAng*TurnSide;
		if (TurnSide < 0) gg_last_ang = MATH_PI - gg_last_ang;	
	}
	gg_last_turnside = Sign(TurnSide);       
    }                                                      
}

void CWorm::ThrowObject()
{
//	local w = GetCurrentWorm();
//	if (w == NullObj || w != this) return;
        GG_calculate_angle();

	float endX = PosX + GG_ang_x * gg_throw_distance;
	float endY = PosY + GG_ang_y * gg_throw_distance;      
    
	int hitX; int hitY; 
	CGObject *hitObject = TraceLine(this, PosX, PosY, endX, endY, 2+4+8+16+32+64+128+256+512+1024+2048+4096+8192+16384+32768+65536+131072+262144+524288, &hitX, &hitY);

	if(hitObject != NullObj)
	{
                        float dsqr;
                        for(i = 1; i < Env->Objs.Count; i += 1)
			{
				obj = CGObject(Env->Objs.Objs[i]);
				if(obj == NullObj) continue;
				
                                        float dx = PosX - hitX;
					float dy = PosY - hitY;
					dsqr = dx*dx+dy*dy;				
                                				
				if(obj->ClType == OC_Landscape) 
                                {
                                   gg_dryfire_s->Play(1.0, 0.0, false);
                                   continue;
                                }
				if(gg_ignore_worms && obj is CWorm) 
				{
					if (!gg_superworm) {continue;}
					else if (gg_ignore_superworms) {continue;}
				}                                                 
				
				if (obj == hitObject)
				{
				        local Weap = GetWeaponByName("Gravity Gun");
                                        GS->Info.AddWeaponEx(WormTeam, Weap->GetWeaponIndex(), Weap->GetPageIndex(), -1);
				
					if (#WEAPON_BOWLINGBALL)
                                        {
                                            if (obj is CBowlingBall)
                                            {
                                        	CBowlingBall* m = CBowlingBall(obj);
                                        	m->Zap(1);
//                                        	m->GravityFactor = 1;
                                            }
                                        } 
                                        
                                        obj->SpX	= GG_ang_x * gg_push_powa * ExplosionFactor; 
					obj->SpY	= GG_ang_y * gg_push_powa * ExplosionFactor;
					
					DoGGTrailBeam(hitX, hitY, sqrt(dsqr));
					
					gg_throw_s->Play(0.75, 0.0, false);
					return;
										
                                }				
			}
			if (gg_bounce_of_landscape)
			{
				GG_BounceOfWall(0.1);
                                DoGGTrailBeam(hitX, hitY, sqrt(dsqr));
                                gg_throw_s->Play(0.40, 0.0, false);			
			}
			else
			{
				gg_dryfire_s->Play(1.0, 0.0, false);
			}	
	} else
	{
	    gg_dryfire_s->Play(1.0, 0.0, false);
	}
}

void CWorm::HoldObject(CGObject* obj, int stop_radius2)
{
//	local w = GetCurrentWorm();
//	if (w == NullObj || w != this) return;
	local last_gg_x;
	local last_gg_y;
	local last_turnside;
	local work;
//	if (gframe % 25) {} else {CheckObjExist(obj);}      // check if object exist
	CheckObjExist(obj);          // we need check this each frame, tho. or it can crash the game
        if (obj == NullObj || holded_obj == NullObj) 
        {
           DisableGGun();
           gg_draw_ghost_ggun = false; 
           gg_used_gg = false; 
           holded_obj = CGObject(NullObj);
           return;
        }
				if(gg_ignore_worms && obj is CWorm) 
				{
					if (!gg_superworm) return;
					else if (gg_ignore_superworms) return;
				}
        last_gg_x = GG_ang_x;
        last_gg_y = GG_ang_y;
        last_turnside = gg_last_turnside;
        
	GG_calculate_angle();

	float endX = PosX + GG_ang_x * stop_radius2;                                                                                                                                 
	float endY = PosY + GG_ang_y * stop_radius2; 
        
        if (gg_hold_work2)
        {
           endX = PosX + GG_ang_x * stop_radius;
           endY = PosY + GG_ang_y * stop_radius; 
        } else stop_radius = stop_radius2; 
	
        if (last_turnside != gg_last_turnside)
        {
		if ((abs(GG_ang_x - last_gg_x) < 0.3 && abs(GG_ang_y - last_gg_y) < 0.3) || portal_just_teleported)
		{
			work = true;
		} 
		else	work = false;
        }
        else 	work = true;
	
        if (ObjState == WS_AIMING) 
        {
           DoGGLightEffect(0.7, PosX + GG_ang_x*25, PosY + GG_ang_y*25, 2, 240);
        }
        if (obj != NullObj && holded_obj != NullObj) DoGGLightEffect(0.7, obj->PosX, obj->PosY, 2, 240);   
	
	local hit = CheckMaskAt(obj, obj->ColMask, endX, endY, -1);
	if (!work)
	{
	    Spark_a_bit(10, obj->PosX, obj->PosY);
            DisableGGun();
            gg_draw_ghost_ggun = false; 
            gg_used_gg = false;
            obj = CGObject(NullObj);
            holded_obj = CGObject(NullObj);
            gg_small_zap2_s->Play(1.5, 0.0, false);
            return;
	}
        if (hit != NullObj) 
        {
            gg_hold_work2 = true;
            gg_timer--;
            if (gg_timer <= 0)
            {
               DisableGGun();
               gg_draw_ghost_ggun = false; 
               gg_used_gg = false;
               obj = CGObject(NullObj);
               gg_hold_work2 = false;
            } else
            {
                if (stop_radius > 6) stop_radius -= 3;
                else stop_radius = stop_radius2;
                if(gg_timer % 5){}
                else 
                {
                   if (obj != NullObj && holded_obj != NullObj) Spark_a_bit(2, obj->PosX, obj->PosY);
                   gg_small_zap2_s->Play(1.0, 0.0, false);
                }
            }
        } else
        {
            if (work)
            {
		local endX1;	local endY1;
                if (stop_radius != gg_stop_radius)
		{
			endX1 = PosX + GG_ang_x * gg_stop_radius;
			endY1 = PosY + GG_ang_y * gg_stop_radius;
			local hit1 = CheckMaskAt(obj, obj->ColMask, endX1, endY1, -1);
			if (hit1 == NullObj) 
			{
				stop_radius = gg_stop_radius;
				gg_hold_work2 = false;
			}
		}
		float dx = obj->PosX - endX;	float dy = obj->PosY - endY;
		float dsqr2 = dx * dx + dy * dy;
		float dist = sqrt(dsqr2);
		if ((stop_radius + 15 - abs(dist)) > 0 || portal_just_teleported)
		{
			gg_timer = gg_unhold_timer;
//			obj->PosX = endX;
//			obj->PosY = endY;
			TeleportObject(obj,endX,endY,obj->MaskIndex,false,2);
			obj->SpX = 0.0;
			obj->SpY = 0.0;
		}
		else
		{
		    work = false;
		}
            }
        }  
}

void CWorm::SuperThrowObject()
{
//	local w = GetCurrentWorm();
//	if (w == NullObj || w != this) return;
	GG_calculate_angle();

	float endX = PosX + GG_ang_x * gg_throw_distance*8;
	float endY = PosY + GG_ang_y * gg_throw_distance*8;      
    
	int hitX; int hitY; 
	CGObject *hitObject = TraceLine(this, PosX, PosY, endX, endY, 2+4+8+16+32+64+128+256+512+1024+2048+4096+8192+16384+32768+65536+131072+262144+524288, &hitX, &hitY);

	if(hitObject != NullObj)
	{
                        float dsqr;
                        for(i = 1; i < Env->Objs.Count; i += 1)
			{
				obj = CGObject(Env->Objs.Objs[i]);
				if(obj == NullObj) continue;
				
					float dx = PosX - hitX;
					float dy = PosY - hitY;
					dsqr = dx*dx+dy*dy;				
				
				if(obj->ClType == OC_Landscape) 
                                {
                                   gg_dryfire_s->Play(1.0, 0.0, false);
                                   continue;
                                }
				if(gg_ignore_worms && obj->ClType == OC_Worm && gg_ignore_superworms)
				{
					gg_dryfire_s->Play(1.0, 0.0, false);
					continue;                                
				}                                				
				
				if (obj == hitObject)
				{
				        local Weap = GetWeaponByName("Gravity Gun");
                                        GS->Info.AddWeaponEx(WormTeam, Weap->GetWeaponIndex(), Weap->GetPageIndex(), -1);
                                        
                                        local do_throw;
					
                                        if (#WEAPON_BOWLINGBALL)
                                        {
                                            if (obj is CBowlingBall)
                                            {
                                        	CBowlingBall* m = CBowlingBall(obj);
                                        	m->Zap(0);
                                        	do_throw = false;
                                            }
                                        } 
                                        if (#WEAPON_MAGNET)
                                        {
                                            if (obj is CMagnet)
                                            {
                                                CMagnet* m = CMagnet(obj);
                                                m->Recharge();
                                                do_throw = true;
                                            }
                                        }
                                        if(#WEAPON_FAN)
                                        {
                                            if(obj is PxFan)
                                            {
                                                local fanobj = PxFan(obj);
                                                fanobj->Recharge();
                                                do_throw = true;
                                            }
                                        }
                                        if(#WEAPON_KSENTRY)
                                        {
                                            if(obj is PxSentryGun)
                                            {
                                                local sentry = PxSentryGun(obj);
                                                sentry->OwnerTeam = WormTeam;
                                                sentry->TurretActive = false;
                                                do_throw = true;
                                            }
                                        }
					if (obj is CMine)
					{
                                                if (obj is CMagnet){}
                                                else {
                                                CMine* mine = CMine(obj);
						if(mine->Radius > 0 && mine->Active == 0 && mine->Prefuse <= 0)
						{
							if(mine->Fuse < 0)
							{
								mine->Fuse = RandomInt(1000, 5000);
							}
							mine->Active = 1;
							mine->nPulses = mine->Fuse / 250;
						}						
						do_throw = true;
						}
					} else
					if (obj is CCrate || obj is COilDrum)
					{
						if (sqrt(dsqr) < (gg_stop_radius + 20))
                                                {
                                                    do_throw = true;
                                                } else
                                                {
                                                    DoExplosion(obj->PosX, obj->PosY, 1, 1, 0, WormTeam);
						    do_throw = false;
		                                }
					} else
					{
						do_throw = true;
					}
				
					if (do_throw)
					{
        	                                obj->SpX	= GG_ang_x * gg_push_powa * ExplosionFactor * 1.5; 
						obj->SpY	= GG_ang_y * gg_push_powa * ExplosionFactor * 1.5;
					}
					gg_throw_s->Play(0.75, 0.0, false);
					DoGGTrailBeam(hitX, hitY, sqrt(dsqr));
					return;
										
                                }				
			}
			if (gg_bounce_of_landscape)
			{
				GG_BounceOfWall(0.15);
                                DoGGTrailBeam(hitX, hitY, sqrt(dsqr));
                                gg_throw_s->Play(0.40, 0.0, false);			
			}
			else
			{
				gg_dryfire_s->Play(1.0, 0.0, false);
			}                        	
	} else
	{
	    gg_dryfire_s->Play(1.0, 0.0, false);
	}
}

void CWorm::DoGGTrailBeam(float hitX, float hitY, float dist)
{
    local num_sprites = abs(dist/10);
    if (num_sprites < 2) num_sprites = 2;
    local last_px;
    local last_py;
    for (i = 1; i <= num_sprites; i++)
    {
        if (i == 1 || i == num_sprites)
        {
            last_px = 0;
            last_py = 0;       
        }
        
        float px = last_px + PosX + GG_ang_x*i*10;
        float py = last_py + PosY + GG_ang_y*i*10;
        
        local rx = RandomFloat(-5.0, 5.0)*abs(GG_ang_x);
        local ry = RandomFloat(-5.0, 5.0)*abs(GG_ang_y);
        
        local rx1 = RandomFloat(-20.0, 20.0)*abs(GG_ang_x);
        local ry1 = RandomFloat(-20.0, 20.0)*abs(GG_ang_y);
        
        if (i != 1 || i != num_sprites)
        {
            rx = 0;
            ry = 0;
        }
        
//        DoGGBeamEffect(px + rx, py + ry, 50, ang);
        local alpha = 255 - 255*i/num_sprites;
        local alpha2 = 255 - 200*i/num_sprites;
        if (i%2){} else DoGGLightEffect(RandomFloat(0.3, 0.5), px + rx1, py + ry1, 40, alpha2);
        DoGGLightEffect(0.1, px + rx, py + ry, 40, alpha);
    }
}
