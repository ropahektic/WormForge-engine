require utils, p_sprite_builder, sprite_curtis_250;

CSprite* cchicken_aiming;
CSprite* fscchickena;
CSprite* fscchickenb;

void crazy_chicken_script::InitGraphic()
{
    CFile *f;
    f = GetAttachment("cchicken.png");
    cchicken_aiming = LoadSprite(f, 4, 0);
    f = GetAttachment("fscchickena.png");
    fscchickena = LoadSprite(f, 32, 1);
    f = GetAttachment("fscchickenb.png");
    fscchickenb = LoadSprite(f, 32, 1);
    
    SetSpriteOverride(462, fscchickena);
    SetSpriteOverride(463, fscchickenb);
}

//---------------------------------CRAZY CHICKEN------------------------------//

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) {
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Crazy Chicken";
     sprite = cchicken_aiming->Index;
     Params->hand_type = 1;
     Params->hand_scale = 1.2;
     Params->hand_radius = -13.2;
     Params->hand_rotation = -0.0;
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;
     Params->weap_rotation = 0.5;
     Params->weap_radius = -11.0;
     Params->animate = true;
     Params->anim_speed = 1.35;
     Params->dont_rotate_with_angle = true;
     Params->hand_ang_if_static = 0.4;
     Params->ang_if_static = 0.0;
     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}
