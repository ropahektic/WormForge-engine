void golf_zones::InitGraphic()
{
	CFile *f;
	f = GetAttachment("red_flag.png");		golfHoleRed = LoadSprite(f, 1, 0);
	f = GetAttachment("yellow_flag.png");		golfHoleYellow = LoadSprite(f, 1, 0);
	f = GetAttachment("blue_flag.png");		golfHoleBlue = LoadSprite(f, 1, 0);
}


CGolfZone::CGolfZone(CEditorObject* aobj)
{
  super(Root, GS);
  pr_x1 = aobj->PosX;
  pr_x2 = aobj->PosX + aobj->GetInt("Size");
  pr_y1 = aobj->PosY;
  pr_y2 = aobj->PosY + aobj->GetInt2("Size");
  if (StrCmp(aobj->GetString("Type"), "HighGrass")) golf_type = 1;
  if (StrCmp(aobj->GetString("Type"), "LowGrass")) golf_type = 2;
  if (StrCmp(aobj->GetString("Type"), "Bunker")) golf_type = 3;
  if (StrCmp(aobj->GetString("Type"), "Green")) golf_type = 4;        
  if (StrCmp(aobj->GetString("Type"), "OutOfBounds")) golf_type = 5; 
}


CGolfSpriterHole::CGolfSpriterHole(CEditorObject* aobj)
{
  super(Root, GS);
  scale = 0.33;
  x = aobj->PosX + 50*scale;
  y = aobj->PosY - 170*scale;
  z = 16.0 + aobj->GetInt("Z_Layer")/100;
  rot = 0.0;
  hx1 = x-25;
  hx2 = x-5;
  hy1 = y+46;
  hy2 = y+58;
  par = aobj->GetInt("Par");
  hole = aobj->GetInt("Hole_Number");
  if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfHoleRed->Index;
  if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfHoleYellow->Index;
  if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfHoleBlue->Index;
  lastHole = aobj->GetBool("LastHole");
}

CGolfHoleStart::CGolfHoleStart(CEditorObject* aobj)
{
  super(Root, GS);
  hole = aobj->GetInt("HoleNumber");
  x = aobj->PosX;
  y = aobj->PosY;
  restart = aobj->GetBool("Restart_Counter");
  if (StrCmp(aobj->GetString("Direction"), "LEFT_to_RIGHT")) offset = -13;
  if (StrCmp(aobj->GetString("Direction"), "RIGHT_to_LEFT")) offset = 13;
}
