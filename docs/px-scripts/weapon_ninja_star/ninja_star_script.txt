require utils, sprite_curtis_457;

CBitmap32* NinjaSBitmap;
CSoundFile* ninjasimpactsnd;


void ninja_star_script::InitGraphic()
{
    local spr       = GetAttachment("ninjas.png");
    NinjaSBitmap     = LoadImage(spr);
}

void ninja_star_script::FirstFrame()
{
	CFile *f;
	f = GetAttachment("ninjasimpact.wav");	ninjasimpactsnd = new CSoundFile(f);
}

override void CMissile::ExplodeAt(fixed x,fixed y)
{

 if(weap->CheckName("Ninja Star") == true)
 {
           CGObject *ninjaob;
     	   for(int i = 0; i < Env->Objs.Count; i += 1)
	   {
		ninjaob = CGObject(Env->Objs.Objs[i]);
		if(ninjaob == this) continue;
		if(ninjaob == NullObj) continue;
		if(ninjaob->IsMaterial == false) continue;
			if(ninjaob->ClType == OC_Worm || ninjaob->ClType == OC_Mine || ninjaob->ClType == OC_OilDrum || ninjaob->ClType == OC_Crate)
			{
				local trg = CGObject(ninjaob);
                                if(x > trg->PosX-20 && x < trg->PosX+20 && y > trg->PosY-20 && y < trg->PosY+20)
                                {
                                NinjaSDamage(trg);
                                super;
                                return;
                                }
			}
           }

           if(GetCurrentWorm() != NullObj)
           {
                   if(x > GetCurrentWorm()->PosX-25 && x < GetCurrentWorm()->PosX+25 && y > GetCurrentWorm()->PosY-25 && y < GetCurrentWorm()->PosY+25)
                   {
                   NinjaSDamage(GetCurrentWorm());
                   super;
                   return;
                   }
           }

           SStickToLandscape(x,y);
           vol = CalculateSoundVolume(x,y);
     	   pan = CalculateSoundPan(x,y);
           ninjasimpactsnd->Play(vol,pan,false);
 }

 super;
}

void CMissile::NinjaSDamage(CGObject* target)
{
                     local damage = 1;
                     if(weap->CheckName("Ninja Star") == true)
                     {damage = 10;}

                     target->SpX = SpX/4;
                     target->SpY = SpY/4;

                     CMessageData msgdamage;
                     msgdamage.params[0] = 3;
                     msgdamage.params[1] = PosX;
                     msgdamage.params[2] = PosY;
                     msgdamage.params[3] = SpX/2;//doesnt work for some reason..
                     msgdamage.params[4] = SpY/2;
                     msgdamage.params[5] = damage;
                     msgdamage.params[6] = 0;
                     target->Message(NullObj, M_GUNEXP, 1032, &msgdamage);
}

void CMissile::SStickToLandscape(fixed x, fixed y)
{
local bmp = NinjaSBitmap;
local top = bmp->top;
int mtp = 0;

      if(SpX == 0.0 && SpY < 0.0){mtp = 0;}
      if(SpX > 0.0 && SpY == 0.0){mtp = 8;}
      if(SpX == 0.0 && SpY > 0.0){mtp = 16;}
      if(SpX < 0.0 && SpY == 0.0){mtp = 24;}

      if(SpX > 0.0 && SpY < 0.0)
      {
      mtp = 3*SpX/sqrt(SpY*SpY);
      if(mtp>7){mtp = 7;}
      if(mtp<0){mtp = 0;}
      }

      if(SpX > 0.0 && SpY > 0.0)
      {
      mtp = 8+3*SpY/SpX;
      if(mtp>15){mtp = 15;}
      }

      if(SpX < 0.0 && SpY > 0.0)
      {
      mtp = 16+3*sqrt(SpX*SpX)/SpY;
      if(mtp>23){mtp = 23;}
      }

      if(SpX < 0.0 && SpY < 0.0)
      {
      mtp = 24+3*sqrt(SpY*SpY)/sqrt(SpX*SpX);
      if(mtp>31){mtp = 31;}
      }

top = mtp*60;
bmp->top = top;

bmp->bottom = bmp->top+60;
GG->land->Draw(bmp, x-30, y-(30+bmp->top));
}
