CSpriterktk_29_beam::CSpriterktk_29_beam(CEditorObject* aobj)
{
  super(Root, GS); 
  Color_R = aobj->GetInt("Color_R");
  Color_G = aobj->GetInt("Color_G");
  Color_B = aobj->GetInt("Color_B");
  Scale = aobj->GetInt("Size");
  scale = Scale/100.0;
  rot = aobj->GetInt("Rotation")/57.325;
  x = aobj->PosX;
  y = aobj->PosY;        
  
  newx = x+(39*sin(rot))*scale;      // number is half of hight of your image
  newy = y-(39*cos(rot))*scale;      // number is half of hight of your image
  
  if (StrCmp(aobj->GetString("Z_Layer"), "Behind_Map")) z = 25;
  if (StrCmp(aobj->GetString("Z_Layer"), "Above_Map")) z = 15;
  if (StrCmp(aobj->GetString("Z_Layer"), "Above_Worm")) z = 5;
  
  pluslayer = aobj->GetInt("Add_Layer")/10;
  
  if (StrCmp(aobj->GetString("Transparency"), "No")) transp = 0;
  if (StrCmp(aobj->GetString("Transparency"), "Yes")) transp = 1;
  
  if (StrCmp(aobj->GetString("Flip"), "No")) flip = 0;
  if (StrCmp(aobj->GetString("Flip"), "Yes")) flip = 262144;   
}

{
    if (Type == M_DRAWQUEUE)
    {
        sprite = ktk_29_beam->Index;
        SetColorMod(RGB(Color_R, Color_G, Color_B), transp); 
        AddSpriteEx(z+pluslayer, newx, newy, sprite+flip, 0.0, rot, scale);
        ClearColorMod();
    }
