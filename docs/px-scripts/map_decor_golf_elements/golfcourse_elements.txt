//Windmill
CWindMillBase::CWindMillBase(CEditorObject* aobj)
{
  super(Root, GS);
  Scale = aobj->GetInt("Size");
  scale = Scale/100.0;
  x = aobj->PosX;
  y = aobj->PosY - 1*scale;
  z = 16.0 + aobj->GetInt("Z_Layer")/100;
  rot = 0.0;
}


void CWindMillBase::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_DRAWQUEUE)
    {
        AddSpriteEx( z, x, y, golfWindMillBase->Index, 0.0, rot, scale);
    }
}

CWindMillBlades::CWindMillBlades(CEditorObject* aobj)
{
  super(Root, GS);
  Scale = aobj->GetInt("Size");
  scale = Scale/100.0;
  x = aobj->PosX;
  y = aobj->PosY - 1*scale;
  z = 16.0 + aobj->GetInt("Z_Layer")/100;
  rot = 0.0;
  
  AnimTimer = 0.0;
  RotationTimer = 0.0;
    
}


void CWindMillBlades::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{

    if (Type == M_DRAWQUEUE)
    {  
        AddSpriteEx( z, x, y, golfWindMillBlades->Index, 0.0, RotationTimer, scale);
        //AddSpriteEx( z, x, y, golfWindMillBlades->Index, AnimTimer, rot, scale);

    }
    if (Type == M_FRAME)
    {  
        if (AnimTimer < 1.0) {
                  AnimTimer += 0.06;
               }
        else AnimTimer = 0.0;
        
        if (RotationTimer < 90.0) {
                  RotationTimer += 0.02;
               }
        else RotationTimer = 0.0;

    }

} 


//Water Wave
CWaterWave::CWaterWave(CEditorObject* aobj)
{
  super(Root, GS);
  Scale = aobj->GetInt("Size");
  scale = Scale/100.0;
  x = aobj->PosX;
  y = aobj->PosY - 1*scale;
  z = 16.0 + aobj->GetInt("Z_Layer")/100;
  rot = 0.0;
  
  waterAnimTimer = 0.0;
  
  if (StrCmp(aobj->GetString("AnimStart"), "Three")) waterAnimTimer = 0.25;
  if (StrCmp(aobj->GetString("AnimStart"), "Six")) waterAnimTimer = 0.50;
  if (StrCmp(aobj->GetString("AnimStart"), "Nine")) waterAnimTimer = 0.75;
  if (StrCmp(aobj->GetString("AnimStart"), "Twelve")) waterAnimTimer = 0.0;        
    
}


void CWaterWave::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{

    if (Type == M_DRAWQUEUE)
    {  
        AddSpriteEx( z, x, y, golfWaterWave->Index, waterAnimTimer, rot, scale);

    }
    if (Type == M_FRAME)
    {  
        if (waterAnimTimer < 1.0) {
                  waterAnimTimer += 0.03;
               }
        else waterAnimTimer = 0.0;

    }

}

//Dolphin
CDolphin::CDolphin(CEditorObject* aobj)
{
  super(Root, GS);
  Scale = aobj->GetInt("Size");
  scale = Scale/100.0;
  x = aobj->PosX;
  y = aobj->PosY - 1*scale;
  z = 16.0 + aobj->GetInt("Z_Layer")/100;
  rot = 0.0;
  
  dolphinAppearTimer = 0.0;
  dolphinRotationTimer = 0.0;
    
}


void CDolphin::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{

    if (Type == M_DRAWQUEUE)
    {  
        AddSpriteEx( z, x, y, golfDolphin->Index, 0.0, dolphinRotationTimer, scale);

    }
    if (Type == M_FRAME)
    {  
        if (dolphinAppearTimer < 150.0) {
               dolphinAppearTimer += 0.1;
               }else{
                dolphinRotationTimer = -51;
               }


        if (dolphinRotationTimer > -50.0) {
                  dolphinRotationTimer -= 0.08;
               }else dolphinRotationTimer = 0.0;
           
    }

} 

//Arrows Direction
CDirectionArrow::CDirectionArrow(CEditorObject* aobj)
{
  super(Root, GS);
  Scale = aobj->GetInt("Size");
  scale = Scale/100.0;
  x = aobj->PosX;
  y = aobj->PosY + 2*scale;
  z = 16.0 + aobj->GetInt("Z_Layer")/100;
  rot = 0.0;
  
      if (StrCmp(aobj->GetString("DirectionArrows"), "Left1")) directionArrowSprite = arrowSide1->Index + 262144;
      if (StrCmp(aobj->GetString("DirectionArrows"), "Right1")) directionArrowSprite = arrowSide1->Index;
      if (StrCmp(aobj->GetString("DirectionArrows"), "Left2")) directionArrowSprite = arrowSide2->Index + 262144;
      if (StrCmp(aobj->GetString("DirectionArrows"), "Right2")) directionArrowSprite = arrowSide2->Index;
      if (StrCmp(aobj->GetString("DirectionArrows"), "LeftDepth1")) directionArrowSprite = arrowSide3->Index + 262144;
      if (StrCmp(aobj->GetString("DirectionArrows"), "RightDepth1")) directionArrowSprite = arrowSide3->Index;
      if (StrCmp(aobj->GetString("DirectionArrows"), "LeftUp1")) directionArrowSprite = arrowUp1->Index + 262144;
      if (StrCmp(aobj->GetString("DirectionArrows"), "RightUp1")) directionArrowSprite = arrowUp1->Index;
      if (StrCmp(aobj->GetString("DirectionArrows"), "AllDirections")) directionArrowSprite = arrowsAllDirections->Index;
