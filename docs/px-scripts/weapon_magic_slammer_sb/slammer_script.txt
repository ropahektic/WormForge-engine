require utils, p_sprite_builder, sprite_curtis_254, sprite_curtis_458;

CSprite* magics_aiming;
CSprite* smagics_aiming;

void slammer_script::InitGraphic()
{
    CFile *f;
    f = GetAttachment("magics.png");
    magics_aiming = LoadSprite(f, 12, 0);
    f = GetAttachment("smagics.png");
    smagics_aiming = LoadSprite(f, 16, 0);
}

//---------------------------------MAGIC SLAMMER------------------------------//

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) {
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Magic Slammer";
     sprite = magics_aiming->Index;
     Params->hand_type = 1;
     Params->hand_scale = 1.3;
     Params->hand_radius = -20.5;
     Params->hand_rotation = -0.0;
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;
     Params->weap_rotation = 0.5;
     Params->weap_radius = -14.5;
     Params->animate = true;
     Params->anim_speed = 0.90;
     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;
  if (Type == M_FRAME){
     if (weap->CheckName("Magic Slammer")) {
         if (magics_anim) {
              local sprite = magics_aiming->Index;
              WeapSpriteParams* Params = new WeapSpriteParams;
              Params->animate = true;
              Params->anim_speed = 0.80;
              new WeapSprite(this, sprite, Params);
              weap->launch.anim.spriteIndex = 0;
              magics_anim = false;
         }
     }
  }
}

override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;
  magics_anim = true;
}
