require utils;

CBitmap32* HatchetBitmap;
CSoundFile* hatchetimpactsnd;
CSoundFile* hatchetlaunchsnd;

void Hatchet_script::InitGraphic()
{	
    local spr       = GetAttachment("hatchet2.png");
    HatchetBitmap     = LoadImage(spr);
}

void Hatchet_script::FirstFrame()
{	
	CFile *f;
	f = GetAttachment("hatchet_impact.wav");	hatchetimpactsnd = new CSoundFile(f);
	f = GetAttachment("hatchet_launch.wav");	hatchetlaunchsnd = new CSoundFile(f);
}

override void CMissile::ExplodeAt(fixed x,fixed y)
{
 
 if(weap->CheckName("Hatchet") == true)
 {
           CGObject *Hatchetobj;
     	   for(int i = 0; i < Env->Objs.Count; i += 1)
	   { 
		Hatchetobj = CGObject(Env->Objs.Objs[i]);
		if(Hatchetobj == this) continue;
		if(Hatchetobj == NullObj) continue;
		if(Hatchetobj->IsMaterial == false) continue;                 			
			if(Hatchetobj->ClType == OC_Worm || Hatchetobj->ClType == OC_Mine || Hatchetobj->ClType == OC_OilDrum || Hatchetobj->ClType == OC_Crate)
			{
				local trg = CGObject(Hatchetobj);
                                if(x > trg->PosX-20 && x < trg->PosX+20 && y > trg->PosY-20 && y < trg->PosY+20)
                                {
                                HatchetDamage(trg);
                                super;
                                return;
                                }
			}
           }
           
           if(GetCurrentWorm() != NullObj)
           {
                   if(x > GetCurrentWorm()->PosX-25 && x < GetCurrentWorm()->PosX+25 && y > GetCurrentWorm()->PosY-25 && y < GetCurrentWorm()->PosY+25)
                   {
                   HatchetDamage(GetCurrentWorm());
                   super;
                   return;
                   }            
           }
           
           HatchetToLandscape(x,y);
			hatchetimpactsnd->Play(1.0,0.0,false);
 }
  
 super;  
}

void CMissile::HatchetDamage(CGObject* target)
{
                     local damage = 1;
                     if(weap->CheckName("Hatchet") == true)
                     {damage = 16;}

                     target->SpX = SpX/4;
                     target->SpY = SpY/4; 
                     
                     CMessageData msgdamage;
                     msgdamage.params[0] = 3;
                     msgdamage.params[1] = PosX;
                     msgdamage.params[2] = PosY;
                     msgdamage.params[3] = SpX/2;//doesnt work for some reason..
                     msgdamage.params[4] = SpY/2;
                     msgdamage.params[5] = damage;
                     msgdamage.params[6] = 0;
                     target->Message(NullObj, M_GUNEXP, 1032, &msgdamage); 
}

void CMissile::HatchetToLandscape(fixed x, fixed y)
{
local bmp = HatchetBitmap;
local top = bmp->top;
int mtp = 0;

      if(SpX == 0.0 && SpY < 0.0){mtp = 0;}
      if(SpX > 0.0 && SpY == 0.0){mtp = 8;}
      if(SpX == 0.0 && SpY > 0.0){mtp = 16;}
      if(SpX < 0.0 && SpY == 0.0){mtp = 24;}
      
      if(SpX > 0.0 && SpY < 0.0)
      {
      mtp = 3*SpX/sqrt(SpY*SpY);
      if(mtp>7){mtp = 7;}
      if(mtp<0){mtp = 0;}
      }
      
      if(SpX > 0.0 && SpY > 0.0)
      {
      mtp = 8+3*SpY/SpX;
      if(mtp>15){mtp = 15;}
      }
      
      if(SpX < 0.0 && SpY > 0.0)
      {
      mtp = 16+3*sqrt(SpX*SpX)/SpY;
      if(mtp>23){mtp = 23;}
      }
      
      if(SpX < 0.0 && SpY < 0.0)
      {
      mtp = 24+3*sqrt(SpY*SpY)/sqrt(SpX*SpX);
      if(mtp>31){mtp = 31;}
      }

top = mtp*60;
bmp->top = top;

bmp->bottom = bmp->top+60;

GG->land->Draw(bmp, x-30, y-(30+bmp->top));
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
  if(Weap->CheckName("Hatchet"))
     {
     hatchetlaunchsnd->Play(2.0, 0.0, false);
     }
     
super;
}
