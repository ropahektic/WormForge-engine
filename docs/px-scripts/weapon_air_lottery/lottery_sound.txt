CSoundFile* payment;
 
void lottery_sound::Init()
{
	payment       = new CSoundFile(GetAttachment("Payment Sound.wav"));
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
  if(Weap->CheckName("Lottery Strike"))
  {  
             payment->Play(1.5, 0.0, false);
  }
  super;
}
