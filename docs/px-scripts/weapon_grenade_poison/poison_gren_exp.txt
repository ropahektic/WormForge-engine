{
    glass_break = new CSoundFile(GetAttachment("glass_break.wav"));
    cluster_priming = new CSoundFile(GetAttachment("cluster_priming.wav"));
}

void CPoison::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
	if(Type == M_DRAWQUEUE)
    	{
        SetColorMod(RGB(250, 110, 12), 1);
        super;

        if(gframe % 10 == 0) super;
            
        ClearColorMod();
	    }
    else super;
}

void CMissile::DoPoison(int x, int y, int power, int team)
{
local radius = 3;

local n = 40;
for(int i = 0; i <= n; i+=1)
	{
	px1 = (cos(2 * MATH_PI * i / n) * radius + 0.5) + x;
	py1 = (sin(2 * MATH_PI * i / n) * radius + 0.5) + y;
	local poison = new CPoison(Root->GetObject(25, 0), px1, py1, power, team);
	}
}

void CMissile::LaunchPoisonClustlets(int ClPosX, int ClPosY, int ExplosionTimer)
{
    CShootDesc SDesc;
    zero(&SDesc);
    SDesc.Team = ClWT;
    SDesc.Worm = ClWN;
    SDesc.X = ClPosX;
    SDesc.Y = ClPosY;
    SDesc.Delay = 5;
    local ldata;
    local poisoncluster;
    local memory;
    local sprite;
    ldata = GetWeaponByName("Poison Grenade");
    memory = ldata;

	sprite = ldata->launch.anim.spriteIndex;
     
    local remember = ldata->launch.timeBeforeExplosion;
    ldata->launch.anim.spriteIndex = p_cluster->Index;
	ldata->launch.anim.trailIndex = 132;
    ldata->launch.anim.trailPower = 10;
    ldata->launch.anim.trailSpeed = 100;
    ldata->launch.timeBeforeExplosion = ExplosionTimer;
    ldata->launch.explodeInto = WExplode_Nothing;
    ldata->launch.countdownFrom = 0;
    ldata->launch.explosion.flags = 1;
     
    SDesc.SpX = RandomFloat(2, -2);
    SDesc.SpY = RandomFloat(-5, -7);
    poisoncluster = new CPoisonCluster(Root->GetObject(25, 0), &ldata->launch, &SDesc);
    WeapSpriteParams* Params = new WeapSpriteParams;
    new WeapSprite(poisoncluster, sprite, Params);

	SDesc.SpX = RandomFloat(2, -2);
    SDesc.SpY = RandomFloat(-5, -7);
    poisoncluster = new CPoisonCluster(Root->GetObject(25, 0), &ldata->launch, &SDesc);
    new WeapSprite(poisoncluster, sprite, Params);
    
	SDesc.SpX = RandomFloat(2, -2);
    SDesc.SpY = RandomFloat(-5, -7);
    poisoncluster = new CPoisonCluster(Root->GetObject(25, 0), &ldata->launch, &SDesc);
    new WeapSprite(poisoncluster, sprite, Params);

	SDesc.SpX = RandomFloat(2, -2);
    SDesc.SpY = RandomFloat(-5, -7);
    poisoncluster = new CPoisonCluster(Root->GetObject(25, 0), &ldata->launch, &SDesc);
    new WeapSprite(poisoncluster, sprite, Params);

	SDesc.SpX = RandomFloat(2, -2);
    SDesc.SpY = RandomFloat(-5, -7);
    poisoncluster = new CPoisonCluster(Root->GetObject(25, 0), &ldata->launch, &SDesc);
    new WeapSprite(poisoncluster, sprite, Params);

	ldata->launch.timeBeforeExplosion = remember;
    ldata = memory;
}

override void CMissile::ExplodeAt(fixed x,fixed y)
{
super;

if (weap != NullObj && weap->CheckName("Poison Grenade"))
    {
    LaunchPoisonClustlets(x, y, 4000);
    cluster_priming->Play(1.0, 0.0, false);
    }

if (weap != NullObj && weap->CheckName("Poison Flask"))
	{
		for(i = 0; i < Env->Objs.Count; i += 1)
		{
    		obj = CGObject(Env->Objs.Objs[i]);
    		if (obj == NullObj) continue;
    		if (obj->ClType == OC_Worm)
        		{
        		CWorm *worm = CWorm(obj);
				DoPoison(x, y+10, 10, worm->WormTeam);
				}
		}
    	glass_break->Play(1.0, 0.0, false);
	}
}

void CPoisonCluster::ExplodeAt(fixed x,fixed y)
{
for(i = 0; i < Env->Objs.Count; i += 1)
	{
    obj = CGObject(Env->Objs.Objs[i]);
    if (obj == NullObj) continue;
    if (obj->ClType == OC_Worm)
        {
        CWorm *worm = CWorm(obj);
		DoPoison(x, y+10, 10, worm->WormTeam);
		}
	}
}
