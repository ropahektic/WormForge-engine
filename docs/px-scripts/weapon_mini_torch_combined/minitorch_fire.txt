//Sequence script by Lucho/reddaz, no need for separate weap.
require utils;   

CWeapon* torchFlame;

void minitorch_fire::FirstFrame()
{
torchFlame = GetWeaponByName("Mini-Torch");
}

//---------------CWorm Class Overrides---------------//
override CWorm::CWorm(CObject * Parent, int aTeam, int aIndex, CWormParams * params)
{
    super;
    TFShotsFired = 0;
    torchFShooting = false;
}

override void CWorm::FireFinal(CWeapon* Weap, CShootDesc* Desc)
{
    if (Weap->CheckName("Mini-Torch") )
    {
       if (TFShotsFired<=25)     //Quantity of shots
       {            
         for (int curShots = TFShotsFired; TFShotsFired==curShots; TFShotsFired++) //For some reason PShotsFired++ normally doesn't work.           
         Desc->SpX = (TurnSide * (0.5 - sqrt((0.5 - FireAngle)*(0.5 - FireAngle)))) * 20;
         Desc->SpY = 20*(0.5 - FireAngle);
         Desc->SpY-(RandomInt(-11,11));   //Simulate gun spread
         super(torchFlame, Desc);    //This method crashes with homing arrows, however, i don't care.
         torchFShooting=true;
         return;   //Return makes it loop.
       }
       torchFShooting=false;   //Don't reset shots here, it'll just keep looping. 
       return;
    }        
    torchFShooting=false; //Shouldn't be neccessary, however i'm insecure.
    super;
}
     
// --- Main Logic Loop --- //
override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
super;
  if (Type == M_FRAME)
  {
    if(CurWeapon->CheckName("Mini-Torch") == true)
    {
       if (torchFShooting)
       {
           // Persist animation.  
           nShoots = 1;
           nShooted = 0;
       }
    }
  }
  if (Type==M_SETWEAPON)
  {
     if(CurWeapon->CheckName("Mini-Torch") == true)
     {TFShotsFired=0;torchFShooting=false;}       //Reset variables
  }  
}
