sprite_curtis_261, sprite_curtis_451, sprite_curtis_452;

CSprite* explosivea_bullet;

CSprite* windya_aiming;
CSprite* windya_bullet;

CSprite* rebounder_aiming;

CSprite* fireb_bullet;

void curtis_bows::InitGraphic()
{
    CFile *f;

    f = GetAttachment("spb_explosivea.png");
    explosivea_aiming = LoadSprite(f, 8, 0);
    f = GetAttachment("explosivea.png");
    explosivea_bullet = LoadSprite(f, 1, 0);

    f = GetAttachment("spb_windya.png");
    windya_aiming = LoadSprite(f, 8, 0);
    f = GetAttachment("windya.png");
    windya_bullet = LoadSprite(f, 8, 0);

    f = GetAttachment("spb_rebounder.png");
    rebounder_aiming = LoadSprite(f, 8, 0);
    f = GetAttachment("spb_hominga.png");
    hominga_aiming = LoadSprite(f, 8, 0);

    f = GetAttachment("spb_multib.png");
    multib_aiming = LoadSprite(f, 12, 0);
    f = GetAttachment("multib.png");
    multib_bullet = LoadSprite(f, 12, 0);

    f = GetAttachment("spb_fireb.png");
    fireb_aiming = LoadSprite(f, 8, 0);
    f = GetAttachment("fireb.png");
    fireb_bullet = LoadSprite(f, 8, 0);
}


//---------------------------------EXPLOSIVE ARROW (POWERED)------------------//

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) {
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Explosive Arrow";
     sprite = explosivea_aiming->Index;
     Params->draw_hand = false;
     Params->hand_radius = 0.0;
     Params->hand_rotation = 0.0;
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;
     Params->weap_rotation = 1.0;
     Params->weap_radius = 3.5;
     Params->animate = true;
     Params->anim_speed = 2.00;
     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;
  if (Type == M_FRAME){
     if (weap->CheckName("Explosive Arrow")) {
         if (explosivea_anim) {
              local sprite = explosivea_bullet->Index;
              WeapSpriteParams* Params = new WeapSpriteParams;
              new WeapSprite(this, sprite, Params);
              weap->launch.anim.spriteIndex = 0;
              explosivea_anim = false;
         }
     }
  }
}
override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;
  explosivea_anim = true;
}
