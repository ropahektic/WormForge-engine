void CHealthZone::DamageHealth()
{
    local worm_active;
    if (health_zone->activator == NullObj) return;
    if (health_zone->activator->ClType == OC_Worm)
    {
       worm_active = CWorm(health_zone->activator);

       local worm = worm_active->WormNumber;
       local team = worm_active->WormTeam;
       local hp_now = GS->Info.GetWormHealth( team, worm);
       if (hp_now > minHP) GS->Info.SetWormHealth( team, worm, hp_now - 1);
       if (worm_active->ObjState == WS_JETPACK) worm_active->JetFuel -= 5*65536;
    }
}

void CHealthZone::AddHealth()
{
    local worm_active;
    if (health_zone->activator == NullObj) return;

    if (health_zone->activator->ClType == OC_Cross)
    {
      if (health_zone->activator->SpX == 0 && health_zone->activator->SpY == 0)      // fixes error with moving grave, aslo errors and game crash with portals.
      {
       PxDeadWorm *deadworm = DW_M->DeadWormFromGrave(health_zone->activator);
       if(deadworm != PxDeadWorm(NullObj))
       {
          deadworm->Resurrect(1);
       }
       else
       {
          ShowMessage("Grave has no bound worm, could not resurrect!");
       }
      }		    
    }
    else if (health_zone->activator->ClType == OC_Worm)
    {
       worm_active = CWorm(health_zone->activator);
       local worm = worm_active->WormNumber;
       local team = worm_active->WormTeam;
       local hp_now = GS->Info.GetWormHealth( team, worm);
       if (hp_now < maxHP) GS->Info.SetWormHealth( team, worm, hp_now + 1);
       if (worm_active->ObjState == WS_JETPACK) worm_active->JetFuel += 5*65536;
       if (worm_active->SickDamage > 0) worm_active->SickDamage--;  // low down sickness untill totally healthy.
    }
    
}

void CHealthZone::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
 if (Type == M_SKIPTURN) return;
 if (Type == M_TURNEND) 
 {
    turns--;
    start_be_inactive = true;
 }
  if (Type == M_TURNBEGIN) 
  {
      if (Dead == false)
      {
      start_be_inactive = false;       // Deactivate timer if turn started earlier then timer deactivated portal.
      inactive_timer = 0;              // Usefull if some1 is afk. Becouse otherwise timer would disable portal`s activity on some1`s turn start :D
      active = true;
      }
 }
 if (Type == M_FRAME)
 {
   timer++;
   if (turns <= 0) 
   {
      Dead = true;
      active = false;
      Free(true);
      return;
//      health_zone->Free(true);
   }
   if (timer % fps) return;
   if (active == false) return;
        if (health_zone == NullObj)
        {
            health_zone = SearchForWeapZone(zone_name);
            return;
        }
        else 
        if (health_zone->active)
        {
           if (health_zone->activator != NullObj)
           {  
             if (start_be_inactive)
             {
                if (health_zone->activator->SpX == 0 && health_zone->activator->SpY == 0)   // bug fixes
                {
                   if (Healing) AddHealth();
                   else DamageHealth();
                }
             } else
             if (Healing) AddHealth();
             else DamageHealth();
           }
        }
   if(start_be_inactive)
   {
      inactive_timer++;
      if (inactive_timer > 7)  // as soon as fps = 50 (1 sec), 7 here = 7 secs.
      {
         active = false;         
         start_be_inactive = false;
         inactive_timer = 0;
      }
   }
 }
}

void CBlast::Boom()
{
    DoExplosion( x, y, PushPower, Damage, unkB, Team);
    if (blast_zone->activator != NullObj)
    {
       if (blast_zone->activator->ClType == OC_Mine)
       {
           CMine* blast_mine = CMine(blast_zone->activator);
           if(blast_mine->Radius > 0 && blast_mine->Active == 0 && blast_mine->Prefuse <= 0)
           {
              if(blast_mine->Fuse < 0)
              {
                 blast_mine->Fuse = RandomInt(1000, 5000);
              }
              blast_mine->Active = 1;
              blast_mine->nPulses = blast_mine->Fuse / 250;
           }
       }
    }
}

void CBlast::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
if (Type == M_TURNEND) turns--;
 
if (Type == M_FRAME)
 	{
   	timer++;
   	if (turns <= 0)
   		{
      	Dead = true;
      	worknow = false;
      	Free(true);
      	return;
   		}
   		
   	if (timer % fps) return;

	if (blast_zone == NullObj)
        {
        blast_zone = SearchForWeapZone(zone_name);
        return;
        }
    else if (worknow)
        {
        if (blast_zone->active)
            {
            if (blast_zone->activator != NullObj)
              	{
                if (blast_zone->activator->ClType == OC_Worm || blast_zone->activator->ClType == OC_OilDrum || blast_zone->activator->ClType == OC_Mine || blast_zone->activator->ClType == OC_Crate || blast_zone->activator->ClType == OC_Arrow || blast_zone->activator->ClType == OC_Missile)
                	{
                  	Boom();
                  	worknow = false;
                	}
              	}
            }
        }

     if (blast_zone != NullObj)
     	{
        if (blast_zone->active == false) worknow = true;
     	}
 	}
}




override void CMissile::ExplodeAt(fixed x,fixed y)
{
    super;
    if (weap->CheckName("Spot Hider"))
    {
    SpotHiderParams->useObj = false;
    SpotHiderParams->turns = 4;
    SpotHiderParams->sprite = blackMask->Index;
    SpotHiderParams->scale = SH_scale;
    SpotHiderParams->speed = 25;
    SpotHiderParams->duration = 15;
    SpotHiderParams->ZPlane = 4.95;
    SpotHiderParams->color = ARGB(100, 0, 0, 0);
    SpotHiderParams->blend_type = 6;
    new CAnimation (Root, x, y, NullObj, SpotHiderParams);
    return;
    }
    if (weap->CheckName("Blast Zone"))
    {
        zone_number++;
        local number = MakeNumberString("blast_zone_", zone_number, "");
        wzone = new CWZone(x, y, 45, 45, 2, false, number, BlastTurns, false);
        wBlast = new CBlast(number, x, y, 30, 30, ClWT, BlastTurns);
    }
    if (weap->CheckName("Health Zone"))
    {
        zone_number++;
        local hp_status = HealthPlusMis;
        local number = MakeNumberString("health_zone_", zone_number, "");
        w2zone = new CWZone(x, y, 40, 30, 2, false, number, HealthTurns, false);
        wHealth = new CHealthZone(number, x, y, 40, 40, ClWT, HealthTurns, hp_status);
    }
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_FRAME)
    {
        if (weap->CheckName("Health Zone"))
        {
            imCheckMis = this;
            HealthPlusMis = HealthPlus;
            if (not_too_much_particles) 
            {
               if (HealthPlusMis) do_sprite_trail_effects(108, PosX, PosY);
               else do_sprite_trail_effects(106, PosX, PosY);
            }
        }
        if (weap->CheckName("Blast Zone"))
        {
            imCheckMis = this;
            if (not_too_much_particles) do_sprite_trail_effects(106, PosX, PosY);
        }                 
    }                                    
    super;
}
