CSprite* blast_zone_aiming;                                                 
CSprite* blast_zone_bullet;                                                 
                                                                                 
void blast_zone_anim::InitGraphic() {
    CFile* f;                                                                    
    f = GetAttachment("blast_zone_aiming.png");                             
    blast_zone_aiming = LoadSprite(f, 1, 0);                                
    f = GetAttachment("blast_zone_bullet.png");                                          
    blast_zone_bullet = LoadSprite(f, 1, 0);                                
}                                                                                
                                                                                 
override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) { 
 super;                                                                          
  if (Type == M_FRAME) {                                                         
    if (gframe == 1) {                                                           
     local weapName; local sprite;
///////......................................................................\\\\\\
//
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Blast Zone";                                               
     sprite = blast_zone_aiming->Index;   
     Params->fix_flip_rotation = true;                                                                                        
     Params->hand_rotation = 0.5;
     Params->hand_radius = 4.0;
     Params->hand_radial_rotation = 0.0;
//     Params->weap_rotation =  1.0;
     Params->hand_type = 3;
     Params->hand_scale = 0.8;               ///// I recommend don`t change it. //////   //                                                       
//
///////``````````````````````````````````````````````````````````````````````//////                                                    
     local WA = new WormSprite(weapName, sprite, Params);
    }                                                                            //
  }                                                                              //
}                                                                                //

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;                                                                           //
  if (Type == M_FRAME){                                                          //
     if (weap->CheckName("Blast Zone")) {                                
         if (blast_zone_anim2) {                                              
            local sprite = blast_zone_bullet->Index;                      
            WeapSpriteParams* Params = new WeapSpriteParams;
            new WeapSprite(this, sprite, Params);             
            weap->launch.anim.spriteIndex = 0;
            blast_zone_anim2 = false;                                         
         } 
     }                                                                                                                                                                                                                          
  }                                                                              //
}                                                                                //
override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;                                                                           //
  blast_zone_anim2 = true;                                                    
}
