void CGolfBall::Zap(int zaptype)
{
	float mul;
	
	if(zaptype == 0)
	{
		ZapTurnsRemaining = 1;
		mul = 1.0;
	}
	else
	{
		mul = 0.8;
	}
	
	Wake();
	
	float ang = RandomFloat(-0.1, 0.1);
	float speed = RandomFloat(5.0, 10.0);
	SpX = sin(ang) * speed * mul;
	SpY = -cos(ang) * speed * mul;
}

void CGolfBall::DoAnimation()
{
/*ZAP*/	if(ZapTurnsRemaining > 0)
	{
		SparkCycle = SparkCycle + SparkAnimSpeed;
		if(SparkCycle > SparkCycleDuration) SparkCycle = SparkCycle - SparkCycleDuration;
	}
}
void CGolfBall::UpdateZPlane()
{
	// Recalculate Z plane periodically to make sure objects are drawn properly when overlapping each other
	if(NextRecalcZPlane <= 0)
	{
		local matches = 0;
		
		for(local i = 1; i < Env->Objs.Count; i+=1)
		{
			local obj = Env->Objs.Objs[i];
			
			if(obj == NullObj) continue;
			if(obj == this) continue;
			if(obj->ClType == EObjectClass(71))
			{
				local ball = CGolfBall(obj);
						
				float dX = ball->PosX - PosX;
				float dY = ball->PosY - PosY;
				float r2 = dX*dX + dY*dY;
					
				if(r2 < 900.0)
				{
					if(ball->ZPlane == ZPlane)
					{
						ball->ZPlane = ball->ZPlane + 1;
					}
					matches = matches + 1;
				}
			}
		}
		
		if(matches == 0)	ZPlane = 5;
		
		NextRecalcZPlane = 200;
	}
	else
	{
		NextRecalcZPlane = NextRecalcZPlane - 1;
	}
}

void CGolfBall::UpdateRotation()
{
	float abs_speedX = RealSpX;
	float abs_speedY = RealSpY;
	
	if(abs_speedX < 0) abs_speedX = 0-abs_speedX;
	if(abs_speedY < 0) abs_speedY = 0-abs_speedY;
	
	float rot = SpinMultiplierX * abs_speedX + SpinMultiplierY * abs_speedY;
	
	if(rot > MaxSpinSpeed) rot = MaxSpinSpeed;
	if(RealSpX<0) rot = 0-rot;
	
	ApplyRotation(rot);
}
//change this to match with backspin
void CGolfBall::ApplyRotation(float r)
{
	Rotation += r;        // += r; for front spin
	while(Rotation > 3.141593)		Rotation -= 6.283186;// 6.283186       >
	while(Rotation <= 0 - 3.141593)	Rotation += 6.283186;
}

void CGolfBall::TakeExplosionDamage(fixed x, fixed y, int dmg)
{
	if(dmg == 0) return;
	
	float fdmg = dmg;
	float dx = PosX - x;
	float dy = PosY - y;
	
	float distsqr = dx*dx + dy*dy;
	float fdmgsqr = fdmg*fdmg*1.25*1.25;
	
	fdmg = (1.0-distsqr/fdmgsqr) * fdmg;
	dmg = fdmg;
	
	if(dmg > 0) TakeDamage(0, dmg);
}

void CGolfBall::ProcessBallOnBallCollision(float Speed)
{
	CGObject *obj;
	
	for(int i = 0; i < Env->Objs.Count; i += 1)
	{
		obj = CGObject(Env->Objs.Objs[i]);
		if(obj == this) continue;
		if(obj == NullObj) continue;
		if(obj is CGolfBall) continue;
		if(obj->ClType == EObjectClass(71)) continue;
		if(obj->ClType == OC_Worm) continue;
		if(obj is CWorm) continue;
                
                if(obj->ClType == ClType)
		{
			if(ColMask->Check(PosX, PosY, obj->ColMask, obj->PosX, obj->PosY))
			{
                        
                        if (justDropped)
		        {
		            Sleep();
		            justDropped = false;
		        }
		        if (RespawnMe)
		        {
                            Sleep();
                            RespawnMe = false;
		        }
                                float ForceX = obj->PosX - PosX;
				float ForceY = obj->PosY - PosY;
				float dist = sqrt(ForceX*ForceX + ForceY*ForceY);
				
				if(dist > 0)
				{
					ForceX = 0.6*sqrt(Speed) * ForceX / dist;
					ForceY = 0.6*sqrt(Speed) * ForceY / dist;
					
					CMessageData msg;
					msg.params[0]	= 0;
					msg.fparams[1]	= PosX;
					msg.fparams[2]	= PosY;
					msg.fparams[3]	= ForceX;
					msg.fparams[4]	= ForceY;
					msg.params[5]	= 1;
					msg.params[6]	= 0;
					
					obj->Message(this, M_GUNEXP, 1032, &msg);
				}
			}
		}
	}
}

void CGolfBall::PhysicsThink()
{
	if(IsAsleep)
	{
		PosX = SleepX;
		PosY = SleepY;
		RealSpX = 0.0;
		RealSpY = 0.0;
		SpX = 0.0;
		SpY = 0.0;
		Stopped = 1;
		InMidAir = 0;
		if (GolfMessage != NullObj && GolfMessage->alpha <= 0) animate = false;
		if (ballOwner != NullObj)
                {
                    if (ballOwner->golfClub == 4) ballOwner->golfPowerChanger = 1.0;
                }
	}
	else
	{
		if (justDropped || RespawnMe) 
                {
                   SpX = 0.0;
//                   if (SpY < 0) SpY = 0;
                }
		
                if (ballOwner != NullObj)
		{
                   CheckGolfHole();
		   CheckGolfZones();
                }
                float Speed = sqrt(RealSpX*RealSpX + RealSpY*RealSpY);
		
		fixed SpX2 = SpX;
		fixed SpY2 = SpY + Env->Gravity;
		
		float bounciness;
		float friction;
		float inertiaX;
		float inertiaY;
		
		if(IsLowSpeed)
		{
			bounciness = LowSpeedBounciness;
			friction = LowSpeedFriction;
			inertiaX = LowSpeedInertiaX;
			inertiaY = LowSpeedInertiaY;
//			if (playApplause) 
//                        {
//                           PlaySound( 2, 0.0, 0.0, 1.0);  // applause
//                           playApplause = false;
//                        }
//			WindFactor = 0.02;
		}
/*ZAP*/		else if(ZapTurnsRemaining > 0)
		{
			bounciness = ZappedBounciness;
			friction = ZappedFriction;
			inertiaX = ZappedInertiaX;
			inertiaY = ZappedInertiaY;
		}
		else
		{
			bounciness = PhysBounciness;
			friction = PhysFriction;
			inertiaX = PhysInertiaX;
			inertiaY = PhysInertiaY;
//			WindFactor = 0.1;  // seems bad idea. works awful
		}
		
		if(SpX2 == 0)
		{
			if(SpY2 == 0)
			{
				SpY2 = SpY + 0.1;
			}
		}
		if ((abs(SpY) + abs(SpX)) > 5) WindFactor = 0.09; 
                else WindFactor = 0.0; 
		
		// Collision detection
		if(Speed > MinDamageSpeed)
		{
			// Enough speed for knocking stuff out of the way, ignore worms
			TraceMaskEx(this, BounceColMask, PosX, PosY, PosX + SpX2, PosY + SpY2, NoWormsColGroup, Res);
		}
		else
		{
			// Regular traces, takes anything solid into consideration
			TraceMaskEx(this, BounceColMask, PosX, PosY, PosX + SpX2, PosY + SpY2, DefaultColGroup, Res);
		}
		if(Res->StartsSolid)
		{
			TraceMaskEx(this, BounceColMask, PosX, PosY, PosX + SpX2, PosY + SpY2, NoWormsColGroup, Res);
			if(Res->StartsSolid)
			{
				// The object is stuck inside something solid, try to unblock it
				TraceMaskEx(this, BounceColMask, PosX, PosY, PosX - 8 * SpX2, PosY - 8 * SpY2, NoWormsColGroup, Res);  // 8 is for 16 pix diameter balls
				TraceMaskEx(this, BounceColMask, Res->LeftSolidX, Res->LeftSolidY, Res->LeftSolidX + 8 * SpX2, Res->LeftSolidY + 8 * SpY2, NoWormsColGroup, Res);
			}
		}
		
		// Couldn't unblock it :(
		if(Res->StartsSolid)
		{
			TraceMaskEx(this, BounceColMask2, PosX, PosY, PosX - 8 * SpX2, PosY - 8 * SpY2, NoWormsColGroup, Res);
			TraceMaskEx(this, BounceColMask2, Res->LeftSolidX, Res->LeftSolidY, Res->LeftSolidX + 8 * SpX2, Res->LeftSolidY + 8 * SpY2, NoWormsColGroup, Res);
			if(Res->StartsSolid)
			{
				//EnteredSolid = true;
				//SpY = SpY - 4.0;
				Sleep();
			}
		}
		
		// Collision processing
		if(Res->Hit)
		{
			First = false;
			// Calculating tangent and normal initial forces
			float SpT = (SpX2 * Res->TangentX) + (SpY2 * Res->TangentY);
			float SpN = (SpX2 * Res->NormalX ) + (SpY2 * Res->NormalY );
			
			// Move the object 1 pixel away from the collision point
			PosX = Res->PreHitX;
			PosY = Res->PreHitY;
			
			// Calculating bounciness and friction forces
			float ForceX = 0 - SpN * Res->NormalX * (bounciness + 1.0) - SpT * Res->TangentX * friction;
			float ForceY = 0 - SpN * Res->NormalY * (bounciness + 1.0) - SpT * Res->TangentY * friction;
			
			// Calculating work of the resulting force relative to the current speed
			float W = (ForceX * SpX2) + (ForceY * SpY2);
			
			SpX = SpX + ForceX;
			SpY = SpY + ForceY;
			
			float sp = Speed;
			if(sp > 1.5)
			{
				for(int i=0 ; i<4 ; i+=1)
				{
					local p = new PxParticle(132, PosX, PosY);
					p->SetLifeTime(30);
					p->SetRandomVelocity(sp*0.5, sp*1.2);
					p->SetAirResistance(0.5);
					p->SetStartAlpha(255);
					p->SetEndAlpha(255);
					p->SetColor(60, 255, 60);
					p->SetStartSize(0.5,0.5);
				}
			}
		}
		
		SpX = SpX + inertiaX * (SpX - LastSpX); //+ BackspinLift;
		SpY = SpY + inertiaY * (SpY - LastSpY);// + BackspinLift;
		RealSpX = PosX - LastPosX;
		RealSpY = PosY - LastPosY;
		
		local bVolume;
		if (SpY > 1) 
                {
                   bounceChecker = true;   // make bounce sound only when ball`s speed over this value
                }
		if (SpY < 0 && bounceChecker) 
                {
                   bVolume = abs(SpY/5) + abs(SpX/5);
                   if (bVolume < 0.4) bVolume = 0.4;
                   else if (bVolume > 2.5) bVolume = 2.5;
                   GolfBounceSound->Play( bVolume, 1.0, false);
                   bounceChecker = false;
                }
		
		
	/*	float InitialBackspin = sqrt(0.24 + ((sqrt(Speed * Speed)) - (sqrt(Speed * Speed))));    //rpm value, + 0 alsowat
                float BackspinLift = (InitialBackspin / Speed);      //the more speed, the less backspin creates lift.
                PhysInertiaY = PhysInertiaY + BackspinLift;//(sqrt(SpY * SpY)) / BackspinLift;
                
                
                /*the more speed, the less backspin creates lift. 
when the ball slows down, the ball is still spinning as fast (at least slow down is not proportionally to speed)
so the backspin effect is more important
so it creates a ballooning effect*/
                                
                
                
                // Should solve an issue where the object keeps rolling(DRUMSTICK:to the left?) on a perfectly horizontal surface,
		// even though friction should prevent it from doing so
		if(IsLowSpeed)
		{
			if(RealSpY > -1.0)
			{
				if(RealSpY < 1.0)
				{
					if(SpX > -0.3)      // cheap but effective
					{
						if(SpX < 0.0) SpX += 0.025;
						else if (SpX < 0.0001) SpX = 0;
					
                                        }
				}
			}
		}
	/*	
		if(IsLowSpeed)
		{
                        if((SpX * SpX) < (LastSpX * LastSpX))     //if there is a deceleration in X (neat! result is always positive even if negative values)
                        {
                                if((RealSpX * RealSpX) < ((RealSpX * RealSpX) + ((PosX - LastPosX) * (PosX - LastPosX)) // if the ball had a lower distance 1 frame ago than 2 ago    /*last realspx*/      
                                  //  10 so  100             100                      100
                
                //if RealSp(X AND Y) is under 1(if the ball moved less than a pixel in any direction), and the speed is very low, divide the speed
                
                /*if(IsLowSpeed)
		{
			if(RealSpY > -1.0)
			{
				if(RealSpY < 1.0)
				{
					if(RealSpX > -1.0)
					{
						if(RealSpX < 1.0) SpX = LastSpX / 2;
					                          SpY = LastSpY / 2;
                                        }
				}
			}
		}*/
		
                LastPosX = PosX;
		LastPosY = PosY;
		LastSpX = SpX;
		LastSpY = SpY;
		
		if(ThrownByExplosion)
		{
/*FLAMES*/			if(SpeedFlames)
			{
				if(Speed < SpeedFlamesStopThreshold)
				{
					SpeedFlamesStopTimer = SpeedFlamesStopTimer - 1;
					if(SpeedFlamesStopTimer <= 0)
					{
						SpeedFlames = false;
					}
				}
				else
				{
					SpeedFlamesStopTimer = SpeedFlamesStopDuration;
				}
			}
			else
			{
				if(Speed > SpeedFlamesStartThreshold)
				{
					SpeedFlamesStopTimer = SpeedFlamesStopDuration;
					SpeedFlames = true;
				}
				ThrownByExplosionTimer = ThrownByExplosionTimer - 1;
				if(ThrownByExplosionTimer <= 0)
				{
					ThrownByExplosion = false;
				}
			}
		}
		if(Speed > MinPushSpeed)
		{
			ProcessBallOnBallCollision(Speed);
		}
		
		if(Speed > MinDamageSpeed)
		{
			ProcessBallOnOtherCollision(Speed);
		}
		if(Speed < LowSpeedThreshold)
		{
			LowSpeedTimer = LowSpeedTimer - 1;
			if(LowSpeedTimer <= 0)
			{
				//local p = new PxParticle(81, PosX, PosY);
				//p->SetLifeTime(30);
				//p->SetVelocity(0,0);
				//p->SetStartAlpha(255);
				//p->SetEndAlpha(255);
				IsLowSpeed = true;
				LowSpeedTimer = TimeForLowSpeed;
			}
		}
		else
		{
			LowSpeedTimer = TimeForLowSpeed;
			IsLowSpeed = false;
		}
		
		SleepSampleTimer = SleepSampleTimer - 1;
		if(SleepSampleTimer <= 0)
		{
				//local p = new PxParticle(79, PosX, PosY);
				//p->SetLifeTime(30);
				//p->SetVelocity(0,0);
				//p->SetStartAlpha(255);
				//p->SetEndAlpha(255);
				
			float dx = PosX - LastRecordPosX;
			float dy = PosY - LastRecordPosY;
			float dist = sqrt(dx*dx+dy*dy);
			
			if(dist < SleepDistanceThreshold) Sleep();
			
			SleepSampleTimer = SleepSamplePeriod;
			LastRecordPosX = PosX;
			LastRecordPosY = PosY;
		}
		if (PosY > Env->Water || OutOfBounds) {
//		    if (ballOwner != NullObj) ballOwner->haveBalls = false;
		    if (ballOwner != NullObj)
		    {
		        aaahh->Play(1.0, 0.0, false);
                        SpX = 0.0;
		        SpY = 8.0;
                        PosX = SleepX;
		        PosY = SleepY-20;
                        if (addPointsBySink) 
                        {
                           ballOwner->holePoints++;
                           ballOwner->totalPoints++;
                           ballOwner->golf_teleported = true;
                        }
                        RespawnMe = true;
//                        Sleep();
                        addPointsBySink = false;
                        OutOfBounds = false;
		    }
		} else addPointsBySink = true;
		spin -= 0.01;
		if (justDropped || RespawnMe){}
		else CalculateLifting();
	}
}

void CGolfBall::SearchNextHole()
{
    for (local i = 1; i < Root->Childs.Count; i++)
    {
        local obj = CObject(Root->Childs.Objs[i]);
        if (#GOLF_ZONES)
        {
            if (obj is CGolfHoleStart)
            {
                local zone = CGolfHoleStart(obj);
                if (ballOwner != NullObj && zone->hole == ballOwner->currentHole)
                {
                    TeleportToNextHole = false;
//                    ReportScore();
                    TeleportOwner(zone->x+zone->offset, zone->y);
                    if (zone->offset < 0) ballOwner->TurnSide = 1;      // make worm look to correct direction
                    else ballOwner->TurnSide = -1;
                    if (IsAsleep) Wake();
                    PosX = zone->x;
                    PosY = zone->y-20;
                    SpX = 0.0;
                    SpY = 7.0;
                    ballOwner->holePoints = 0;
                    if (zone->restart) ballOwner->currentHole = 1;
//                    RespawnMe = true;
                    justDropped = true;
//                    ballOwner->HaveBall = CGolfBall(NullObj);
//                    ballOwner = CWorm(NullObj);
                }
            }
        }
    }
}

int CGolfBall::CalculatePoints(int par)
{
   if (ballOwner == NullObj) return;

// used NASA`s formula (http://www.grc.nasa.gov/WWW/k-12/airplane/beach.html)
// L = (4 * pi^2 * b^3 * s * rho * V) * 4 / 3
// b is a ball radius; s is a speed of rotations (resolutions per second); V is a velocity; pho is a density 
void CGolfBall::CalculateLifting()
{
    if (spin >= 0) Lift = (4 * MATH_PI*MATH_PI * radius*radius*radius * spin * density * abs(SpX)) * 4 / 3;
    else Lift = 0;
    if (Lift < 0.8 && Lift > 0) SpX = SpX * 0.995;
    if (spin < 0) SpX = SpX * 0.995;
    else if (usedClub == 1) SpX = SpX * 1.01;
    else if (usedClub == 2) SpX = SpX * 1.002;
    else if (usedClub == 3) SpX = SpX * 1.001;
    else if (usedClub == 4) SpX = SpX * 1.001;
    if (Lift > 0) SpY -= Lift;
}
