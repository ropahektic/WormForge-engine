void tunnelmaker::InitGraphic()
{
    pixel  = MakeBitmap(1, 1);
    fill10 = MakeBitmap(10, 10);
    fill20 = MakeBitmap(20, 20);
    fill30 = MakeBitmap(30, 30);
    fill40 = MakeBitmap(40, 40);
    fill50 = MakeBitmap(50, 50);
    fill60 = MakeBitmap(60, 60);
    fill70 = MakeBitmap(70, 70);
    fill80 = MakeBitmap(80, 80);
    fill90 = MakeBitmap(90, 90);
    fill100 = MakeBitmap(100, 100);
        
    CFile *f;
    f = GetAttachment("pointery.png");	yellowpointer = LoadSprite(f, 1, 0);
    f = GetAttachment("firecircle.png");	rcircle = LoadSprite(f, 1, 0);
}

void tunnelmaker::FirstFrame()
{
    SpecialX = GS->LevelSX / 2;
    SpecialY = GS->LevelSY / 2;
    Brushspeed = 6;
    bsz = 20;
    cratetype = CT_HEALTH;
    curfill = fill20;
    minefuse = 3000;
    FixEraser = true;
    TeamColorSetDfl = true;
    teamcolored = true;

}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
   if (Weap->CheckName("Mouse Brush"))
   {
       local cteam = Root->CurTeam;
       local snd = 159;
       if (cteam == 1 || cteam == 6) snd = 159;
       else {snd = 105 + 54 * cteam;}
   
       ShowMessage("Do NOT click the mouse!");
       PlaySound(snd,0,0,1.0);
       
       if (canwarn)
       {
           GG->WriteToChat(1,"Notice: do not click the mouse. Hold the given keys to paint/erase/etc.",false);
           GG->WriteToChat(1,"I don't know will or not the painter support clicking. One day it will.",false);
           canwarn = false;
       }
   }
   
   else super;
}
