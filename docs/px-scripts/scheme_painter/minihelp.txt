void minihelp::FirstFrame()
{
 new CHelpGUI(Root);
 HGUION = true;
 HGUIEnabled = true;
}

void CHelpGUI::Enable()
{
 if((IsKeyPressed(17) && IsKeyPressed(72)) && HGUIEnabled && !pressed) 
   { 
      HGUION = false; 
      pressed = true; 
   } 
 
 if((!IsKeyPressed(17) && !IsKeyPressed(72)) && HGUIEnabled  && pressed) 
   { 
      HGUIEnabled = false; 
      pressed = false; 
   } 
 
 if((IsKeyPressed(17) && IsKeyPressed(72)) && !HGUIEnabled && !pressed) 
   { 
      HGUION = true; 
      pressed = true; 
   } 
 
 if((!IsKeyPressed(17) && !IsKeyPressed(72)) && !HGUIEnabled && pressed) 
   { 
      HGUIEnabled = true; 
      pressed = false; 
   }         
}

void CHelpGUI::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{

 if (Type == M_FRAME)
    {
    Enable();     
    ProcessOutput();   
    }
    
 if (Type == M_DRAWQUEUE)
    {
     if (HGUION)
     {
      AddInterfaceBitmap(0, (GS->Disp->xRes / 2) - HGUI->SX / 2,(GS->Disp->yRes / 2) - HGUI->SX,HGUI,360,640,0);
     }
    }
}
