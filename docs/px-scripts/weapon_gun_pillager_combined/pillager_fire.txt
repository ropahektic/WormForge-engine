require utils, p_sprite_builder;   

CWeapon* g_PillagerWeap;

void pillager_fire::FirstFrame()
{
g_PillagerWeap = GetWeaponByName("Pillager");
}

//---------------CWorm Class Overrides---------------//
override CWorm::CWorm(CObject * Parent, int aTeam, int aIndex, CWormParams * params)
{
    super;
    PShotsFired = 0;
    pillShooting = false;
}

override void CWorm::FireFinal(CWeapon* Weap, CShootDesc* Desc)
{
    if (Weap->CheckName("Pillager") )
    {
       if (PShotsFired<=19)     //Quantity of shots
       {            
         for (int curShots = PShotsFired; PShotsFired==curShots; PShotsFired++) //For some reason PShotsFired++ normally doesn't work.
         Desc->SpX = (TurnSide * (0.5 - sqrt((0.5 - FireAngle)*(0.5 - FireAngle)))) * 20;
         Desc->SpY = 20*(0.5 - FireAngle);           
         Desc->Y-18;    //hotfix
         Desc->SpY + RandomInt(-4,4);     //Spread
         super(g_PillagerWeap, Desc);
         pillShooting=true;
         return;   //Return makes it loop.
       }
       pillShooting=false;   //Don't reset shots here, it'll just keep looping. 
       return;
    }        
    pillShooting=false; //Shouldn't be neccessary, however i'm insecure.
    super;
}
     
// --- Main Logic Loop --- //
override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
super;
  if (Type == M_FRAME)
  {
    if(CurWeapon->CheckName("Pillager") == true)
    {
       if (pillShooting)
       {
           // Persist animation.  
           nShoots = 1;
           nShooted = 0;
       }
    }
  }
  if (Type==M_SETWEAPON)
  {
     if(CurWeapon->CheckName("Pillager") == true)
     {PShotsFired=0;pillShooting=false;}       //Reset variables
  }  
}
