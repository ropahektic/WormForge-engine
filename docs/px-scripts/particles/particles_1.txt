void CParticleMan::ParticleBoom(int n, float x, float y, float sp, CParticleDesc* desc)
{
  for (local i = 0; i < n; i += 1)
  {
     local prtcl = new CParticle(this, desc);
     prtcl->x = x;
     prtcl->y = y;
     prtcl->spx = sp * srandom_b() / 2.0;
     prtcl->spy = sp * srandom_b() / 2.0;
  };
};

CParticle : CObject;

CParticle::CParticle(CParticleMan* aman, CParticleDesc* pdesc)
{
  man = aman;
  super(man, GS); //lol
  x = 0.0;
  y = 0.0;
  spx = 0.0;
  spy = 0.0;
  counter = 0;
  gravity = pdesc->gravity;
  wind = 0.0;
  randomness = pdesc->randomness;
  rotspeed = pdesc->rotspeed;
  life = 0.0;
  lifetime = pdesc->lifetime + pdesc->lifevar * srandom_b();
  desc = pdesc;
  startsize = pdesc->startsize + pdesc->sizevar * srandom_b();
  endsize = pdesc->endsize;
  if (pdesc->randomrot){rot = srandom_p() * 6.0;}
  else {rot = 0.0;}
  
  color = ARGB(0, 255, 255, 255);
  destroyme = false;
};

void CParticle::Render()
{
    if (destroyme) return;
    local part = life / lifetime;
    local size = startsize + (endsize - startsize) * part;
    local sindex = parsprites[desc->texindex]->Index;
    local alpha = desc->startA + (desc->endA - desc->startA) * part;
    local col = ARGB(alpha, desc->colorR, desc->colorG, desc->colorB);

    CQuad q;
    
    FillSpriteQuad(&q, sindex, 0, 0); // 0, 0
    TransformQuad(&q, rot, size, size, x, y);
    q.blend = 0;
    for (local i = 0; i < 4; i += 1) q.v[i].color = col;

    RenderQuad(&q, 0);
};

void CParticle::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{
  if (destroyme) return;
  if (Type == M_FRAME)
  {
	counter += 1;
    if (desc->collide)
    {
      local trgx = x + spx;
      local trgy = y + spy;
      while (man->CheckPoint(trgx, trgy))  //cheap way
      {
         if (srandom_p() < 0.5)
         {
            spx = spy * -0.94;
            spy = spx * 0.94;
         } else {
            spx = spy * 0.94;
            spy = spx * -0.94;
         };
         trgx = x + spx;
         trgy = y + spy;         
         if (abs(spx) < 0.1 && abs(spy) < 0.1) break;
      };
    };
    if (desc->toxic > 0)
    {
      if (counter % 8 == 0)
      {
        for (i = 0; i < Env->Objs.Count; i += 1)
        {
          local ob = Env->Objs.Objs[i];
          local obj = CGObject(ob);
          if (obj == NullObj) continue;
          if (obj->ClType != OC_Worm) continue;
          if (abs(obj->PosX - x) < 20)
          {
            if (abs(obj->PosY - y) < 20)
            {
              local force = desc->toxic;
              mdata = new CMessageData;
              mdata->params[0] = force;
              mdata->params[1] = -1;
              mdata->params[2] = 0;
              obj->Message(this, M_POISON, 1032, mdata);
              delete mdata;
            };
          };
        };
      };
    };
    x += spx;
    y += spy;
    spx += wind;
    spy += gravity;
    rot += rotspeed;
    life += 1.0;
    spx = spx * desc->damping;
    spy = spy * desc->damping;
    
    if (randomness > 0.0)
    {
       spx += srandom_b() * randomness;
       spy += srandom_b() * randomness;
    };
    
    if (life > lifetime)
    {
      destroyme = true;
    };
  };
};
