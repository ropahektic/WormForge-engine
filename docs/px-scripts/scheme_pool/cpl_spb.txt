override void CWorm::RenderWorm()
{
SetColorMod(GetTeamColorRGB(GetTeamColor(WormTeam)+1), 8);
AddSpriteEx(13, PosX, PosY, cpbaz->Index, 1.0, 0, 0.8);
ClearColorMod();
}




//------------------------------------------------------------------------------
override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;
if (Type == M_FRAME)
{
	if (weap->CheckName("Cue"))
		{
		if (cuebl_anim)
			{
			local sprite = cuebl_bul->Index;
			WeapSpriteParams* Params = new WeapSpriteParams;
			Params->animate = false;
			new WeapSprite(this, sprite, Params);
			weap->launch.anim.spriteIndex = 0;
			cuebl_anim = false;
			}
		}

	if (weap->CheckName("Remote Cue"))
		{
		if (rcueb_anim)
			{
			local sprite = rcueb_bul->Index;
			WeapSpriteParams* Params = new WeapSpriteParams;
			Params->animate = false;
			new WeapSprite(this, sprite, Params);
			weap->launch.anim.spriteIndex = 0;
			rcueb_anim = false;
			}
		}
		
	if (weap->CheckName("Hyper Cue"))
		{
		if (hcueb_anim)
			{
			local sprite = hcueb_bul->Index;
			WeapSpriteParams* Params = new WeapSpriteParams;
			Params->animate = true;
			Params->anim_speed = 1.00;
			new WeapSprite(this, sprite, Params);
			weap->launch.anim.spriteIndex = 0;
			hcueb_anim = false;
			}
		}
		
	if (weap->CheckName("Cheap Shot"))
		{
		if (cheap_anim)
			{
			local sprite = cheap_bul->Index;
			WeapSpriteParams* Params = new WeapSpriteParams;
			Params->animate = true;
			Params->anim_speed = 4.00;
			new WeapSprite(this, sprite, Params);
			weap->launch.anim.spriteIndex = 0;
			cheap_anim = false;
			}
		}
	}
}

{
super;
cuebl_anim = true;

hcueb_anim = true;
cheap_anim = true;
}
//------------------------------------------------------------------------------
override void CWorm::PlaySoundBank(int sound,fixed param1)
{
if(CurWeapon->CheckName("Cue")
|| CurWeapon->CheckName("Remote Cue")
|| CurWeapon->CheckName("Hyper Cue")
|| CurWeapon->CheckName("Guidance Cue")
|| CurWeapon->CheckName("Cheap Shot")
|| CurWeapon->CheckName("Ball Launcher"))

	{
	if (sound == 140 || sound == 194) return;         // FIRE!
	else if (sound == 177 || sound == 231) return;    // WATCH THIS...
	else super;
	}
else
	{
	super;
	}
}
