require utils;

CColMask* col_wpn;

void cpl_col_wep::Init(){}

void cpl_col_wep::FirstFrame()
{
col_wpn = new CColMask(20,20,MakeCircleMask(20));
}

override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata)
{
super;
Cwpn = true;
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
if(Cwpn == true)
	{
	if(Type == M_FRAME)
		{
		WEPCol(15.0);
		}
	}
super;
}

void CMissile::WEPCol(float Speed)
{
CGObject *plwepob;

for(int i = 0; i < Env->Objs.Count; i += 1)
	{
	plwepob = CGObject(Env->Objs.Objs[i]);
	if(plwepob == this) continue;
	if(plwepob == NullObj) continue;
	if(GetCurrentWorm() != NullObj)
		{
		if(plwepob == GetCurrentWorm()) continue;
		}
	if(plwepob->ClType == OC_Worm
	|| plwepob->ClType == OC_Mine
	|| plwepob->ClType == EObjectClass(70)
	|| plwepob->ClType == EObjectClass(71)
	|| plwepob->ClType == EObjectClass(73)
	|| plwepob->ClType == EObjectClass(74))
		{
		if(col_wpn->Check(PosX, PosY, plwepob->ColMask, plwepob->PosX, plwepob->PosY))
			{
			float ForceX = plwepob->PosX - PosX;
			float ForceY = plwepob->PosY - PosY;
			float dist = sqrt(ForceX*ForceX + ForceY*ForceY);

			if(dist > 0)
				{
				ForceX = 0.6*sqrt(Speed) * ForceX / dist;
				ForceY = 0.6*sqrt(Speed) * ForceY / dist;

				CMessageData msg;
				msg.params[0]	= 0;
				msg.fparams[1]	= PosX;
				msg.fparams[2]	= PosY;
				msg.fparams[3]	= ForceX;
				msg.fparams[4]	= ForceY;
				msg.params[5]	= 0;
				msg.params[6]	= 0;

				plwepob->Message(this, M_GUNEXP, 1032, &msg);
				}
			}
		}
	}
}
