override CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
{
    DoRm = false;
    super;
}

void FinishRound()
{
        for (local i = 1; i < Env->Objs.Count; i++)
        {
            local foundObj = CGObject(Env->Objs.Objs[i]);
            local wrm = CWorm(foundObj);
            if (wrm is CWorm)
            {
                wrm->DoRm = true;
                wrm->PosY = Env->Water + 2000;
            }
        }
        
        CMessageData *loldata = new CMessageData;
        loldata->params[0] = 1;
        Root->Message(Root, M_TURNBEGIN,4,loldata);
}

     if (Turnfinishing && (SIndex == 136 || SIndex == 190 || SIndex == 244 || SIndex == 298 || SIndex == 352 || SIndex == 406))

int modulei(int i)
{
    if (i < 0) return i*(-1);
    else if (i >= 0) return i;
}

float modulef(float f)
{
    if (f < 0) return f*(-1);
    else if (f >= 0) return f;
}

int CTurnGame::GetCurColor()
{
    local i = 1;
    CWorm* w = GetCurrentWorm();
    
    if (w != NullObj)
    {
        i = w->LabelColor;
    }
    
    return i; 
}
