bool GenerateCrSpawnMap(int side, int mSX, int mSY){
    
    if(NewlyCrate == NullObj){return false;}
    
    local CrSpawnUnderY = 305;
    local CrSpawnOverY = 1088; //not used?
    
    
    CrSearchGridSize = 50;
    local extraX = 0;
    local side2X = 0;
    if(mSX == 2200 && mSY == 1504){
        CrSpawnUnderY = 305;
        CrSpawnOverY = 1088;
        ax = int(880 / CrSearchGridSize);
        ay = int(790 / CrSearchGridSize);
        side2X = 1325;
    }
    if(mSX == 2200 && mSY == 2000){
        CrSpawnUnderY = 713;
        CrSpawnOverY = 1500;
        ax = int(730 / CrSearchGridSize);
        ay = int(790 / CrSearchGridSize);
        extraX = 80;
        side2X = 1377;
    }
    

    local startx = 0;
    local starty = CrSpawnUnderY;
    
    local dOut = 0;
    local DFbmp = DigFoam;
    
    SMlastdistL = 0;
    SMlastdistR = 0;
    for(int iy = 0; iy < ay; iy++ ){
        for(int ix = 0; ix < ax; ix++ ){
            
            local cy = (iy * CrSearchGridSize) + starty;
            local cx = 0;
            if(side == 0){
                cx = (ix * CrSearchGridSize) + extraX;
            }else{
                cx = (ix * CrSearchGridSize) + side2X ;
            }
                
                local Obje = Env->CheckCollision( NewlyCrate, cx, cy );
                if(Obje != NullObj){
                    if(Obje->ClType == OC_Landscape){
                        if(side == 0){
                            MapEmptySpaceL[(iy * ax) + ix] = 0; //not enough room
                        }else{
                            MapEmptySpaceR[(iy * ax) + ix] = 0;
                        }
                        continue;
                    }else if(Obje->ClType == OC_Crate){
                        if(side == 0){
                            MapEmptySpaceL[(iy * ax) + ix] = 0; //spawning on another crate
                        }else{
                            MapEmptySpaceR[(iy * ax) + ix] = 0;
                        }
                        continue;
                    }
                
                }
                
                local hitx1 = 0;
                local hity1 = 0;
                local hit1 = TraceLine(NewlyCrate, cx, cy, cx, CurrWaterLvl + 10, -1, &hitx1, &hity1);

                if (hit1 != NullObj) 
                {
                    //GG->land->Draw(DFbmp, cx, cy);
                    local disty = 0;
                    disty = abs(cy - hity1);
                    
                    if(disty > 100){disty = 100;}
                    if(disty == 0){disty = 1;}
                    
                    if(side == 0){
                        dOut = SMlastdistL + disty;
                        SMlastdistL = dOut;
                        MapEmptySpaceL[(iy * ax) + ix] = dOut;
                    }else{
                        dOut = SMlastdistR + disty;
                        SMlastdistR = dOut;
                        MapEmptySpaceR[(iy * ax) + ix] = dOut;
                    }

                    continue;
                }else{
                    if(side == 0){
                        MapEmptySpaceL[(iy * ax) + ix] = 0; //water
                    }else{
                        MapEmptySpaceR[(iy * ax) + ix] = 0;
                    }
                   
                    continue;
                }
                
            
        }
    }
    
    CrMapGenerated = true;
    return true;
}

int NewCrX;
int NewCrY;
bool CrateSpawnSuccess;
void GetNewCrateSpawn(int mSX, int mSY, int rngseedA){ 
    CrateSpawnSuccess = false;
    
    local RndSide = RandomInt(0,1);
    //local RndSide = (StartTurnCounter + 1) % 2; CAUSE OF DESYNC?
    //if(RndSide < 50){ RndSide = 0; }else{ RndSide = 1;}
    if (GenerateCrSpawnMap(RndSide, mSX, mSY) == false){
        CrateSpawnSuccess = false; return;
    }
    
    
    //pick a value between 1 and SMlastdist
    local Rtarget = 0;
    if(RndSide == 0){
        //Rtarget = RandomInt(1,SMlastdistL);
        Rtarget = rngseedA % SMlastdistL;
    }else{
        //Rtarget = RandomInt(1,SMlastdistR);
        Rtarget = rngseedA % SMlastdistR;
    }
    
    local lastIndex = 0;
    //local NoSpots = true;
    //Find closest value to target in array
    
    //Optimize ?
    
    for(int i = 0; i < (ax * ay); i++ ){
        local MapInt = 0;
        if(RndSide == 0){
            MapInt = MapEmptySpaceL[i];
        }else{
            MapInt = MapEmptySpaceR[i];
        }
        
        if(MapInt != 0){
            if(MapInt < Rtarget){
                    lastIndex = i;
                    CrateSpawnSuccess = true;
                continue;
            }else{
                break;
            }
        }
    }
    

    
   // if(NoSpots){return false;} //no spot found
   
   
   if(mSX == 2200 && mSY == 1504){
       if(RndSide == 0){
        NewCrX = ((lastIndex % ax) * CrSearchGridSize);
        NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 305;
       }else{
        NewCrX = ((lastIndex % ax) * CrSearchGridSize) + 1325;
        NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 305;  
       }
    }
    if(mSX == 2200 && mSY == 2000){
       if(RndSide == 0){
        NewCrX = ((lastIndex % ax) * CrSearchGridSize) + 80;
        NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 713;
       }else{
        NewCrX = ((lastIndex % ax) * CrSearchGridSize) + 1377;
        NewCrY = (int(lastIndex / ax) * CrSearchGridSize) + 713;  
       } 
    }

    }

//Returns the number of cells in a ring around (x,y) that are alive.

int countAliveNeighbours(int inX, int inY){
    local count = 0;
    local i = 0;
    local j = 0;
    local neighbour_x = 0;
    local neighbour_y = 0;
    
    local widthRow = cSizeX;
    local size_x = cSizeX;
    local size_y = cSizeY;
    
    for(i = -1; i < 2; i++){
        for(j = -1; j < 2; j++){
            neighbour_x = inX + i;
            neighbour_y = inY + j;
            //If we're looking at the middle point
            if(i == 0 && j == 0){
                //Do nothing, we don't want to add ourselves in!
            }
            //In case the index we're looking at it off the edge of the map
            /*else if(neighbour_x < 0 || neighbour_y < 0 || neighbour_x >= size_x || neighbour_y >= size_y){
                count = count + 1;
            }*/
            //Otherwise, a normal check of the neighbour
            else if( CaveMap[ neighbour_x + ( neighbour_y * widthRow)] == 1){
                count = count + 1;
            }
        }
    }
    return count;
}

    local LocalCircle = CicleMaskMedium;
    //How hidden does a spot need to be for treasure?
    //I find 5 or 6 is good. 6 for very rare treasure.
    local treasureHiddenLimit = 30;

    local nbs = 0;
    placedT = 0;
    local widthRow = cSizeX;
    local size_x = cSizeX;
    local size_y = cSizeY;
    treasureHiddenLimit = inLimit;
    for (x=0; x < size_x; x++){

    if(placedT < 2){//if < 2 sentrys could spawn, spawn two in a more common place.
        
        for (x=0; x < size_x; x++){
        for (y=0; y < size_y; y++){
            if(CaveMap[x + (y * widthRow)] == 0){

    }

        CaveMap[(yi * widthRow) + size_x - 1 ] = 0;
	for(xi=0; xi<size_x; xi++)
		CaveMap[xi] = 0;
        CaveMap[((size_y - 1)*widthRow) + xi] = 0;
 }
 
 
 void CaveGeneration()
 {
	local xi = 0;
    local yi = 0;
    local ii = 0;
    local jj = 0;
    local size_x = cSizeX;
    local size_y = cSizeY;
    local widthRow = cSizeX;
    
    local adjcount_r1 = 0;
 	local adjcount_r2 = 0;
 
	for(yi=1; yi<size_y-1; yi++)
	for(xi=1; xi<size_x-1; xi++)
 	{
 		adjcount_r1 = 0;
 		adjcount_r2 = 0;
 
 		for(ii= -1; ii<=1; ii++)
		for(jj= -1; jj<=1; jj++)
 		{
 			if(CaveMap[((yi+ii)*widthRow)+xi+jj] != 0)
 				adjcount_r1++;
 		}
 		for(ii = yi - 2; ii<=yi+2; ii++)
 		for(jj = xi - 2; jj<=xi+2; jj++)
 		{
 			if( abs( ii - yi ) == 2 && abs( jj - xi ) == 2 )
 				continue;
 			if(ii<0 || jj<0 || ii>=size_y || jj>=size_x)
 				continue;
 			if(CaveMap[(ii*widthRow)+jj] != 0)
 				adjcount_r2++;
 		}
 		if(adjcount_r1 >= r1_cutoff || adjcount_r2 <= r2_cutoff)
 			CaveMap2[(yi*widthRow)+xi] = 1;
 		else
 			CaveMap2[(yi*widthRow)+xi] = 0;
 	}
 	for(yi=1; yi<size_y-1; yi++)
 	for(xi=1; xi<size_x-1; xi++)
 		CaveMap[(yi*widthRow)+xi] = CaveMap2[(yi*widthRow)+xi];
 }

int TeamLowestHealth;

int TeamHealthDiff(){
    teamLhp = 0;
    teamRhp = 0;
    local whp = 0;
for (int j = 6; j >= 1; j-=1)
                      {
                        for (int i = 8; i >= 1; i-=1)
                        {
                          
                            local Worm = CWorm(Root->GetObject(2, 16 * j + i));
                            if (Worm)
                            {
                                if(Worm->TeamSide == 1){
                                    whp = GS->Info.GetWormHealth(Worm->WormTeam, Worm->WormNumber);
                                    if(whp < 0){whp = 0;}
                                    teamLhp += whp;
                                }else{
                                    whp = GS->Info.GetWormHealth(Worm->WormTeam, Worm->WormNumber);
                                    if(whp < 0){whp = 0;}
                                    teamRhp += whp;
                                }
                            };
                        
                        };
                      };
    
    if(teamLhp < teamRhp){
        TeamLowestHealth = 1;
    }else{
        TeamLowestHealth = 2;
    }
    
    return abs(teamLhp - teamRhp);
    
}

void FindClosestEnemy(){
    NearestEnemyWorm = NullObj;
    local currW = GetCurrentWorm();
    local j; local i;
    local shortestDistance = 3000;
    if(currW == NullObj){ return; }
    
       for (j = 6; j >= 1; j-=1)
                          {
                            for (i = 8; i >= 1; i-=1)
                            {
                              
                                local Worm = CWorm(Root->GetObject(2, 16 * j + i));
                                if (Worm)
                                {
                                    if(Worm->TeamSide == currW->TeamSide){continue;}
                                    local distX = abs(Worm->PosX - currW->PosX);
                                    local distY = abs(Worm->PosY - currW->PosY);
                                    local tmpdist = sqrt((distX*distX) + (distY*distY));
                                    if(tmpdist < shortestDistance){
                                        shortestDistance = tmpdist;
                                        NearestEnemyWorm = Worm;
                                    }  

                                };
                            
                            };
                          };
}
