require utils, p_sprite_builder, utils_curtis, sprite_curtis_253;

CSprite* bumperb_aiming;
CSprite* sbumperb_aiming;

void bumper_script::InitGraphic()
{
    CFile *f;
    f = GetAttachment("bumperb.png");
    bumperb_aiming = LoadSprite(f, 6, 0);
    f = GetAttachment("sbumperb.png");
    sbumperb_aiming = LoadSprite(f, 12, 0);
}

//---------------------------------BUMPER BOMB--------------------------------//

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) {
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Bumper Bomb";
     sprite = bumperb_aiming->Index;
     Params->hand_type = 1;
     Params->hand_scale = 1.3;
     Params->hand_radius = -16.8;
     Params->hand_rotation = -0.0;
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;
     Params->weap_rotation = 0.5;
     Params->weap_radius = -13.0;
     Params->animate = true;
     Params->anim_speed = 1.25;
     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}
override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;
  if (Type == M_FRAME){
     if (weap->CheckName("Bumper Bomb")) {
         if (bumperb_anim) {
              local sprite = bumperb_aiming->Index;
              WeapSpriteParams* Params = new WeapSpriteParams;
              Params->animate = true;
              Params->anim_speed = 1.10;
              new WeapSprite(this, sprite, Params);
              weap->launch.anim.spriteIndex = 0;
              bumperb_anim = false;
         }
     }
  }
}

override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;
  bumperb_anim = true;
}
