void CCrate::SetBoobyTrapConfig()
{
     // When a random weapon crate appears, it has x probability to be a booby-trap
     // 1 -> 100%      // 4 -> 25%     // 7 -> 14%    // 10 -> 10%
     // 2 -> 50%       // 5 -> 20%     // 8 -> 12%    // 50 -> 2%
     // 3 -> 33%       // 6 -> 16%     // 9 -> 11%    //100 -> 1%
     
     boobytrap_random = false;          //Spawn auto boobytrap crates
     boobytrap_probability = 100;        //Default 20 % probability
}

CBoobyTrap::CBoobyTrap(CCrate * crate_parent)
{
     if(crate_parent == NullObj) return;
     
     super(Root->GetObject(25,0), GS);
     boobytrap_crateparent = crate_parent;
     boobytrap_explode = false;
     boobytrap_mine = CMine(NullObj);
     boobytrap_dwalking = 42;
     boobytrap_dairing = 50;
     
     CMineParams MParams;
     zero(&MParams);
     
     MParams.Prefuse = 0;
     MParams.Flags = 4+8+16+32+64+256+512+4096+32768+65536;   //1024
     MParams.Bias = 30; 
     MParams.Damage = 40;  
     MParams.BlastPower = 70;
     MParams.Fuse = 0;
     MParams.Radius = 30;
	
     CShootDesc MDesc;
     zero(&MDesc);
     MDesc.Team = 0;
     MDesc.Worm = 0;
     MDesc.X = boobytrap_crateparent->PosX;
     MDesc.Y = boobytrap_crateparent->PosY;
     MDesc.SpX = boobytrap_crateparent->SpX;
     MDesc.SpY = boobytrap_crateparent->SpY;
        
     boobytrap_mine = new CMine(Root->GetObject(25, 0), &MParams, &MDesc, false, 0);
     boobytrap_mine->PosX = boobytrap_crateparent->PosX;
     boobytrap_mine->PosY = boobytrap_crateparent->PosY;
     boobytrap_mine->boobytrap_explodedbyworm = false;
     boobytrap_mine->ClType = EObjectClass(80081);
     boobytrap_mine->GravityFactor = 0;                                     
}

override void CCrate::CCrate(CObject* Parent,CCrateDesc* desc,EDropType type)
{
     boobytrap_asweapon = desc->highlight;
     desc->highlight = false;
     
     super;
     
     boobytrap_random = true;
     boobytrap_enable = false;
     boobytrap_probability = 5;
     
     if(desc->type == CT_WEAPON || desc->type == CT_UTILITY || desc->type == CT_HEALTH || desc->type == CT_MWEAPON)
     {
          if(boobytrap_asweapon)
          {
               boobytrap_probability = 1;
          }
          else
          {
               SetBoobyTrapConfig();
          }
          
          if(int(boobytrap_probability) <= 0)
          {
               boobytrap_probability = 5;
          }
          
          if( RandomInt(1, boobytrap_probability) == 1)
          {
               if(boobytrap_random)
               {
					LayerBackup = Layer;
                    ColGroupBackup = ColGroup;
                    MaskIndexBackup = MaskIndex;
               
                    boobytrap = new CBoobyTrap(this);
                    
                    boobytrap_enable = true;
                    Layer = 0;
                    MaskIndex = 0;
                    ColGroup = 2+2048+4194304;
               }
          }
     }
}

void CBoobyTrap::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
     if(Type == M_FRAME)
     {
          if(boobytrap_crateparent != NullObj && boobytrap_mine != NullObj)
          {
               boobytrap_mine->PosX = boobytrap_crateparent->PosX;
               boobytrap_mine->PosY = boobytrap_crateparent->PosY;
               boobytrap_mine->SpX = boobytrap_crateparent->SpX;
               boobytrap_mine->SpY = boobytrap_crateparent->SpY;
               
               local cworm = GetCurrentWorm();
               
               if(cworm != NullObj)
               {              
                   if(cworm->ObjState == WS_ROPING || cworm->ObjState == WS_BUNGEE)
                   {
                        boobytrap_mine->Radius = boobytrap_dairing;
                   }
                   else
                   {
                        boobytrap_mine->Radius = boobytrap_dwalking;                                                
                   }
              }
          }
          else
          {
               Free(true);
          }
     }
     
     if(Type == M_DRAWQUEUE) {}
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
   CWorm* w = GetCurrentWorm();
   
   if(CurWeapon->CheckName("Crate Shower Mode") && ObjState == WS_FIRED)
   {
      local wep = GS->GetWeaponByName("Crate Shower");
      GS->Info.SetWeaponEx(WormTeam, wep->GetWeaponIndex(), wep->GetPageIndex(), 1);   
   }
   
   if(CrateShowerMode == 1)
   {
      if(Type == M_TURNEND)
      {
         local wep = GS->GetWeaponByName("Crate Shower");
         GS->Info.SetWeaponEx(WormTeam, wep->GetWeaponIndex(), wep->GetPageIndex(), 0);
         CrateShowerMode == 0;   
      }
   }      
   
   if(Type == M_FRAME)
   {
      if(CurWeapon->CheckName("Select Crate"))
      {
         if(keySync->KeyPressedNow(49))
         {
            selcratype = 0; 
            ShowMessage("Weapon Crate");
            PlayLocalSound(92, 1.0, 1.0, 1.0);
            if(AnimCanChangeCrateType)
	    {
	    Anim->SetAnim(362);
	    }
         }
         else if(keySync->KeyPressedNow(50))
         {
            selcratype = 1;
            ShowMessage("Utility Crate");
            PlayLocalSound(92, 1.0, 1.0, 1.0);
            if(AnimCanChangeCrateType)
	    {
	    Anim->SetAnim(362);
	    }
         } 
         else if(keySync->KeyPressedNow(51))
         {
            selcratype = 2;
            ShowMessage("Health Crate");
            PlayLocalSound(92, 1.0, 1.0, 1.0);
            if(AnimCanChangeCrateType)
	    {
	    Anim->SetAnim(362);
	    }
         } 
      }
      
      if(CurWeapon->CheckName("Trap Crate"))
         {
         local worm = GetCurrentWorm();
        
         if(keySync->KeyPressedNow(49))
            {
            selboobytrap_distance = 200; 
            ShowTeamMessage(7,"200 pixels distance",worm->WormTeam);
            }
         else if(keySync->KeyPressedNow(50))
            {
            selboobytrap_distance = 300;
            ShowTeamMessage(7,"300 pixels distance",worm->WormTeam);
            } 
         else if(keySync->KeyPressedNow(51))
            {
            selboobytrap_distance = 400;
            ShowTeamMessage(7,"400 pixels distance",worm->WormTeam);
            }
         else if(keySync->KeyPressedNow(52))
            {
            selboobytrap_distance = 500;
            ShowTeamMessage(7,"500 pixels distance",worm->WormTeam);
            } 
         else if(keySync->KeyPressedNow(53))
            {
            selboobytrap_distance = 600;
            ShowTeamMessage(7,"600 pixels distance",worm->WormTeam);
            }         
         }            
   }
   
   super;
}

override void CCrate::Free(bool FreeMem)
{
     if(boobytrap_enable)
     {
		  Layer = LayerBackup;
          ColGroup = ColGroupBackup;
          MaskIndex = MaskIndexBackup;
     }
         
     super;
}

override void CMine::PlayLocalSound(int SIndex,int Unk,fixed Volume,fixed Pan)
{
     if(ClType == EObjectClass(80081))
     {
         if(SIndex == 90)
         {
             Active = 1;
         }
     }
     else super;
}

void CBoobyTrap::Free(bool FreeMem)
{
    if(FreeMem)
    {
        if(boobytrap_mine != NullObj)
        {
             boobytrap_mine->Free(true);
        }
        super;
    }
}

override CGObject * CMine::GetTarget(int Radius)
{
    if(ClType == EObjectClass(80081))
    {
         CGObject * cobj = super;
         
         if(cobj is CWorm)
         {
              boobytrap_explodedbyworm = true;
         }
    }
    super;
}

override void CMine::Explosion()
{
     if(ClType == EObjectClass(80081) && boobytrap_explodedbyworm)
     {
     ShowMessage("** Booby Trapped **");
     }
     super;
}

				int hitX = 0;
				int hitY = 0;

					crate->PosX = newtrcrX;
					crate->PosY = newtrcrY;
					crate->X = newtrcrX;
					crate->Y = newtrcrY;
					return true; // found a spawn point, return. This means exit this entire function.
					}
				}
			}
		crate->Free(true); // we did not find a spawn point, lets destroy the crate we have created
		return false;
		
	}
	
	else if (Weap->CheckName("Select Crate"))

	    if      (selcratype == 0) cratype = CT_WEAPON;

CGObject * CWorm::SCFindCrate(int l, int r, int t, int b)     // left, right, top, bottom
{
      for(int i = 0; i < Env->Objs.Count; i++)
      {
          local obj = CGObject(Env->Objs.Objs[i]);
          if(obj == NullObj) continue;     
          if(obj->ClType == OC_Crate)
             {             
             if(obj->IsInBox( l, t, r, b))                    // left, top, right, bottom
             return obj;
             }          
      }
      return CGObject(NullObj);
}

override void CSpriteAnim::SetAnimEx(int sindex, fixed frame)
{
	if(GS->Tick >= 1)
	{
		CWorm *w = GetCurrentWorm();
		if(w != NullObj && w->Anim == this)
		{
			if(sindex == 362 && w->CurWeapon->CheckName("Select Crate")) //prod deploy
			{
				if(selcratype == 0)
				{
				super(wcpp->Index, frame);
				}
				else if(selcratype == 1)
				{
				super(ucpp->Index, frame);
				}
				else if(selcratype == 2)
				{
				super(hcpp->Index, frame);
				}
				w->AnimCanChangeCrateType = true;
				return;
			}
			else
			{
				w->AnimCanChangeCrateType = false;
			}			
		}
	}	
	super;
}
