// But draws sprites placing it's center on the coordinates
// This is why we need offsets
// offsetX is always half of width
// offsetY is always half of height

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
CWorm *w = GetCurrentWorm();

if (w != NullObj)
    {
    if(Weap->CheckName("Mario Builder (x4)") || Weap->CheckName("Mario Builder (x5)"))
		{
	    local x = SelectX/65536;
        local y = SelectY/65536;
        // Offset is needed here to move bitmap's center to the cursor
        GG->land->Draw(bmp, x - offsetX, y - offsetY);
        vol = CalculateSoundVolume(PosX,PosY);
     	pan = CalculateSoundPan(PosX,PosY);
        brick->Play(vol, pan, false);

        if(ReadyForMusic == true)
            {
            mario_theme->Play(vol, pan, false);
            ReadyForMusic = false;
            }

		// Makes a hole for anything inside of the drawn shape to prevent falling through terrain abuse
        for(i = 0; i < Env->Objs.Count; i++)
      		{
          	local obj = CGObject(Env->Objs.Objs[i]);
          	if(obj == NullObj) continue;
          	if(obj != NullObj)
             	{
             	if(obj->IsInBox(x - offsetX, y - offsetY, x + offsetX, y + offsetY))                  // left, top, right, bottom
             		{
		            GG->land->ApplyMask(4, obj->PosX-5, obj->PosY+5);
		            GG->land->ApplyMask(4, obj->PosX-5, obj->PosY-5);
		            GG->land->ApplyMask(4, obj->PosX+5, obj->PosY+5);
		            GG->land->ApplyMask(4, obj->PosX+5, obj->PosY-5);
					}
				}
			}
		}
	}
super;
}

void CWorm::LimitCursorMovement()
{
// Crude way to deal with coordinate mismatch

if(GS->landSelectX >= GS->LevelSX - offsetX && GS->landSelectX < 32000) GS->landSelectX = GS->LevelSX - offsetX;
if(GS->landSelectY >= GS->LevelSY - offsetY && GS->landSelectY < 32000) GS->landSelectY = GS->LevelSY - offsetY;

if(GS->landSelectX <= offsetX || GS->landSelectX > 32000) GS->landSelectX = offsetX;
if(GS->landSelectY <= offsetY || GS->landSelectY > 32000) GS->landSelectY = offsetY;
}
