override CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
{
super;
WallJumpMultiplier = 4;
WallJumpDelay = 0;
ColGroup = ColGroup - 4;
}


override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
      if(Type == M_FRAME)
      {
              if(SpY > 6){SpY = 6;}    //weeeeeeeeeeeeeeeee...
              if(SpY < -16){SpY = -16;}  
              if(WallJumpDelay >= 1) {WallJumpDelay = WallJumpDelay - 1;}
      }
      
      if(Type == M_ENTER)
      {
      //GG->WriteToChat( 3, "enter!", false);
      if(WallJumpDelay == 0)
                       {
                       WallJump();
                       //GG->WriteToChat( 3, "wj!", false);
                       }
      WallJumpDelay = 30;
      }  
super;    
}


void CWorm::WallJump()
{
     if(ObjState == WS_JUMP || ObjState == WS_SLIDING || ObjState == WS_FLYING2)
     {
       //wall to the right
       int hit2x;
       int hit2y;
       CGObject *hit2 = TraceLine(this, PosX, PosY, PosX+20, PosY, 2, &hit2x, &hit2y);   
              
       //wall to the left
       int hit3x;
       int hit3y;
       CGObject *hit3 = TraceLine(this, PosX, PosY, PosX-20, PosY, 2, &hit3x, &hit3y); 
       
                if(hit2 != NullObj && hit3 != NullObj)
		{   
		   local x2 = sqrt((hit2x - PosX)*(hit2x - PosX));
		   local x3 = sqrt((hit3x - PosX)*(hit3x - PosX));
		   local side = 0;
		   if(x2 < x3){ side = -1;}
		   else{ side = 1;}
		   
		   
                       if(SpY > 0){SpY = 0;}         			
                       SpY = SpY - WallJumpMultiplier - sqrt(SpX*SpX);
                       SpX = (WallJumpMultiplier+1 + sqrt(SpX*SpX)) * side;
                       TurnSide = side;
                       if(ObjState != WS_JUMP){ SetState(WS_JUMP);}
		}
		else
		{
		    if(hit2 != NullObj)
		    {
		       if(SpY > 0){SpY = 0;}
                       SpY = SpY - WallJumpMultiplier - sqrt(SpX*SpX);
                       SpX = -1*(WallJumpMultiplier+1 + sqrt(SpX*SpX));
                       TurnSide = -1;
                       if(ObjState != WS_JUMP){ SetState(WS_JUMP);}
		    }
		    
		    if(hit3 != NullObj)
		    {
		       if(SpY > 0){SpY = 0;}
                       SpY = SpY - WallJumpMultiplier - sqrt(SpX*SpX);
                       SpX = WallJumpMultiplier+1 + sqrt(SpX*SpX);
                       TurnSide = 1;
                       if(ObjState != WS_JUMP){ SetState(WS_JUMP);}
		    }
		}
     }
}
