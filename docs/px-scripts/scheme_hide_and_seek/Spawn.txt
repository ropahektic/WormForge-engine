void SpawnPoint(CEditorObject* obj)
{
  x = obj->PosX;
  y = obj->PosY;
  team = obj->GetInt("Team");
  num = obj->GetInt("Num");
  if(team == 0)team = 1; //forgot it from 0 to 6 or from 1 to 7 :D fuk dat shit
  if(team == 7)team = 6;

  for (j = 1; j < 7; j++) //team
  {
   if(j == team)
   {
    for (i = 1; i < 9; i++)     //worm
    {
      if(i == num)
      {
        local Worm = CWorm(Root->GetObject(2, 16 * j + i));
        if (Worm != NullObj)
        {
          Worm->PosX = x;
          Worm->PosY = y;
          x = x + 5;
        };
      }
    };
   };
  };
};
