require utils, p_sprite_builder;

CSprite* collector_aiming;

void collector_script::InitGraphic()
{
    CFile *f;    
    f = GetAttachment("collector.png");
    collector_aiming = LoadSprite(f, 24, 0);
} 

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) { 
 super;                                                                         
  if (Type == M_FRAME) {                                                         
    if (gframe == 1) {                                                           
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Crate Collector";                                              
     sprite = collector_aiming->Index;
     Params->hand_type = 1;     
     Params->hand_scale = 1.15; 
     Params->hand_radius = -14.1;                                                 
     Params->hand_rotation = -0.0;                                              
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;     
     Params->weap_rotation = 0.5;
     Params->weap_radius = -11.5;
     Params->animate = true;
     Params->anim_speed = 1.2;                                             
     local WA = new WormSprite(weapName, sprite, Params);
    }                                                                            
  }                                                                              
}

override void CMissile::ExplodeAt(fixed x,fixed y)
{
if (weap->CheckName("Crate Collector"))
  	{
    PlaySound(155,0,0,1.0);
  	}
super;
}
