CMineManager : CObject;
CMineSpriter : CObject;

CMineManager *MM;

CSprite* ElectroExpSprite;
CSprite* StunSprite;

void minesprite::InitGraphic()
{
    local spr4       = GetAttachment("electroexp.png");
    ElectroExpSprite     = LoadSprite(spr4, 8, 0);

    local spr5       = GetAttachment("stunnedspr.png");
    StunSprite     = LoadSprite(spr5, 8, 0);
}

void minesprite::Init(){}

void minesprite::FirstFrame()
{
	MM = new CMineManager();
}

CMineManager::CMineManager()
{
	super(Root, GS);
}

void __DrawStunMine()
{
	MM->Draw();
}

void CMineManager::Draw()
{
	for(int i=0 ; i<Childs.Count ; i+=1)
	{
		local p = CMineSpriter(Childs.Objs[i]);
		if(p == CMineSpriter(NullObj)) {continue;}
		p->Draw();
	}
}

void CMineManager::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{
	if(Type == M_FRAME)
	{
		super;
	}
	if(Type == M_DRAWQUEUE)
	{
		AddCustomRender(4, "__DrawStunMine");
	}

	if(Type == M_FRAME)
	{
		for(int i=0 ; i<Childs.Count ; i+=1)
		{
			local p = CMineSpriter(Childs.Objs[i]);
			if(p == CMineSpriter(NullObj)) continue;

			if(p->Dead)
			{
				p->Free(true);
				break;
			}
		}
	}
}

void CMineManager::MakeSpriter(float x, float y, CStunMine* link)
{
     local Buildin = new CMineSpriter(this, x, y);
     Buildin->PosX = x;
     Buildin->PosY = y;
     Buildin->LinkedBuilding = link;
}

CMineSpriter::CMineSpriter(CMineManager* MM, float x, float y)
{
    super(MM, GS);
    PosX = x;
    PosY = y;
    SpX = 0.0;
    SpY = 0.0;
    Dead = false;

    CStunMine* LinkedBuilding = NullObj;
}

void CMineSpriter::Draw()
{
    if(Dead) return;

    if(LinkedBuilding != NullObj)
    {
        if(StrCmp(LinkedBuilding->ClassType,"Electro Explosion"))
        {
             CQuad q;
             FillSpriteQuad(&q, ElectroExpSprite->Index, LinkedBuilding->AnimPhase, 0);
             TransformQuad(&q,0,1,1,LinkedBuilding->PosX,LinkedBuilding->PosY);
             q.blend = 0;
             for (local i = 0; i < 4; i += 1) q.v[i].color = -1;
	     RenderQuad(&q, 0);
        }
        if(StrCmp(LinkedBuilding->ClassType,"Stun Sprite"))
        {
             CQuad q;
             FillSpriteQuad(&q, StunSprite->Index, LinkedBuilding->AnimPhase, 0);
             TransformQuad(&q,0,1,1,LinkedBuilding->PosX,LinkedBuilding->PosY);
             q.blend = 0;
             for (local i = 0; i < 4; i += 1) q.v[i].color = -1;
	     RenderQuad(&q, 0);
        }
    }
}

void CMineSpriter::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
     if(Type == M_FRAME)
     {
               if(LinkedBuilding != NullObj)
               {
               Destroy();
               PosX = 0;
               PosY = 0;
               }
               else {Dead = true; return;}
     }
}

void CMineSpriter::Destroy()
{
      if (LinkedBuilding->Health < 1) //Destroying
      {
      Dead = true; return;
      }
}
