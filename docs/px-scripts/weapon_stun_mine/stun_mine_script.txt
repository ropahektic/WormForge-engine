void CStunMine::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
//basic - death, falling, animphase
AnimPhaseProcess();
Destroy();
   if (Type == M_FRAME)
   {
      if(SpX != 0.0)SpX = 0.0;
      if(SpY != 0.0)SpY = 0.0;
   }

  if(Type == M_GUNEXP || Type == M_EXPLOSION)
  {
  return;
  }

super;
}


void CStunMine::Collide ( CGObject *obj, int type )
{
super;
}

void CStunMine::Reflect ( CGObject *obj, int type )
{
super;
if(type == 2 || type == 4)obj->SpX = -1 * obj->SpX;
if(type == 1 || type == 3)
	{
    if(obj->ClType == OC_Missile)
    	{
        obj->SpY = -1 * obj->SpY;
        }
 	}
}

void CStunMine::Destroy()
{
      if (Health < 1) //Destroying
      {
      Free(true); return;
      }

      if(PosY > GS->LevelSY)
      {
      Health = 0;
      }
}

void CStunMine::AnimPhaseProcess()
{
     if(AnimPhase < 1.0)
     {
     AnimPhase = AnimPhase + AnimPhaseSpeed;
     }

     if(AnimPhase > 0.99)
     {
       if(StrCmp(ClassType,"Electro Explosion"))
       {
       Health = 0;
       }
       AnimPhase = 0.0;
     }
}

void CreateSMSprite()
{
    for(int i = 0; i < Env->Objs.Count; i += 1)
	   {
		obj = CGObject(Env->Objs.Objs[i]);
		if(obj == NullObj) continue;
		if(obj->ClType == EObjectClass(88))
			{
            local building = CStunMine(obj);
            if(building->HasSprite == false)
                {
                building->HasSprite = true;
                MM->MakeSpriter(building->PosX, building->PosY, building);
                }
			}
        }
}
