void CSaw::CSawCollission()
{
    local damage = 15;
    w = findWorm(this);
    if (w == NullObj) return;
    if (w is CWorm == false) return;

    if (w!=NullObj && w->hitFrame >= 7)
    {                                   
        w->timesSliced = w->timesSliced + 1; //should go up always, gets reset at sturn start anyways
        if (w->timesSliced <= 20)
        {
            damage = ApplySawKnockback(w, PosX, PosY, 0.77);
        
            DoSound();
        
              
            timesCollided = timesCollided + 1;
        }
        else if (w->timesSliced == 20) ShowMessage("Damn boy!");       
        
        if (w->ObjState!=WS_FROZEN)
           if (!w->invincible)
              DamageWorm(w, damage);
        w->hitFrame = 0;
     }
}

void CSaw::DamageWorm(CWorm* w, int calc)
{
       local team = w->WormTeam;
       local worm = w->WormNumber;
       local hp_now = GS->Info.GetWormHealth(team, worm);

       if (w->dmgDIVIDER != 0.0) calc / w->dmgDIVIDER;   //Armor Vest adjustment.
       
       if (w->timesSliced <= 25) //Higher cap than knockback in case worm is still alive
       {
       int rnd = RandomInt(1,3);

       if (rnd == 1) w->SetState( WS_FLYING ); if (rnd == 2) w->SetState( WS_FLYING2 ); if (rnd == 3) w->SetState( WS_POWERFLY );
       }
       if (w->invincible) return;

       if (hp_now <= 15 && w->Endure)
       {
       GS->Info.SetWormHealth(team, worm, 15);
       return;
       }
       if (hp_now - calc <= 15 && w->Endure)
       {
         GS->Info.SetWormHealth(team, worm, 15);
         return;
       }

       if (hp_now - calc <= 0)
       {
         GS->Info.SetWormHealth(team, worm, 0);
         return;
       }

       if (hp_now > 1) GS->Info.SetWormHealth(team, worm, hp_now - calc);
}

int CSaw::ApplySawKnockback(CGObject* worm, fixed sawX, fixed sawY, float sawSizeFactor)
{
    if (worm == NullObj) return 0;

    // Directional Math (Away from Saw Center)
    float dx = worm->PosX - sawX;
    float dy = worm->PosY - sawY;
    float dist = sqrt(dx*dx + dy*dy);
    if (dist < 0.1) dist = 0.1;

    float nx = dx / dist;
    float ny = dy / dist;

    // Incoming Speed Dynamics
    float vx = worm->SpX;
    float vy = worm->SpY;
    float speedIn = sqrt(vx*vx + vy*vy);

    // Return Force (Conservation of Energy)
    // Takes 75% of incoming speed and adds a gentle base push
    float basePush = 1.5 + (sawSizeFactor * 1.5);
    float bounceStrength = basePush + (speedIn * 0.75);

    // Dot Product Reduction
    float dot = vx * nx + vy * ny;
    if (dot > 0.0)
    {
        bounceStrength = bounceStrength * 0.5; // Properly lowers force if already moving away
    }

    // Velocity Assembly
    // Keep 80% of tangential sliding velocity for smooth arcs
    float tangentX = vx - (dot * nx);
    float tangentY = vy - (dot * ny);

    float finalSpX = (nx * bounceStrength) + (tangentX * 0.8);
    float finalSpY = (ny * bounceStrength) + (tangentY * 0.8);

    float total = sqrt(finalSpX*finalSpX + finalSpY*finalSpY);

    // Enforce a Gradual Minimum
    // If the final speed is too weak, scale it up to 4.0
    if (total < 4.0 && total > 0.01)
    {
        finalSpX = (finalSpX / total) * 4.0;
        finalSpY = (finalSpY / total) * 4.0;
        total = 4.0;
    }
    else if (total <= 0.01) // If completely stationary
    {
        finalSpX = nx * 4.0;
        finalSpY = ny * 4.0;
        total = 4.0;
    }

    // Dynamic Speed Capping
    maxAllowed = 12.5;
    if (worm is CWorm == false) maxAllowed = 17.5;

    if (total > maxAllowed)
    {
        finalSpX = (finalSpX / total) * maxAllowed;
        finalSpY = (finalSpY / total) * maxAllowed;
        total = maxAllowed;
    }

    worm->SpX = finalSpX;
    worm->SpY = finalSpY;

    // Damage Calculation
    dmg = int((speedIn * 1.2) + (sawSizeFactor * 6.0));
    if (dmg > 33) dmg = 33;
    if (dmg < 15) dmg = 15;
    return dmg;
}

CWorm* CSaw::findWorm(CSaw* missile)    //Sentry copy paste, thanks entuser i guess
{
    CWorm* bestTarget = NullObj;

    for (int i = 0; i < Env->Objs.Count; i++)
    {
        CGObject* obj = CGObject(Env->Objs.Objs[i]);  
        
        if (obj == NullObj) continue;
        
        if (obj is CSaw) continue;
        
        objtargetX = obj->PosX;                 
        objtargetY = obj->PosY; 
        
        float objdist = sqrt((PosX - objtargetX) * (PosX - objtargetX) + (PosY - objtargetY) * (PosY - objtargetY));   //proximity
        
        if (objdist<hitboxSize)
        {
        if (obj is CMissile)
        {
             CMissile* miss = CMissile(obj);

             if (miss->launchdata.action == WAction_Bounce)
                {
                    if (miss->sawhitlimit>20) { continue; }
                    else
                    {
                        ApplySawKnockback(miss, PosX, PosY, 0.67); 
                    
                        miss->sawhitlimit = miss->sawhitlimit+1;
                    
                        continue;
                    }
                } 
             else 
                {
                    miss->ExplodeAt(miss->PosX, miss->PosY);
                    miss->Free(true);     
                    continue;
                }
        }
        if (obj is CMine) 
        {      
             CMine* mine = CMine(obj);
             if (mine->sawhitlimit >20)
                 { continue; }
             else
             {
             mine->sawhitlimit = mine->sawhitlimit + 1;
             ApplySawKnockback(mine, PosX, PosY, 0.5);    
             continue;
             }
        }
        if (obj is COilDrum || obj is CCrate)  
        {
             expobj = CGObject(obj);        
             DoExplosion(expobj->PosX, expobj->PosY, 0, 5, 5, 0);                             
             continue;
        }
        if (obj is CArrow || obj is icya)
        {
             arrobj = CArrow(obj);
             PlayLocalSound(28,5, 1.0, 1.0);
             arrobj->Free(true);
        }
        }
        if (objdist>hitboxSize) continue;  
        if (obj is CWorm == false) continue;
        // Only consider worm objects
        if (obj == NullObj || obj->ClType != OC_Worm)
            continue;

        //distT = abs(obj->PosX - PosX);    //proximity

        CWorm* worm = CWorm(obj);

        float targetX = worm->PosX;
        float targetY = worm->PosY;
        float missileX = PosX;
        float missileY = PosY;
        float dist = sqrt((missileX - targetX) * (missileX - targetX) + (missileY - targetY+2) * (missileY - targetY+2));   //proximity
        if (dist > hitboxSize) continue;   
        // Comprehensive validity check
        if (worm == NullObj ||
            worm->ObjState == WS_SINKING ||
            worm->ObjState == WS_DEATH ||
            worm->TargHealth <= 0)
        {
            continue;
        }
                    //bestScore = score;
           bestTarget = worm;

    }
    return bestTarget;
}

CSawSpawn::CSawSpawn(CEditorObject* obj)

   super(Root, GS);

}

void CSawSpawn::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{

     {
      if (Type == M_FRAME)
      {

      }
     }
}  

void CSawSpawn::DoSpawn()
{
     CShootDesc SDesc;
     zero(&SDesc);

     SDesc.Team = 0;
     SDesc.X = bx;
     SDesc.Y = by;
     SDesc.SpX = 0;
     SDesc.SpY = 0;

     saw = new CSaw(GetObject(25, 0), &SDesc);

     SawCollisionUnit = SawRadius;
     saw->sawScale = SawRadius / 100;
     saw->ColMask = new CColMask(SawCollisionUnit, SawCollisionUnit, MakeCircleMask(SawCollisionUnit));
     saw->hitboxSize = saw->sawScale * 200 * saw->scaleToHitboxRatio - 2;
}
