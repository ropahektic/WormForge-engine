void CWorm::thinkingShow(CWorm* worm)
{
     CMessageData thinking_show_msg;
     
     thinking_show_msg.params[0] = worm->WormTeam;
     
     worm->Message(NullObj, M_THINKINGSHOW, 4, &thinking_show_msg);
}

void CWorm::setWeapon(CWorm* worm, string weapon_name, bool visibly)
{
     local wp = GetWeaponByName(weapon_name); 
     
     CMessageData set_weapon_msg;
     
     set_weapon_msg.params[0] = worm->WormTeam;
     set_weapon_msg.params[1] = worm->WormNumber;
     set_weapon_msg.params[2] = wp->GetWeaponIndex();
     
     if(visibly == false)
     {
          set_weapon_msg.params[3] = 0;
          
          worm->Message(NullObj, M_SETWEAPON, 1032, &set_weapon_msg);
     }
     else
     {
          set_weapon_msg.params[3] = 1;  
          
          worm->Message(NullObj, M_SETWEAPON, 16, &set_weapon_msg);
     }
}

void CWorm::enableWeapons(CWorm* worm)
{
     CMessageData enable_weapons_msg;
     
     worm->Message(NullObj, M_ENABLEWEAPONS, 0, &enable_weapons_msg);
}

void CWorm::disableWeapons(CWorm* worm)
{
     CMessageData disable_weapons_msg;
     
     worm->Message(NullObj, M_DISABLEWEAPONS, 0, &disable_weapons_msg);
}
