override void CWorm::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{
    local CurrWorm = GetCurrentWorm();
    isCurrentWorm = (CurrWorm != NullObj && CurrWorm == this);
	if (isCurrentWorm == true)
		{
		if (Type == M_FRAME)
			{
			if (hcrateNextSpawn == 0)
				{
				hcrateNextSpawn = 18000 + (GS->GetRandom() % 7200);  //mc: this is the time interval for health crate. 25200 + %10800 frames = 5-7 minutes.
				}

			if (hcrateCounter >= hcrateNextSpawn)
				{
				hcrateNextSpawn = 0;
				hcrateCounter = 0;

				if (ShouldSpawnCrate()){local rez = SpawnNikolaHealth();}
				
				if (rez == false) {hcrateNextSpawn = 60;} 			// spawn failed, try again in 60 frames
				else {ShowMessage("Health crate spawned.");}
				}


			ncrateCounter = ncrateCounter + 1;
			ucrateCounter = ucrateCounter + 1;
			hcrateCounter = hcrateCounter + 1;

			if (ncrateCounter <= 0){ncrateCounter = 1;}
			if (ucrateCounter <= 0){ucrateCounter = 1;}
			if (hcrateCounter <= 0){hcrateCounter = 1;}
			}
    	}
super;
}

bool SpawnUtilityCrate()
{
    string uWeps[14];
	int uWepsCount = 13;
	
	uWeps[0] = "Crate Grabber";
	uWeps[1] = "Wind Changer";
	uWeps[2] = "Health Zone";
	uWeps[3] = "Ninja Rope";
	uWeps[4] = "Magnetic Missile";
	uWeps[5] = "Sentry Gun";
	uWeps[6] = "Sentry Laser";
	uWeps[7] = "Select Worm";
	uWeps[8] = "Low Gravity";
	uWeps[9] = "Double Jump";
	uWeps[10] = "Damage*2";
	uWeps[11] = "Crate Spy";
	uWeps[12] = "Double Turn Time";
	uWeps[13] = "Crate Shower";

	if (mapsizeX < 100 || mapsizeY < 100)
        {
		mapsizeX = GS->LevelSX;
		mapsizeY = GS->LevelSY;
		}

	CCrateDesc cDesc;
	zero(&cDesc);
	cDesc.X = -1000;
	cDesc.Y = -1000;
	cDesc.type = CT_UTILITY; // CT_WEAPON CT_UTILITY CT_HEALTH

	int ruWep = GS->GetRandom() % (uWepsCount + 1);
	cDesc.wType = WCT_DEFINED;
	cDesc.amount = 1;
	local wep1 = GS->GetWeaponByName(uWeps[ruWep]);

        if (wep1 != NullObj)
        {
		cDesc.weapon = 100 * wep1->GetPageIndex() + wep1->GetWeaponIndex();
	}
        if (wep1 == NullObj){
           local wep1 = GS->GetWeaponByName("Low Gravity");
           string smrd = uWeps[ruWep];
           cDesc.weapon = 100 * wep1->GetPageIndex() + wep1->GetWeaponIndex();
        }

	local nwC = new CCrate(Root->GetObject(25, 0), &cDesc, DT_CHUTE); // creates the crate but it does not have a spawn point yet
	// the crate is created because we need to access its collision mask

	int trywX = mapsizeX / 5;
	int trywY = mapsizeY / 5;
	int startwX = GS->GetRandom() % trywX;
	int startwY = GS->GetRandom() % trywY;
	for (int n = 0; n < (trywX * trywY); n++)
	{
		int nwX = ((startwX + (n % trywX)) % trywX) * 5;
		int nwY = ((startwY + (n / trywY)) % trywY) * 5;
		CGObject * ow = Env->CheckCollision(nwC, nwX, nwY);
		if (ow == NullObj)
		{

			int hitX = 0;
			int hitY = 0;
			if (TraceMask(nwC, nwC->ColMask, nwX, nwY, nwX, nwY + 1000, 4332670, &hitX, &hitY) != NullObj) {
				nwC->PosX = nwX;
				nwC->PosY = nwY;
				nwC->X = nwX;
				nwC->Y = nwY;
				return true; // found a spawn point, return. This means exit this entire function.
			}
		}
	}
	nwC->Free(true); // we did not find a spawn point, lets destroy the crate we have created
	return false;
}

bool ShouldSpawnCrate(){
     local crn = 0;
     for(local i = 1; i < Env->Objs.Count; i+=1){
               local obj = Env->Objs.Objs[i];
               if(obj != NullObj){
                  if(obj->ClType == OC_Crate) crn++;
               }
     }
     if (crn > 4) return false;
     return true;
}
