override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if(Type == M_TURNBEGIN || Type == M_TURNEND || Root->Timer == 0) AutoWeaponManagement();
    
    if(Type == M_AFTERSHOOT && this == GetCurrentWorm()) AutoWeaponManagement();
    
    if(Type == M_TURNEND) CastedConcentrateToday = false;

    if(Type == M_TURNEND && IsSneaky == true && gamestarted < 2) gamestarted = gamestarted + 1;

    if(Type == M_FRAME)                                                         //nolag plx
    {
      if (gframe < 10)
        {
            for(int i = 1; i < Env->Objs.Count; i++)
    		{
        	local obj = Env->Objs.Objs[i];

	        if(obj == NullObj) continue;
    	    if((obj is CWorm) == false) continue;

        	CWorm* worm = CWorm(obj);

	        if(worm == NullObj) continue;

	        if(StrCmp(GetTeamName(WormTeam), "RED LEADER") || StrCmp(GetTeamName(WormTeam), "BLUE LEADER"))
	            {
            	for (local wormIndex = 3; wormIndex <= 8; wormIndex++)
					{
   			 		GS->Info.SetWormHealth( WormTeam, wormIndex, 0);
      				GS->Info.Teams[WormTeam-1].Worms[wormIndex-1].State = 135;
      				CWorm* wormy = Root->GetObject( 2, 16 * WormTeam + wormIndex);
      				if (wormy == NullObj) continue;
      				wormy->Free(true);
  					}
  				}

            if(StrCmp(GetTeamName(WormTeam), "RED SUPPORT") || StrCmp(GetTeamName(WormTeam), "BLUE SUPPORT"))
	            {
            	for (local wormIndex = 4; wormIndex <= 8; wormIndex++)
					{
   			 		GS->Info.SetWormHealth( WormTeam, wormIndex, 0);
      				GS->Info.Teams[WormTeam-1].Worms[wormIndex-1].State = 135;
      				CWorm* wormy = Root->GetObject( 2, 16 * WormTeam + wormIndex);
      				if (wormy == NullObj) continue;
      				wormy->Free(true);
  					}
  				}
			}
        }

	  if(GS->Info.GetWormHealth(WormTeam, WormNumber) > MaxHealth)
      {
      GS->Info.SetWormHealth(WormTeam, WormNumber, MaxHealth);
      TargHealth = MaxHealth;
      }

      if((ObjState == WS_DEAD || ObjState == WS_SINKING || ObjState == WS_DEATH) && Class == "King" && gg == false)  //my name is king and i pay you muni
      {
       for(local i = 1; i < 6; i++)
       {
       local guy = CWorm(Root->GetObject(2, 16 * WormTeam + i));
       if(guy != NullObj){GS->Info.SetWormHealth(guy->WormTeam, guy->WormNumber, 0);}
       }
       gg = true;  //this is the end, my beautiful friend
      }
    }
    super;
}

void CWorm::AutoWeaponManagement()
{
    RemoveWeaps();
    AddWeaps();
}

void CWorm::RemoveWeaps()
{
  local TeamNumber = GetTeamNumber(WormTeam);
  for(local i=1; i<68; i++)
  {
    local weap = GS->GetWeaponByIndex(1, i);
    if(weap == NullObj) continue;
    if(weap->CheckName("Skip Go") || weap->CheckName("Surrender") || weap->CheckName("Auto-ballista") || weap->CheckName("Magic Potion")) continue;

    GS->Info.SetWeaponEx(TeamNumber, i, weap->GetPageIndex(), 0);
  }
}

void CWorm::AddWeaps()
{
  local TeamNumber = GetTeamNumber(WormTeam);
  for(local i=0; i<71; i++)
  {
    if(Weapons[i] == 0) continue;

    local weap = GetWeaponByName(Weapons[i]);
    if(weap == NullObj) continue;

    GS->Info.SetWeaponEx(TeamNumber, weap->GetWeaponIndex(), weap->GetPageIndex(), -1);
  }
}

void CWorm::AddSpells()
{
    if(CastedConcentrateToday == true)
    	{
    	local TeamNumber = GetTeamNumber(WormTeam);
    	
		local weap = GetWeaponByName("Healing Spell");
    	GS->Info.SetWeaponEx(TeamNumber, weap->GetWeaponIndex(), weap->GetPageIndex(), -1);
    	weap = GetWeaponByName("Fireball Spell");
    	GS->Info.SetWeaponEx(TeamNumber, weap->GetWeaponIndex(), weap->GetPageIndex(), -1);
    	weap = GetWeaponByName("Dimensional Door");
    	GS->Info.SetWeaponEx(TeamNumber, weap->GetWeaponIndex(), weap->GetPageIndex(), -1);
    	weap = GetWeaponByName("Concentrate");
    	GS->Info.SetWeaponEx(TeamNumber, weap->GetWeaponIndex(), weap->GetPageIndex(), 0);
    	}
}

override int CTurnGame::GetNextTeam()    //between turns stuff
{
   for(local i = 1; i < Env->Objs.Count; i+=1)
   {
   local obj = Env->Objs.Objs[i];
   if(obj == NullObj) continue;
        if(obj->ClType == OC_Worm)
        {
        CWorm* warm;
        warm = obj;
        warm->WasSeenThisTurn = false;
        warm->CastedConcentrateToday = false;
        }
   }
//GG->WriteToChat( 1, "override int CTurnGame::GetNextTeam()", false);
super;
}
