CSprite* nuke_rocket_aiming;

void nuke_rocket_anim::InitGraphic() {
    CFile* f;                                                                    
//    f = GetAttachment("nuke_rocket.png");
    f = GetAttachment("nuke_rocket_bullet.png");                             
    nuke_rocket_aiming = LoadSprite(f, 1, 0);                                                            
}                                                                                
                                                                                 
override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) { 
 super;                                                                          
  if (Type == M_FRAME) {                                                         
    if (gframe == 1) {                                                           
     local weapName; local sprite;

     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Nuke Rocket";                                               
     sprite = nuke_rocket_aiming->Index;                                     
     Params->hand_radius = 13.0;                                                         
     Params->hand_rotation = -0.6;
     Params->hand_radial_rotation = -0.95;
     Params->weap_radial_rotation = 1.0;
     Params->weap_rotation = 0.45;
     Params->weap_radius = 8.5;
     Params->hand_type = 2;
     Params->dont_rotate_with_angle = true; 
     Params->ang_if_static = -0.1;
     Params->hand_ang_if_static = 0.4;                                       
                                                  
     local WA = new WormSprite(weapName, sprite, Params);
    }                                                                           
  }                                                                              
}                                                                                

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;                                                                          
  if (Type == M_FRAME){                                                          
     if (weap->CheckName("Nuke Rocket") || im_cnuke || nr_first) {                                
         if (nuke_rocket_anim2) {                                              
            local sprite = nuke_rocket_aiming->Index;                      
            WeapSpriteParams* Params = new WeapSpriteParams;
            local wa = new WeapSprite(this, sprite, Params); 
            wa->type = 5;             
            //weap->launch.anim.spriteIndex = 0;   this just crashes and it's a toddler failsafe, index is already 0
            nuke_rocket_anim2 = false;                                         
         } 
     }                                                                                                                                                                                                                          
  }                                                                              
}                                                                                
override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;                                                                           
  nuke_rocket_anim2 = true;                                                    
}
