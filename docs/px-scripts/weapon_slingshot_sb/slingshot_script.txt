require utils, p_sprite_builder, utils_curtis, sprite_curtis_456;

CSprite* slings_aiming;

void slingshot_script::InitGraphic()
{
    CFile *f;
    f = GetAttachment("spb_slings.png");
    slings_aiming = LoadSprite(f, 8, 0);
}

//---------------------------------SLINGSHOT----------------------------------//

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) {
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "SlingShot";
     sprite = slings_aiming->Index;
     Params->draw_hand = false;
     Params->hand_radius = 0.0;
     Params->hand_rotation = 0.0;
     Params->hand_radial_rotation = 0.0;
     Params->weap_radial_rotation = 0.0;
     Params->weap_rotation = 1.0;
     Params->weap_radius = 3.5;
     Params->animate = true;
     Params->anim_speed = 1.10;
     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}

//---------------------------------STONE BLASTER------------------------------//

 override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
 {
   if(Weap->CheckName("Stone Blaster"))
   {
   Desc->SpX = (TurnSide * (0.5 - sqrt((0.5 - FireAngle)*(0.5 - FireAngle)))) * 40;
   Desc->SpY = 40*(0.5 - FireAngle);
   super(GetWeaponByName("Stone"), Desc);
   return;
   }
 super;
 }
