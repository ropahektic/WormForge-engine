void CWorm::MakeThemMines()
{
   local tempObj = new CGObject(Root, 17, 9);
    tempObj->ColGroup = 6190;   // random numbers
    tempObj->MaskIndex = 1;
    tempObj->Layer = 1;
    tempObj->IsMaterial = true;
    tempObj->ClType = EObjectClass(500);
    tempObj->PosX = 0;
    tempObj->PosY = 0;
    tempObj->SpX = 0;
    tempObj->SpY = 0;
    tempObj->SpeedDivider = 1;
   
   CMineParams MParams;
   CShootDesc SDesc;
    zero(&MParams);
    zero(&SDesc);
    
    SDesc.Team = 0;
    SDesc.X = RandomInt(0, GS->LevelSX);
    SDesc.Y = RandomInt(0, GS->LevelSY);
    
    SDesc.Delay = ExplTimer;
    
    MParams.Prefuse = 0;
    MParams.Flags = 60;
    MParams.Bias = 0;
    MParams.Damage = 50;
    MParams.BlastPower = 100;
    MParams.Fuse = 3000; //20 frames
    MParams.Radius = 48;
    
    if (iMine == 1)
    {
    if (Env->CheckCollision(tempObj, SDesc.X, SDesc.Y) != NullObj) {iMine = 2;}
    else {mine = new CMine(Root->GetObject(25, 0), &MParams, &SDesc, false, 0); iMine = 2;}
    }
    if (iMine == 2)
    {
    RanX = RandomInt(0, GS->LevelSX);
    RanY = RandomInt(0, GS->LevelSY);
    if (Env->CheckCollision(tempObj, RanX, RanY) != NullObj) {iMine = 3;} 
    else {drum = new COilDrum(Root->GetObject(25, 0), RanX, RanY, true); iMine = 3;}
    }    
    if (iMine == 3 && WCrateNumNow < WCrateNum)
    {
    local Weap = "Electric Charge";
    local Amount = 5;
    RanX = RandomInt(0, GS->LevelSX);
    RanY = RandomInt(0, GS->LevelSY);
    CCrateDesc CDesc;
    zero(&CDesc);
    CDesc.X = RanX;
    CDesc.Y = RanY;
    CDesc.type = CT_WEAPON;
    local wep1 = GS->GetWeaponByName(Weap);
  
    if (Env->CheckCollision(tempObj, RanX, RanY) != NullObj) {iMine = 4;} 
    else {
    local t = new CCrate(Root, &CDesc, DT_NONE);
    t->desc.weapon = 100 * wep1->GetPageIndex() + wep1->GetWeaponIndex();
    t->desc.wType = WCT_DEFINED;
    t->desc.amount = Amount;
    iMine = 4;
    }
    WCrateNumNow += 1;
    }
    if (iMine == 4 && HCrateNumNow < HCrateNum)
    {
    RanX = RandomInt(0, GS->LevelSX);
    RanY = RandomInt(0, GS->LevelSY);
    CCrateDesc CDesc;
    zero(&CDesc);
    CDesc.X = RanX;
    CDesc.Y = RanY;
    CDesc.type = CT_HEALTH;
    CDesc.health = 150;
    CDesc.highlight = false;
    if (Env->CheckCollision(tempObj, RanX, RanY) != NullObj) {iMine = 1;} 
    else {drum = new CCrate(Root->GetObject(25, 0), &CDesc, DT_NONE); iMine = 1;}
    HCrateNumNow += 1;
    }
    if (HCrateNumNow >= HCrateNum) iMine = RandomInt(1, 2);
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  super;
     if(Type != M_FRAME) return;
     if (Type == M_FRAME)
     {
       if(placeNow)
       {
          if (Tur_k < numTurrets*placingDelay && Tur_k < turretsLimit)
          {
             if (Tur_k/placingDelay == int(Tur_k/placingDelay))
             {
                MakeThem();
                MakeThemMines();
             }
          Tur_k += 1;
          } else 
          {
             if (Tur_k >= numTurrets*placingDelay || Tur_k >= turretsLimit) 
             {
                ShowMessage("Placing turrets: DONE!");
                ShowMessage("Have Fun!");
                if (Health > 0)
                {
                  for (local i = 1; i <= 6; i+=1)
                  {
                  for (local j = 1; j <= 8; j+=1)
                  {
                  if (Root->GetObject(2, j + i * 16))
                  {
                   GS->Info.SetWormHealth(i, j, Health);
                  };
                  };
                  };  
                };
                placeNow = false;
             }
          }
       }
     }
     if (ObjState == WS_SINKING || ObjState == WS_DEATH)
       {
         if (placeRespawnPosNow)
         {
            if (wait_lil > 15) wait_lil = 0;
            if (wait_lil == 15)
            {
            ShowMessage("You will be respawned after 1 turn");
            ShowMessage("You will be respawned after 1 turn");
            ShowMessage("You will be respawned after 1 turn");
//           local weapnowforme = GS->GetWeaponByName("Electric Charge");
//            GS->Info.AddWeapon(WormTeam, weapnowforme, 3);
            placeRespawnPosNow = false;
            }
            wait_lil += 1;
         }
       }
     if(ObjState == WS_IDLE) {
                 LastPosX = PosX;
                 LastPosY = PosY;
                 ILayer = Layer;
                 IColGroup = ColGroup;
                 rControl->SetRespawnPosition(WormNumber, WormTeam, LastPosX, LastPosY);
         }
     if (Type == M_TURNBEGIN)
     {
        placeRespawnPosNow = true;
     }
     if (ObjState == WS_ROPING || ObjState == WS_JUMP || ObjState == WS_SLIDING)
     {
        placeRespawnPosNow = true;
     }
     return;
}
