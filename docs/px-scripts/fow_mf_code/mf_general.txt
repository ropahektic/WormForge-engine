int NextKcrateX;
int WantToSpawnKCrate(int RNGA,int RNGB, int SX, int SY, int startturns)
{
    KCrateList->Compact();
    if(KCrateList->Count >= 2){return 3;} //No more 2 at the time

    local bonus = int(startturns);

    if(bonus > 30 || bonus < 0){bonus = 30;}
    if((RNGA % 100) < (5 + bonus))
		{
        local sp = 20;
        local xstep = SX / sp;
        NextKcrateX = xstep * ((RNGB % sp) + 1);

        if(SX == 2200 && SY == 1504)
			{
            if(NextKcrateX < 105){NextKcrateX = 105; return 1;}
            if(NextKcrateX > 2098){NextKcrateX = 2098; return 1;}
            if(NextKcrateX > 1010 && NextKcrateX < 1195)
				{
                if(RNGB % 2 == 0){NextKcrateX = 555;}
                else {NextKcrateX = 1690;}
                }
            }

		return 1;
        }

	return 0;
}

//Returns the number of cells in a ring around (x,y) that are alive.

int countAliveNeighbours(int inX, int inY)
{
    local count = 0;
    local i = 0;
    local j = 0;
    local neighbour_x = 0;
    local neighbour_y = 0;

    local widthRow = cSizeX;
    local size_x = cSizeX;
    local size_y = cSizeY;

    for(i = -1; i < 2; i++)
		{
        for(j = -1; j < 2; j++)
			{
            neighbour_x = inX + i;
            neighbour_y = inY + j;
            //If we're looking at the middle point
            if(i == 0 && j == 0){} //Do nothing, we don't want to add ourselves in!

            //In case the index we're looking at it off the edge of the map
            //Otherwise, a normal check of the neighbour

            else if( CaveMap[ neighbour_x + ( neighbour_y * widthRow)] == 1){count = count + 1;}
        	}
    	}
    return count;
}

bool GenerateCrSpawnMap(int side, int mSX, int mSY){

    if(NewlyCrate == NullObj){return false;}

    local CrSpawnUnderY = 305;
    local CrSpawnOverY = 1088; //not used?

    CrSearchGridSize = 50;
    local extraX = 0;
    local side2X = 0;
    if(mSX == 2200 && mSY == 1504)
		{
        CrSpawnUnderY = 305;
        CrSpawnOverY = 1088;
        ax = int(880 / CrSearchGridSize);
        ay = int(790 / CrSearchGridSize);
        side2X = 1325;
    	}
    if(mSX == 2200 && mSY == 2000)
		{
        CrSpawnUnderY = 713;
        CrSpawnOverY = 1500;
        ax = int(730 / CrSearchGridSize);
        ay = int(790 / CrSearchGridSize);
        extraX = 80;
        side2X = 1377;
    	}

    local startx = 0;
    local starty = CrSpawnUnderY;

    local dOut = 0;
    local DFbmp = DigFoam;

    SMlastdistL = 0;
    SMlastdistR = 0;
    for(iy = 0; iy < ay; iy++ )            	//int
		{
        for(ix = 0; ix < ax; ix++ )         //int
			{
            local cy = (iy * CrSearchGridSize) + starty;
            local cx = 0;
            if(side == 0){cx = (ix * CrSearchGridSize) + extraX;}
			else {cx = (ix * CrSearchGridSize) + side2X ;}

            local Obje = Env->CheckCollision( NewlyCrate, cx, cy );
            if(Obje != NullObj)
				{
                if(Obje->ClType == OC_Landscape)
					{
                    if(side == 0){MapEmptySpaceL[(iy * ax) + ix] = 0;} //not enough room
					else {MapEmptySpaceR[(iy * ax) + ix] = 0;}
                    continue;
                    }
				else if(Obje->ClType == OC_Crate)
					{
                    if(side == 0){MapEmptySpaceL[(iy * ax) + ix] = 0;} //spawning on another crate
                    else {MapEmptySpaceR[(iy * ax) + ix] = 0;}
                    continue;
                    }
                }

            local hitx1 = 0;
            local hity1 = 0;
            local hit1 = TraceLine(NewlyCrate, cx, cy, cx, CurrWaterLvl + 10, -1, &hitx1, &hity1);

            if (hit1 != NullObj)
                {
                local disty = 0;
                disty = abs(cy - hity1);
                if(disty > 100){disty = 100;}
                if(disty == 0){disty = 1;}

            	if(side == 0)
					{
                    dOut = SMlastdistL + disty;
                    SMlastdistL = dOut;
                    MapEmptySpaceL[(iy * ax) + ix] = dOut;
                    }
				else
					{
                    dOut = SMlastdistR + disty;
                    SMlastdistR = dOut;
                    MapEmptySpaceR[(iy * ax) + ix] = dOut;
                    }
                continue;
                }

			else
				{
                if(side == 0){MapEmptySpaceL[(iy * ax) + ix] = 0;} //water
                else {MapEmptySpaceR[(iy * ax) + ix] = 0;}
                continue;
                }
        }
    }

    CrMapGenerated = true;
    return true;
}

bool WormNotThere(int pX, int pY, bool moreminus)
	{
    CWorm * TWrm;
    if(WormsOnClouds == NullObj){return true;}
    for(int n = 0; n<WormsOnClouds->Count; n++)
        {
        TWrm = WormsOnClouds->Objs[n];
        local morem = 10;
        if(moreminus){morem = 16;}
        if( (abs(TWrm->PosX - pX - morem) < 50) && (abs(TWrm->PosY - pY - morem) < 50) )
			{
            return false;
            }
        }

	return true;
}

				   Worm->PosY > (yoff - 120) && Worm->PosY < (yoff + (size_y * 17)) )
				   	{

                    }
                }
            }
        }

    return false;
}

int TeamLowestHealth;

int TeamHealthDiff()
{
    teamLhp = 0;
    teamRhp = 0;
    local whp = 0;
	for (int j = 6; j >= 1; j-=1)
    	{
        for (int i = 8; i >= 1; i-=1)
            {
            local Worm = CWorm(Root->GetObject(2, 16 * j + i));
            if (Worm)
                {
                if(Worm->TeamSide == 1)
					{
                    whp = GS->Info.GetWormHealth(Worm->WormTeam, Worm->WormNumber);
                    if(whp < 0){whp = 0;}
                    teamLhp += whp;
                    }
				else
					{
                    whp = GS->Info.GetWormHealth(Worm->WormTeam, Worm->WormNumber);
                    if(whp < 0){whp = 0;}
                    teamRhp += whp;
                    }
                }
            }
        }

    if(teamLhp < teamRhp){TeamLowestHealth = 1;}
    else{TeamLowestHealth = 2;}

    return abs(teamLhp - teamRhp);
}

int GenerateWeaponID(int RNGn, int RNGnA, int RNGnB)
{
	local NTurn = StartTurnCounter;
	if(NTurn > 100){NTurn = 100;}
	if(RNGnB <= ( NTurn * 0.5 ) ) {return EndGameBonus[RNGnA % 3];}

	if(RNGn <= 7) {return DropVeryRarely[(RNGnA % 11)];}
	else if(RNGn <= 18) {return DropLess[RNGnA % 5];}
	else if(RNGn <= 43) {return DropMore[RNGnA % 4];}
	else {return DropAverage[RNGnA % 25];}
}

bool CLemming::WaitSome(int amount, int stateat)
{
    if(stateat != StateM){WaitTimer = 0; WaitInEffect = false; return true;}
    WaitTimer++;

    if(WaitTimer > amount)
		{
        WaitTimer = 0; WaitInEffect = false;
        return true;
    	}
	else
		{
        WaitInEffect = true;
        return false;
    	}
}

bool CLemming::CheckCollision()
{
    local Cdir = 0;
    if(WalkingDir == 0){ Cdir = -1; }else{ Cdir = 1; }
    local Obje = Env->CheckCollision( this, PosX+Cdir, PosY-1 );
    if(Obje != NullObj)
		{
        if(Obje->ClType == OC_Landscape)
			{
            StateM = 2; //Collision
            return false;
        	}
    	}
  	return true;
}

    local Obje = Env->CheckCollision( this, PosX, PosY-2 );
    if(Obje != NullObj){

        }
    }
    local Obje = Env->CheckCollision( this, PosX, PosY+2 );
    if(Obje != NullObj){
            if(Obje->ClType == OC_Landscape){
            return 1;
        }
    }
    local Obje = Env->CheckCollision( this, PosX-2, PosY );
    if(Obje != NullObj){
            if(Obje->ClType == OC_Landscape){
            return 2;
        }
    }

    if(Obje != NullObj){
            if(Obje->ClType == OC_Landscape){
            return 3;
        }
    }

  return 4;
}
