void CLemming::LookForFire()
{
	local Obje = Env->CheckCollision( this, PosX, PosY );
    if(Obje != NullObj)
		{
        if(Obje->ClType == OC_SeaBubble || Obje->ClType == OC_Smoke || Obje->ClType == OC_Gas)
			{
            OnFire = true;
            if(Fr < 9000) {OnFireFrame = Fr;}
			else {Fr = 0; OnFireFrame = 0;}
        	}
    	}
}

void CLemming::IsNotMoving()
{
    if(PosX == PosHistoryX && PosY == PosHistoryY) {StandingStill = true;}
		
    PosHistoryX = PosX;
    PosHistoryY = PosY;
}



void CLemming::PickRandomDir()
{
    if(RandomHundo > 50) {WalkingDir = 0;}
	else {WalkingDir = 1;}
}

void CLemming::WalkThere(){
    local Cdir = 0;
    if(WalkingDir == 0){ Cdir = -1; }else{ Cdir = 1; }
    if(InMidAir == 0){

        SpX = SpX+(Cdir* 0.5);
        SpY = SpY-3.0;

    }else{
        if(Fr % 100 == 0){
            if (RandomHundo > 60){
                SpY = SpY-5.0;
                SpX = SpX+(Cdir * 2.0);
            }
         }
    }
}
void CLemming::TurnAround(){
    if(WalkingDir == 0){
        WalkingDir = 1;
    }else{
        WalkingDir = 0;
    }
}
void CLemming::NeedsToGrow(){
    if(WantSize != AtSize){
        StartGrowingNow = true;
    }
}

   if(dir == 3){
        SpY = SpY -2.0;
        SpX = SpX -2.0;
   }
}
void CLemming::DoGrow(){

    if(AtSize > 100){StateM = 1; return; }
    if(Fr % 4 == 0){
        if(AtSize < WantSize){
            AtSize = AtSize + 1;
            ColMask = new CColMask(AtSize,AtSize,MakeCircleMask(AtSize));
        }
        if(AtSize > WantSize){
            AtSize = AtSize + 1;
            ColMask = new CColMask(AtSize,AtSize,MakeCircleMask(AtSize));
        }
        if(AtSize == WantSize){
            StateM = 1; return;
        }
    }
}

{
if(Type == M_DRAWQUEUE)
    {
        local dsize = (AtSize * 7.5) / 100;
        local zdsize = (AtSize * 1.0) / 10;

        CSprite * tmpSpr;
        if(OnFire){
            tmpSpr = lwlkb;
        }else{
            tmpSpr = lwlk;
            //tmpSpr = SprKCrate;
        }


        if(WalkingDir == 0){
            AddSpriteEx(10,PosX,PosY,tmpSpr->Index+262144,abs(sin((GS->Tick/5))) * abs(SpX),CurrRota,dsize);
        }else{
            AddSpriteEx(10,PosX,PosY,tmpSpr->Index,abs(sin((GS->Tick/5))) * abs(SpX),CurrRota,dsize);
        }

        if(StandingStill == true){
            
            AddSpriteEx(9,PosX,(PosY-8)-((GS->Tick/10) % 6),lslp->Index, abs(sin((GS->Tick/15))) ,0.0, zdsize);
            AddSpriteEx(9,PosX+5,(PosY-14)-(((GS->Tick/10)+33) % 6),lslp->Index, abs(sin((GS->Tick/25))) ,0.0, zdsize);
        }

    }

    if(Type == M_FRAME)
    {
        if(Fr < 10000){Fr += 1;}else{Fr = 0;}

        if(Fr % 100 == 0){
            RandomHundo = GS->GetRandom()%100;
        }

        if(Fr % 42){if(OnFire == false){ LookForFire(); }}

        if (Sinking == 1) Free(true);

        if(Fr % 500 == 0){NeedsToGrow();}

        if(Fr % 25 == 0){IsNotMoving();}else{if(PosX != PosHistoryX && PosY != PosHistoryY){ StandingStill = false; } }
        if(Fr % 102 == 0 ){ NearbyWorm();
        if(EnemyIsClose == true ){
            //GG->WriteToChat(11, ItoA(AtSize), false);
            if(AtSize < 10){
                DoExplosion (  PosX,  PosY, 50, 20, 0, 0 );
            }else if(AtSize < 12){
                DoExplosion (  PosX,  PosY, 50, 30, 0, 0 );
            }else if(AtSize < 15){
                DoExplosion (  PosX,  PosY, 50, 40, 0, 0 );
            }else if(AtSize < 18){
                DoExplosion (  PosX,  PosY, 50, 50, 0, 0 );
            }else if(AtSize < 21){
                DoExplosion (  PosX,  PosY, 50, 70, 0, 0 );
            }else if(AtSize < 24){
                DoExplosion (  PosX,  PosY, 50, 90, 0, 0 );
            }else{
                DoExplosion (  PosX,  PosY, 50, 120, 0, 0 );
            }

            Free(true);
            }
        }

        if(Fr % 28 == 0){ CheckAllObj(); }


        if(Fr % 200 == 0 && WaitInEffect == false){
            GetNextState = true;
        }

        if(GetNextState == true){
            GetNextState = false;
            RandomHundo = GS->GetRandom()%100;
            if(StartGrowingNow == true){
                StateM = 3;

            }else if(TurnAroundNow == true){
                //not used

            }else if(StandingStill == true){

                if(RandomHundo > 75){ //25% chance to sleep
                    StateM = 0;
                }else{
                    StateM = 1;
                }
            }


        }


        if(StateM == 0){ //standing / sleeping
                if(WaitSome(500, 0)){
                    RandomHundo = GS->GetRandom()%100;
                    PickRandomDir();
                    StateM = 1;
                }
        }else if(StateM == 1){ //walking
            if(CheckCollision()){
                WalkThere();
            }
        }else if(StateM == 2){
            if(WaitSome(20, 2)){
                TurnAround();
                StateM = 1;
            }
        }else if(StateM == 3){ //Grow
        local testcol = CheckCollisionGrow();
            if(testcol == 4){
                DoGrow();
            }else{ //collision 0up1down 2left3right
                MoveDirection(testcol);
            }
        }
    }
}
