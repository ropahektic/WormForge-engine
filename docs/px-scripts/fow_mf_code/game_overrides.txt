void FWormSel() //Fast walk!
{
    CWorm* currWW = GetCurrentWorm();
    if(currWW != NullObj)
    	{
    	CMessageData msgselectMole;
        msgselectMole.params[0] = currWW->WormTeam;
        msgselectMole.params[1] = currWW->WormNumber;
        msgselectMole.params[2] = 64;
        msgselectMole.params[3] = 0;
        currWW->Message(NullObj, M_SETWEAPON, 1032, &msgselectMole);
        CMessageData msgspaceT;
        msgspaceT.params[0] = currWW->WormTeam;
        currWW->Message(NullObj, M_SPACE, 1032, &msgspaceT);
    	}
}

void FindClosestEnemy()
{
    NearestEnemyWorm = NullObj;
    local currW = GetCurrentWorm();
    local j; local i;
    local shortestDistance = 3000;
    if(currW == NullObj){ return; }

    for (j = 6; j >= 1; j-=1)
    	{
        for (i = 8; i >= 1; i-=1)
            {
            local Worm = CWorm(Root->GetObject(2, 16 * j + i));
            if (Worm)
                {
                if(Worm->TeamSide == currW->TeamSide){continue;}
                local distX = abs(Worm->PosX - currW->PosX);
                local distY = abs(Worm->PosY - currW->PosY);
                local tmpdist = sqrt((distX*distX) + (distY*distY));

				if(tmpdist < shortestDistance)
					{
                    shortestDistance = tmpdist;
                    NearestEnemyWorm = Worm;
                    }
                }
            }
        }
}



override void CTurnGame::FinishTurn(int reason)
{
    //spawn extra crate

    CCrateDesc CDesc;
    local RNGt = GS->GetRandom();
    for(int w = 0; w < 3; w++)
		{
        zero(&CDesc);

        CDesc.X = -1000;
        CDesc.Y = -1000;

        CDesc.type = CT_WEAPON;//cratype;
        CDesc.wType = WCT_RANDOM;//WCT_RANDOM;
        CDesc.weapon = 127;

        //CDesc.health = 25;//RandomInt(1,30);
        local daJetCrate = new CCrate(Root->GetObject(25, 0), &CDesc, DT_TELEPORT);
        daJetCrate->FixedWeapon = 0; //random wep
        daJetCrate->PosX = -1000 + (w * 30);
        daJetCrate->PosY = -1000;
        daJetCrate->X = -1000;
        daJetCrate->Y = -1000;
        daJetCrate->PositionSet = true;

        RNGt = GS->GetRandom();
        local mzX = mapsizeX;
        local mzY = mapsizeY;
        GetNewCrateSpawn(mzX, mzY, RNGt);
        if(CrateSpawnSuccess == true)
			{
            daJetCrate->PosX = NewCrX;
            daJetCrate->PosY = NewCrY;
            daJetCrate->X = NewCrX;
            daJetCrate->Y = NewCrY;
            daJetCrate->PositionSet = true;
            daJetCrate->LookAtMe(NewCrX, NewCrY, 5);
            }
        else
			{
            //could not spawn crate
            daJetCrate->PositionSet = true;
            daJetCrate->Freeing = true;
            daJetCrate->Free(true);
        	}
    }

    local tcworm = GetCurrentWorm();

    if(mapsizeX < 100 || mapsizeY < 100)
		{
        mapsizeX = GS->LevelSX;
        mapsizeY = GS->LevelSY;
    	}

    local mSX = mapsizeX;
    local mSY = mapsizeY;
    local SpawnKCrateResult = 5;

    if(isAnyWormOnClouds(mSY, mSX) == false){ClearCloudBridge(mSY, mSX);}

    if(SpawnKCrateOverride == true)
		{
        SpawnKCrateOverride = false;
        NewKCrate(NextKcrateX, mSX, mSY);
        }
	else
		{
        // A TINY BIT OF BALANCING!
        //is one team lower health than the other by x amount?
        if(TeamHealthDiff() > 100)
			{
        	//TeamLowestHealth
        	//Do we want to spawn crate?

			SpawnKCrateResult = WantToSpawnKCrate(GS->GetRandom(),GS->GetRandom(), mSX, mSY, StartTurnCounter);
	        if(SpawnKCrateResult == 1)
				{
	            //is crate spawning on the low health teams turn?
	            //its the low health teams turn next, spawn crate
	            if(tcworm->TeamSide != TeamLowestHealth){NewKCrate(NextKcrateX, mSX, mSY);}
	            else
					{
	                if(SpawnKCrateResult != 3){SpawnKCrateOverride = true;}
	                }
            	}
            }

		else //normal spawn
			{
            SpawnKCrateResult = WantToSpawnKCrate(GS->GetRandom(),GS->GetRandom(), mSX, mSY, StartTurnCounter);
            if(SpawnKCrateResult == 1){NewKCrate(NextKcrateX, mSX, mSY);}
            }
        }

	local whp = 0;

    StartTurnCounter = StartTurnCounter + 1; // must maybe be here for desync reasons
    CurrWaterLvl = Env->Water;

super;
}
