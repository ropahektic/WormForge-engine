//Variables!
CBuyInterface::CBuyInterface(CObject* parent)
   {
   super(Root, GS);   
   ShopGUI = MakeBitmap(64,80);
   IconBitmap = MakeBitmap(32,32);
   PosX=0;
   PosY=0;
   Weap="Bazooka";
   OrigName="Bazooka"; //Name string occasionaly breaks, this is to prevent so
   Amount=1;
   Page=1;
   CPrice=0;
   DPrice=0;
   GotWeap=false;
   ReadyToGo=false; //This becomes true once the ShopObject successfully finds its stock
   StockNum=1;
   SelWeap=1;
   Updated=false;
   }

//Draw all bitmaps!
void CBuyInterface::DrawBitmap()
   {
    AddBitmap(12,PosX,PosY,ShopGUI,64,80,0);
    DrawIcons();
    AddBitmap(11,PosX-24,PosY+16,CoinBitmap,16,16,0);
    AddBitmap(11,PosX-24,PosY+32,DiamondBitmap,16,16,0);
   }

//Non-bitmap stuff!  
void CBuyInterface::DrawInfo()
   {           			 
   local a = 1;
   local b = 1; 
   local selstr = StrCon(StrCon(StrCon(StrCon("(",ItoA(SelWeap)),"/"),ItoA(StockNum)),")");                                          
   local amstr = StrCon(ItoA(Amount),"x ");
   local cpstr = StrCon("x",ItoA(CPrice));
   local dpstr = StrCon("x",ItoA(DPrice));
   ShopGUI->Rect(0,0,64,80,ARGB(255,0,0,0));	
   GS->Disp->DrawText(3,ShopGUI, 2, 34, amstr, &a, &b);
   GS->Disp->DrawText(3,ShopGUI, 34, 34, selstr, &a, &b);
   GS->Disp->DrawText(7,ShopGUI, 18, 50, cpstr, &a, &b);
   GS->Disp->DrawText(7,ShopGUI, 18, 66, dpstr, &a, &b);	
   }

//Make sure it actually draws.
void CBuyInterface::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
   {
   if (ReadyToGo) {
   if (Type == M_FRAME) 
   {
   if (!GotWeap) //Oh wait, I don't know the weapon yet;
       {
         MyWeap = GS->GetWeaponByName(Weap);
         Icon = GS->GetWeaponIcon(Page,MyWeap->GetWeaponIndex()); 
         IconBitmap = Icon;
         GotWeap=true; //k
       }
   else
       {
       if (Updated) {
          IconBitmap = MakeBitmap(32,32);
          Icon = GS->GetWeaponIcon(Page,MyWeap->GetWeaponIndex()); 
          IconBitmap = Icon;
          Updated=false;
       }       
       DrawInfo();    
       }
   }
   if (GotWeap) {
   if (Type == M_DRAWQUEUE)
       {
       DrawBitmap();
       }
   }
   }
}
