CZone* SearchForZone(string ZName)
{
  for (local i = 0; i < gTrapParent->Childs.Count; i += 1)
  {
    local obj = CZone(gTrapParent->Childs.Objs[i]);
    if (obj is CZone)
    {
      if (StrCmp(obj->zone_name, ZName))
      {
        return obj;
      };
    };
  };
  return NullObj;
};

CZone::CZone(CEditorObject* aobj)
{
  super(gTrapParent, GS);
  pr_x1 = aobj->PosX;
  pr_x2 = aobj->PosX + aobj->GetInt("Size");
  pr_y1 = aobj->PosY;
  pr_y2 = aobj->PosY + aobj->GetInt2("Size");
  AcitveWormOnly = aobj->GetBool("ActiveWormOnly");
  fps = aobj->GetInt("EachNFrames");

  active = false;
  activator = CGObject(NullObj);
  timer = 0;
  zone_name = StrSave(aobj->GetName());
};

void CZone::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if (Type != M_FRAME) return;
  timer += 1;
  if (timer % fps) return;        //performance

  active = false;

  if (AcitveWormOnly)
  {
    local obj = GetCurrentWorm();
    if (obj == NullObj) return;
    if (pr_x1 < obj->PosX && pr_x2 > obj->PosX)
    {
      if (pr_y1 < obj->PosY && pr_y2 > obj->PosY)
      {
        active = true;
        activator = obj;
      };
    };
    return;
  };

  for (local i = 1; i < Env->Objs.Count; i += 1)
  {
    local obj = CWorm(Env->Objs.Objs[i]);
    if (obj == NullObj) continue;
    if (obj->ClType != OC_Worm) continue;

    if (pr_x1 < obj->PosX && pr_x2 > obj->PosX)
    {
      if (pr_y1 < obj->PosY && pr_y2 > obj->PosY)
      {
        active = true;
        activator = obj;
        break;
      };
    };
  };
};

void CTrap::Shoot()
{
  CShootDesc sdesc;
  zero(&sdesc);

  sdesc.X = PosX;
  sdesc.Y = PosY;

  local fSpread = (Spread / 180.0) * 3.141;

  local da = Angle + srandom_b() * fSpread;
  if (isBadSphere)
  {
    local daa = (EndAngle - Angle) / nAngles;
    da = daa * curAngle + Angle;

    sdesc.X += Spread * sin(da);
    sdesc.Y -= Spread * cos(da);

    curAngle += 1;
    if (curAngle > nAngles) curAngle = 0;
  };

  local dx = sin(da) * (Force + srandom_b() * Force * 0.1);
  local dy = -cos(da) * (Force + srandom_b() * Force * 0.1);

  sdesc.SpX = dx;
  sdesc.SpY = dy;
  sdesc.Delay = gTimer;

  local ldata;

  if (Type == 5)
  {
    CMineParams MParams;
    MParams.Prefuse = 0;
    MParams.Flags = 60;
    MParams.Bias = 0;
    MParams.Damage = MineDamage;
    MParams.BlastPower = 100;
    MParams.Fuse = MineFuse; //20 frames
    MParams.Radius = MineSensorRadius;
    MParams.Unknown = 0;

    local Mine = new CMine(Root->GetObject(25, 0), &MParams, &sdesc, false, 0);
    Mine->autoTurnDestruct = MineRemoveOnTurn;
    Mine->autoDestroyTimer = MineDestroyTimer;
    return;
  }

  if (Type == 1)
    ldata = GetWeaponByName("Bazooka");
  if (Type == 2)
    ldata = GetWeaponByName("Grenade");
  if (Type == 3)
    ldata = GetWeaponByName("Mortar");
  if (Type == 4)
  {
    ldata = GetWeaponByName("Homing Missile");
    sdesc.AddX = zone->activator->PosX;
    sdesc.AddY = zone->activator->PosY;
  };

  if (ldata == NullObj) return;

  if (Type == 4) Missile_HTarget = zone->activator;

  if (Type == 6)
    ldata = GetWeaponByName("Sheep Launcher");   // Firework1 in obj enum
  if (Type == 7)
    ldata = GetWeaponByName("Banana Bomb");    // Firework2 in obj enum

  local missile = new CMissile(Root->GetObject(25, 0), &ldata->launch, &sdesc);
};

void CTrap::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if (Type != M_FRAME) return;

  if (Root->IsTimerActive() == false)
  {
    if (timerToDisable <= 0)
    {
      return;
    } else {
      timerToDisable -= 1;
    };
  } else {
    timerToDisable = 60;
  };
  if (zone == NullObj)
  {
    zone = SearchForZone(ZoneName);
    return;
  };

  if (ShootNow)
  {
     if (zone->active)
     {
        if (timer3 == 0)
        {
           Shoot();
           timer3 = 1;
        }
     } else {
            timer3 = 0;
            }
  } else {
    if (active == false)                           //the following code is complex
    {                                              //because it supports pre-delay,
      if (zone->active)                            //shooting delay, and post-delay
      {                                            //lol.
        timer += 1;
        if (timer == ShootDelay)
        {
          active = true;
          curAngle = 0;
//          ShowMessage("I M ACTIVATED ASDASDASD!");
          timer2 = 0;
        };
      } else {
        timer = 0;
      };
    } else {
      if (timer2 > Delay)
      {
        timer2 = 0;
        Shoot();
      } else {
        timer2 += 1;
      }
    };

    if (zone->active == false)
    {
      timer -= 1;
      if (timer <= 0)
      {
        active = false;
        timer = 0;
        timer2 = 0;
      };
    };
  };
};

CEGameControl::CEGameControl(CEditorObject* aobj)
{
  gControl = this;
  super(gTrapParent, GS);
  noWind = aobj->GetBool("NoWind");
  gNoWaiting = aobj->GetBool("NoWaiting");
  Health = aobj->GetInt("ForceHealth");
  MaxHealth = aobj->GetInt("MaxHealth");
  if (Health > 0)
  {
    for (local i = 1; i <= 6; i+=1)
    {
      for (local j = 1; j <= 8; j+=1)
      {
        if (Root->GetObject(2, j + i * 16))
        {
          GS->Info.SetWormHealth(i, j, Health);
        };
      };
    };
  };

  if (Root->GetObject(2, 1 + 16))
  {
     WormsHealth = GS->Info.GetWormHealth(1, 1);
  };

  for (local i = 0; i < 48; i += 1) gCheckpoints[i] = NullObj;

  respawn = aobj->GetBool("EnableRespawn");
  nlifes = aobj->GetInt("Lifes");
  rDelay = aobj->GetInt("RespawnDelay");

  if (respawn)
  {
    rControl->Enable(rDelay);
  }
  cInitialized = false;
};

void CEGameControl::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if (Type != M_FRAME) return;
  if (noWind)
  {
    Env->Wind = 0;
  };

  if (cInitialized == false)
  {
    for (local j = 1; j <= 6; j += 1)
    {
      gAliveWorms[j] = 0;
      for (local i = 1; i <= 8; i += 1)
      {
        local objWorm = CWorm(Root->GetObject(2, i + j * 16));
        if (objWorm)
        {
          local wIndex = i + j * 8 - 9;
          gStartPosX[wIndex] = objWorm->PosX;
          gStartPosY[wIndex] = objWorm->PosY;
          rControl->SetRespawnPosition(i, j, objWorm->PosX, objWorm->PosY);
          gWormNames[wIndex] = objWorm->GetName();
          gAliveWorms[j] += 1;
        };
      };
    };
    cInitialized = true;
  };
};

  {
    local tindex = worm->WormTeam;

    PosX += offX;
  };
};

      {
       PlaySound(104, 10, 0.80, 1.0);

       ShowMessage("Checkpoint! ;D");
//       orderCheck--;
      } else if (order == 1) {PlaySound(104, 10, 0.80, 1.0); ShowMessage("Checkpoint! ;D");}
      gCheckpoints[wIndex] = this;
      rControl->SetRespawnPosition(wormObj->WormNumber, wormObj->WormTeam, PosX - 5, PosY - 15);
    }
  }
}


CMapTeleport::CMapTeleport(CEditorObject* aobj)
{
  super(gTrapParent, GS);
  pr_x1 = aobj->PosX;
  pr_x2 = aobj->PosX + aobj->GetInt("Size");
  pr_y1 = aobj->PosY;
  pr_y2 = aobj->PosY + aobj->GetInt2("Size");
  isRight = aobj->GetBool("IsToRightSide");
  isExit = aobj->GetBool("ExitOnly");
  tpMissiles = aobj->GetBool("TeleportMissiles");
  trapTeleport = aobj->GetBool("TrapTeleport");
  landTeleport = aobj->GetBool("LandTeleport");

  if (isExit) landTeleport = false;

  hSize = aobj->GetInt("HighlightSize") / 2;
  hColor = aobj->GetInt("Color");
  hX = 0; hY = 0; hTraced = false;

  hSize = (pr_x2 - pr_x1) / 2;



  if (isRight)
  {
    tp_x = int(pr_x1 + ((pr_x2 - pr_x1) / 3) * 2);
  } else {
    tp_x = int(pr_x1 + (pr_x2 - pr_x1) / 3);
  };

  tp_y = int((pr_y1 + pr_y2) / 2);

  timer = 0;
  waiting = false;
  LinkedTo = StrSave(aobj->GetString("LinkedTo"));
  t_name = StrSave(aobj->GetName());
  linked = CMapTeleport(NullObj);

  if (landTeleport)
  {
    touchSensor = new CTouchSensor(this, pr_x1, pr_y1, pr_x2 - pr_x1, pr_y2 - pr_y1);
  }
};

void CMapTeleport::TeleportObject(CGObject* obj)
{
 if (obj->ColMask->Check(linked->tp_x, linked->tp_y, CGObject(Env->Objs.Objs[0])->ColMask, 0, 0))
 {
    GG->land->MakeHole( 15, linked->tp_x, linked->tp_y);
 }

 if (obj->ClType == OC_Worm)
 {
  waiting = true;
  linked->waiting = true;

  local worm = CWorm(obj);

  if (worm->ObjState == WS_WALKING) worm->SetState(WS_IDLE);
  if (worm->ObjState == WS_ROPING) worm->SetState(WS_IDLE);

  worm->PosX = linked->tp_x;
  worm->PosY = linked->tp_y;
  if (isRight == linked->isRight)
  {
    worm->SpX = -worm->SpX;
    worm->TurnSide = -worm->TurnSide;
  }
 } else {
     obj->PosX = linked->tp_x;
     obj->PosY = linked->tp_y;
     if (isRight == linked->isRight) obj->SpX = -obj->SpX;
 }
}

void CMapTeleport::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if (Type == M_TOUCHSENSOR_COLLIDE)
  {
    if (linked != NullObj)
    {
       local ts = CTouchSensor(sender);
       local activator = ts->activator;
       if (activator->ClType != OC_Worm)
       {
         if (!tpMissiles || (activator->ClType != OC_Missile)) return;
       } else {
         if (!trapTeleport && (activator != GetCurrentWorm())) return;
       }
       if (timer <= 0)
       {
          MData->params[0] = 1; //disable collision
       }
    }
    return;
  }
  if (Type == M_TOUCHSENSOR)
  {
     if (linked == NullObj) return;

     if (timer > 0)
     {
       timer = 2;
       return;
     }

     local ts = CTouchSensor(sender);
     local activator = ts->activator;

     if (activator->ClType != OC_Worm)
     {
       if (!tpMissiles || (activator->ClType != OC_Missile)) return;
     }  else {
       if (!trapTeleport && (activator != GetCurrentWorm())) return;
     }

     TeleportObject(activator);

     if (activator->ClType == OC_Worm)
     {
       ts->noCollide = true;
     } else {
       ts->noCollide = false;
     }

     timer = 2;
     linked->timer = 2;
     return;
  }
  if (Type == M_DRAWQUEUE)
  {

     if (hTraced && hSize > 0 && hColor != 0)
     {
        CQuad q;
        q.blend = 0; q.tex = 0;
        q.v[0].x = hX - hSize;
        q.v[0].y = hY;
        q.v[1].x = hX - hSize;
        q.v[1].y = hY - 20;
        q.v[2].x = hX + hSize;
        q.v[2].y = hY - 20;
        q.v[3].x = hX + hSize;
        q.v[3].y = hY;

        cAdd = int( float(random() * 25) ) << 24;

        q.v[0].color = hColor | $A0000000 + cAdd;
        q.v[1].color = hColor & $00FFFFFF;
        q.v[2].color = hColor & $00FFFFFF;            //hColor & $00FFFFFF
        q.v[3].color = hColor | $A0000000 + cAdd;
        AddSpriteQ(26, &q, 0, 0, 0);
     }
  }

  if (Type != M_FRAME) return;
  if (linked == NullObj)
  {
    if (StrCmp(LinkedTo, "")) return;
    for (local i = 0; i < gTrapParent->Childs.Count; i += 1)
    {
      local obj = CMapTeleport(gTrapParent->Childs.Objs[i]);
      if (obj is CMapTeleport)
      {
        if (StrCmp(obj->t_name, LinkedTo))
        {
          linked = obj;
          linked->linked = this;
          if (hColor != 0)
          {
            linked->hColor = hColor;
          } else if (linked->hColor != 0)
          {
            hColor = linked->hColor;
          }
          break;
        }
      }
    }
    return;
  }

  if (landTeleport)
  {
    if (timer > 0) timer--;
    return;
  }

  if (!hTraced)
  {
     if (hColor != 0 && hSize != 0)
     {
        local fx = (pr_x1 + pr_x2) / 2;
        local fy = (pr_y1 + pr_y2) / 2;
        local obj = TraceLine(NullObj, fx,  fy, fx, fy + 40, -1, &hX, &hY);
        hTraced = true;
        hY += 2;
        if (obj == NullObj) hSize = 0;
     }
  }

  if (isExit)
  {
    waiting = false;
    return;
  }


  if (trapTeleport)
  {
     for (local i = 1; i < Env->Objs.Count; i++)
     {
        local w = CWorm(Env->Objs.Objs[i]);
        if (w == NullObj) continue;
        if (w->ClType != OC_Worm) continue;         //faster than IS operator, btw.

        if (pr_x1 < w->PosX && pr_x2 > w->PosX && pr_y1 < w->PosY && pr_y2 > w->PosY)
        {
          TeleportObject(w);
        }
     }
     return;
  }

  if (tpMissiles)
  {
     for (local i = 1; i < Env->Objs.Count; i++)
     {
        local m = CMissile(Env->Objs.Objs[i]);
        if (m == NullObj) continue;
        if (m->ClType != OC_Missile) continue;         //faster than IS operator, btw.

        if (pr_x1 < m->PosX && pr_x2 > m->PosX && pr_y1 < m->PosY && pr_y2 > m->PosY)
        {
          if (m->tLastTeleport != this)
          {
            m->tLastTeleport = linked;
            TeleportObject(m);
          }
        } else {
          if (m->tLastTeleport == this) m->tLastTeleport = NullObj;
        }

     }
  }

  local worm = GetCurrentWorm();
  if (worm == NullObj)
  {
     waiting = true;
     return;
  }
  if (pr_x1 < worm->PosX && pr_x2 > worm->PosX)
  {
    if (pr_y1 < worm->PosY && pr_y2 > worm->PosY)
    {
      if (waiting == false && linked->waiting == false)
      {
            TeleportObject(worm);
            PlaySound(67, 10, 1, 1);
      }
      return;
    }
  }
  waiting = false;
}

bool CCustomLibTask::MakeArrow()
{
    local dmg = admg;     // damage

//    powaX = ShootSX;        // combination of these 2 are angle and power :)))
//    powaY = ShootSY;
//                       Visual Extrapolation:
//
//                             X:1,Y:-1
//                               *
//                               |
//                 X:-1,Y:1 *\       /* X:1,Y:-1
//
//              X:-1,Y:0 *---          ---* X:1,Y:0
//
//                          /        \
//               X:-1,Y:-1 *     |    * X:1,Y:1
//                               *
//                            X:0,Y:0
// Other words:
//            (X:10.0,Y:0.0)=90_degree.      (Y:10.0,X:0.0)=0_degree. (X:10.0,Y:10.0)=135_degree. (X:10.0,Y:-10.0)=45_degree
//            (X:-10,Y:0.0)=-90_degree(270). etc

    CShootDesc SDesc;
    zero(&SDesc);

    SDesc.Team = 1;
    SDesc.Worm = 0;

    SDesc.X = ShootX;   // set X position to shot
    SDesc.Y = ShootY;   // set Y position to shot

    SDesc.SpX = ShootSX;
    SDesc.SpY = ShootSY;

    local arrow;


    arrow = new CArrow(Root->GetObject(25, 0), &admg, &SDesc);
    if (CGObject(arrow)->IsValid())
    {
      arrow->GravityFactor = gravArrow;
      arrow->DontStick = noStick;
    }
    return true;

};

void CCustomLibTask::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
   if (Type == M_TOUCHSENSOR)
  {
    if (timerToDisable <= 0) return;
    if (timerToEnable > 0) return;
    local activator = CTouchSensor(sender)->activator;
    HandleObject(activator);
  }
  if (Type != M_FRAME) return;
  if (Root->IsTimerActive() == false)
  {
    if (timerToDisable <= 0)
    {
      timerToEnable = 13;
      return;
    } else {
      timerToDisable -= 1;
    }
  } else {
    timerToDisable = 60;
  }
  timer += 1;
  if (timerToEnable > 0)
  {
    timerToEnable -= 1;
    return;
  }

  if (timer % fps) return;        //performance

 if (zone == NullObj)
   {
     zone = SearchForZone(ZoneName);
     return;
   }
 if (zone->active == false)
 {
     MakeNow = true;
     return;
 }

//  if (Type == M_TOUCHSENSOR)
//  {
//    if (timerToDisable <= 0) return;
//    if (timerToEnable > 0) return;
//    local activator = CTouchSensor(sender)->activator;
//    HandleObject(activator);
//  }

     local obj = CWorm(zone->activator);
     if (obj is CWorm)
     {
        if (MakeOnce)
        {
            if(MakeNow)
            {
                HandleObject(obj);
                MakeNow = false;
            }
        }
        else
        {
            HandleObject(obj);
        }
     }
};

CSpriter_1by1::CSpriter_1by1(CEditorObject* aobj)
{
  super(gTrapParent, GS);
  ZoneName = StrSave(aobj->GetString("ProximityZone"));
  zone = CZone(NullObj);
  PosX = aobj->PosX;
  PosY = aobj->PosY;

  order = aobj->GetInt("Order");
//  checkOrder = 0;
  ang1 = aobj->GetInt("RotationAngle");
  ang = (ang1 / 360 + MATH_HALF_PI) * 6.3;
  scl = aobj->GetInt("Scale");
  scl = float(scl)/100.0;
  if (ang > 0) rotate = true;
  if (scl < 1) scale = true;
  SprNum = aobj->GetInt("SpriteNum");
  PlayOnce = aobj->GetBool("PlayOnce");
  PlayBackwards = aobj->GetBool("PlayBackwards");
  playnow =  true;
  waSoundNum = aobj->GetInt("PlayWaSound");
  if (waSoundNum == 0) {playSound = false;} else playSound = true;
  playSoundNow = playSound;
  checkLoop = 0;

  local as = aobj->GetInt("AnimSpeed");
  as = float(as)/1000.0;
  AnimationSpeedT = 0.0;     // aobj->GetInt("AnimSpeed") / 100;
  AnimationSpeedT = as;
  AnimationPhaseT = 0.0;

  ZPlane = aobj->GetInt("ZPlane");
  CSpriter_1by1* Spriter_1by1;
};

void CSpriter_1by1::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
   if (zone == NullObj)
   {
     zone = SearchForZone(ZoneName);
     return;
   }
 if (zone->active == false)
 {
     AnimationPhaseT = 0.0;
     playnow = true;
     playSoundNow = true;
     checkLoop = 0;
     return;
 }
   local worm = CWorm(zone->activator);
   if (worm is CWorm)
   {
      if(Type == M_DRAWQUEUE)
      {
         if (order == fc)
        {
                 if (playnow)
                 {
                     if (rotate || scale)
                     { AddSpriteEx(ZPlane,PosX,PosY,SprNum,AnimationPhaseT,ang,scl);
                     } else AddSprite(ZPlane,PosX,PosY,0,SprNum,AnimationPhaseT);
                 }
                 if (playSound) {if (playSoundNow) PlaySound(waSoundNum, 10, 1, 1); playSoundNow = false;}
        }
      }
   }
   if(Type == M_FRAME)
   {
      if (PlayBackwards)
      {
          if (checkLoop == 0) {AnimationPhaseT = AnimationPhaseT + AnimationSpeedT;}
          if(AnimationPhaseT > 0.99){ if(checkLoop == 0) {AnimationPhaseT = AnimationPhaseT - AnimationSpeedT; checkLoop = 1;}}
          if (checkLoop == 1) { AnimationPhaseT = AnimationPhaseT - AnimationSpeedT;}
          if(checkLoop == 1){ if(AnimationPhaseT <= 0.00){AnimationPhaseT = AnimationPhaseT + AnimationSpeedT; if (PlayOnce){checkLoop = -1; playnow = false;}else{checkLoop = 0;}}}
      } else
      {
          if(AnimationPhaseT > 0.99){AnimationPhaseT = 0.0; if (PlayOnce)playnow = false;}
          else if (playnow){AnimationPhaseT = AnimationPhaseT + AnimationSpeedT;}
      }
   }
}

void CCoinBonus::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
     if (Type == M_TURNBEGIN) //Get all health values at turn start
     {                        //to compare with previous turn's.
        local coinbonus = 0;
        local totaldamage = 0;
        for (local j = 1; j <= 6; j += 1) //Every Team
        {
        for (local i = 1; i <= 8; i += 1) //Every Worm
        {
            local objWorm = CWorm(Root->GetObject(2, i + j * 16));
            if (objWorm)
            {
               local curhealth = GS->Info.GetWormHealth(j, i);
               if ((j != lastteam) && (lastteam != -1)) //No bonus for friendly fire!
               {
                 if (curhealth<initialhp[i + j * 10]) { //Ouch, that hurt.
                  totaldamage += initialhp[i + j * 10]-curhealth; //Bonus!
                 }
               }
               initialhp[i + j * 10] = curhealth; //Got it!
            }
            else //It might have died.
            {
              if (initialhp[i + j * 10] > 0) //Did it really die?
             {
                if ((j != lastteam) && (lastteam != -1)) //No bonus for suiciding!
                {
                totaldamage += initialhp[i + j * 10]; //Bonus for you!
                }
                initialhp[i + j * 10] = 0; //Reset the HP anyways
             }
            }
        }
        }
        if ((totaldamage>=sbdamage) && (lastteam > 0)) { //Eligible for Small Bonus?
           local minibonus = (totaldamage/sbdamage)*sbcoins;
           local megabonus = 0;
           local gigabonus = 0;
           if (totaldamage >= bbdamage) { //Eligible for Better Bonus?!
              megabonus = (totaldamage/bbdamage)*bbcoins;
              if (totaldamage >= hbdamage) { //Eligible for HUGE Bonus?!!
                 gigabonus = (totaldamage/hbdamage)*hbcoins;
              }
           }
           local wep = GS->GetWeaponByName("Gold Coin");
           GS->Info.AddWeaponEx(lastteam,wep->GetWeaponIndex(),wep->GetPageIndex(),minibonus);
           //Compliment player and display bonus values
           if (gigabonus>0)
           {ShowMessage(StrCon(StrCon(StrCon(StrCon("Awesome! ",GetTeamName(lastteam))," got "),ItoA(minibonus+megabonus+gigabonus))," coins!"));}
           else if (megabonus>0)
           {ShowMessage(StrCon(StrCon(StrCon(StrCon("Nice! ",GetTeamName(lastteam))," got "),ItoA(minibonus+megabonus))," coins!"));}
           else
           {ShowMessage(StrCon(StrCon(StrCon(GetTeamName(lastteam)," got "),ItoA(minibonus+megabonus+gigabonus))," coins."));}
        }
     }
     if (Type == M_TURNEND) //Which team has just played?
     {
        lastteam = Root->CurTeam;
     }
}

void CShopInterface::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{

     if (Type == M_FRAME)
     {
      if (GotSound) //Got our CSound already? Ready to work!
      {
      if (zone == NullObj) //We actually have a link
        {
        zone = SearchForZone(ZoneName); //Where is it?
        return;
        }
      else {
           if (GotStock)
           { //We have our weapons!
           if (zone->active) //Worm detected!
           {
            CWorm *ActWorm = GetCurrentWorm();
            if(ActWorm != NullObj) {
                if (keySync->KeyPressedNow(66)) { //B Pressed, attempting to buy!
                   local wep = GS->GetWeaponByName("Gold Coin");
                   local wep2 = GS->GetWeaponByName("Diamond");
                   //Got enough money?
                   if (GS->Info.AddWeaponEx(Root->CurTeam,wep->GetWeaponIndex(),wep->GetPageIndex(),0) >= shop_Price[(MyShopID*20)+SelWeap] && GS->Info.AddWeaponEx(Root->CurTeam,wep2->GetWeaponIndex(),wep2->GetPageIndex(),0) >= shop_Price2[(MyShopID*20)+SelWeap] )
                   {
                   ShowMessage("Succesfully bought!");
                   SoundObj = CSound(NullObj);
	           PurchaseSound->Play(1.0, 0.0, false);
                   GS->Info.AddWeaponEx(Root->CurTeam,wep->GetWeaponIndex(),wep->GetPageIndex(),-shop_Price[(MyShopID*20)+SelWeap]);
                   GS->Info.AddWeaponEx(Root->CurTeam,wep2->GetWeaponIndex(),wep2->GetPageIndex(),-shop_Price2[(MyShopID*20)+SelWeap]);
                   GS->Info.AddWeaponEx(Root->CurTeam,shop_ActualWeap[(MyShopID*20)+SelWeap]->GetWeaponIndex(),shop_Page[(MyShopID*20)+SelWeap],shop_Amount[(MyShopID*20)+SelWeap]);
                   }
                   else {
                   ShowMessage("Not enough coins/diamonds!");
                   }
                }
                else if (keySync->KeyPressedNow(78)) { //N Pressed, next Weapon please
                 if (SelWeap<StockNum) SelWeap+=1;
                 else SelWeap = 1;
                        //Update our loyal interface
                        sm->SelWeap = SelWeap;
                        sm->Weap = shop_weapname[(MyShopID*20)+SelWeap];
                        sm->OrigName = shop_weapname[(MyShopID*20)+SelWeap];
                        sm->MyWeap = shop_ActualWeap[(MyShopID*20)+SelWeap];
                        sm->Amount = shop_Amount[(MyShopID*20)+SelWeap];
                        sm->CPrice = shop_Price[(MyShopID*20)+SelWeap];
                        sm->DPrice = shop_Price2[(MyShopID*20)+SelWeap];
                        sm->Page = shop_Page[(MyShopID*20)+SelWeap];
                        sm->Updated = true;
                }
            }
           }
           }
           else
           { //No Stock
             obj2 = SearchForStock(shop_obj2name[(MyShopID*20)+1]); //At least one has to exist.
             if (obj2 != NullObj) {
             for(local i = (MyShopID*20)+1; i <= (MyShopID*20)+20; i ++)
             {
              obj2 = SearchForStock(shop_obj2name[i]);
              if (obj2 != NullObj) {
                shop_weapname[i] = obj2->st_weapname; //Filling our arrays...
                shop_Page[i] = obj2->st_Page;
                shop_Amount[i] = obj2->st_Amount;
                shop_Price[i] = obj2->st_Price;
                shop_Price2[i] = obj2->st_Price2;
                shop_ActualWeap[i] = GS->GetWeaponByName(shop_weapname[i]);
                }
                else {
                shop_weapname[i] = "Bazooka"; //Default values
                shop_Page[i] = 1;
                shop_Amount[i] = 1;
                shop_Price[i] = 1;
                shop_Price2[i] = 0;
                }
             }
              GotStock=true;
              sm->SelWeap = SelWeap; //Keep our BuyInterface updated!
              sm->Weap = shop_weapname[(MyShopID*20)+SelWeap];
              sm->OrigName = shop_weapname[(MyShopID*20)+SelWeap];
              sm->Amount = shop_Amount[(MyShopID*20)+SelWeap];
              sm->CPrice = shop_Price[(MyShopID*20)+SelWeap];
              sm->DPrice = shop_Price2[(MyShopID*20)+SelWeap];
              sm->Page = shop_Page[(MyShopID*20)+SelWeap];
              sm->ReadyToGo = true;
             }
             else {
              ShowMessage("No ITEMSTOCK found!");
             }
           }
      }
      }
      else {
      //There we go, our cash register sound effect
      CFile *f;
      f = GetAttachment("kaching.wav");
      PurchaseSound = new CSoundFile(f);
      GotSound=true;
      }
     }
}
