void dan_castle_ktk::InitGraphic()
{
	CFile *f;
	f = GetAttachment("dan_castle.png");			dan_castle = LoadSprite(f, 1, 0);
	f = GetAttachment("dan_cave.png");			dan_cave = LoadSprite(f, 1, 0);
	f = GetAttachment("dan_house.png");			dan_house = LoadSprite(f, 1, 0);
	f = GetAttachment("dan_tower.png");			dan_tower = LoadSprite(f, 1, 0);       	
}

////////////////////////////////////////////////////////////////////////////////

CSpriterdan_castle::CSpriterdan_castle(CEditorObject* aobj)
{
  super(Root, GS); 
  Color_R = aobj->GetInt("Color_R");
  Color_G = aobj->GetInt("Color_G");
  Color_B = aobj->GetInt("Color_B");
  Scale = aobj->GetInt("Size");
  scale = Scale/100.0;
  rot = aobj->GetInt("Rotation")/57.325;
  x = aobj->PosX;
  y = aobj->PosY;        
  
  newx = x+(258*sin(rot))*scale;      // number is half of hight of your image
  newy = y-(258*cos(rot))*scale;      // number is half of hight of your image
  
  if (StrCmp(aobj->GetString("Z_Layer"), "Behind_Map")) z = 25;
  if (StrCmp(aobj->GetString("Z_Layer"), "Above_Map")) z = 15;
  if (StrCmp(aobj->GetString("Z_Layer"), "Above_Worm")) z = 5;
  
  pluslayer = aobj->GetInt("Add_Layer")/10;
  
  if (StrCmp(aobj->GetString("Transparency"), "No")) transp = 0;
  if (StrCmp(aobj->GetString("Transparency"), "Yes")) transp = 1;
  
  if (StrCmp(aobj->GetString("Flip"), "No")) flip = 0;
  if (StrCmp(aobj->GetString("Flip"), "Yes")) flip = 262144;   
}

void CSpriterdan_castle::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_DRAWQUEUE)
    {
        sprite = dan_castle->Index;
        SetColorMod(RGB(Color_R, Color_G, Color_B), transp); 
        AddSpriteEx(z+pluslayer, newx, newy, sprite+flip, 0.0, rot, scale);
        ClearColorMod();
    }
}
