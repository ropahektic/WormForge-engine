void CWorm::RenderNewWorm() {
    local n = WormTeam*10 + WormNumber;
    if (Invisible) animColor = ARGB(150, 255, 255, 255);
    else animColor = ARGB(255, 255, 255, 255);
    CQuad q;
    FillSpriteQuad(&q, currentSpriteIndex[n], currentSpriteFrame[n], 1);
    TransformQuad(&q, WAP->Rotation, WAP->Size * -TurnSide, WAP->Size , currentSpriteX[n], currentSpriteY[n]);
    for (local i = 0; i < 4; i += 1) q.v[i].color = animColor;
    if (Invisible) q.blend = 1;
    else q.blend = 0;
    RenderQuad(&q, 0);
    AddSpriteQ( WAP->ZPlane, &q, currentSpriteIndex[n], currentSpriteFrame[n], 1);
}

override void CWorm::RenderWorm() {
  if (imSmallWorm) RenderNewWorm();
  super;
}

void CWorm::SmallDeactivate()
{
          MaskIndex = wormsOriginalMaskIndex;
          ColMask = wormsOriginalMask;
          PosY -= 9;
          SpX = 0.0;   // we need this to avoid glitches
          SpY = 0.0; 
          local hit1 = CheckMaskAt(this,smallWormMask,PosX-5,PosY-7, -1);
          local hit2 = CheckMaskAt(this,smallWormMask,PosX+5,PosY-7, -1);
          local hit3 = CheckMaskAt(this,smallWormMask,PosX-3,PosY-3, -1);
          local hit4 = CheckMaskAt(this,smallWormMask,PosX+3,PosY-3, -1);
            if (hit1 != NullObj || hit2 != NullObj || hit3 != NullObj || hit4 != NullObj) {
            GG->land->MakeHole( 5, PosX, PosY-4);
            GG->land->MakeHole( 4, PosX-6, PosY+4);
            GG->land->MakeHole( 4, PosX+6, PosY+4);
            GG->land->MakeHole( 4, PosX-6, PosY);
            GG->land->MakeHole( 4, PosX+6, PosY);
            GG->land->MakeHole( 4, PosX, PosY-1);
            }
          ShowMessage("Small worm mode: OFF");
          imSmallWorm = false;
          hideWorm = false;
//          params2->ReadyDeactive = false;
          animatedWorm = false;
          GravityFactor = GravityFactor * 1.333333;
          SpeedDivider = SpeedDivider * 0.666666;
}

void CWorm::SmallActivate()
{
         PosY -= 5;
         wormsOriginalMask = ColMask;
         WAP->Size = 0.85;
         animColor = ARGB(255, 90, 180, 255);    // lol
         wormsOriginalMaskIndex = MaskIndex;
         MaskIndex = 2; // mine = 2
         ColMask = smallWormMask60;
         ShowMessage("Small worm mode: ON");            
         imSmallWorm = true;
         hideWorm = true;
         animatedWorm = true;
         GravityFactor = GravityFactor * 0.75;
         SpeedDivider = SpeedDivider * 1.5; 
}
