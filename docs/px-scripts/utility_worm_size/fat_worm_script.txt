void CWorm::FatDeactivate()
{
//         if (imFatWorm && ObjState == WS_IDLE) {
          MaskIndex = wormsOriginalMaskIndex;
          PosY -= 9;
          SpX = 0.0;   // we need this to avoid glitches
          SpY = 0.2; 
          local hit1 = CheckMaskAt(this,smallWormMask,PosX-5,PosY-7, -1);
          local hit2 = CheckMaskAt(this,smallWormMask,PosX+5,PosY-7, -1);
          local hit3 = CheckMaskAt(this,smallWormMask,PosX-3,PosY-3, -1);
          local hit4 = CheckMaskAt(this,smallWormMask,PosX+3,PosY-3, -1);
            if (hit1 != NullObj || hit2 != NullObj || hit3 != NullObj || hit4 != NullObj) {
            GG->land->MakeHole( 5, PosX, PosY-4);
            GG->land->MakeHole( 4, PosX-6, PosY+4);
            GG->land->MakeHole( 4, PosX+6, PosY+4);
            GG->land->MakeHole( 4, PosX-6, PosY);
            GG->land->MakeHole( 4, PosX+6, PosY);
            GG->land->MakeHole( 4, PosX, PosY-1);
            }
          ColMask = wormsOriginalMask;
          ShowMessage("Fat worm mode: OFF");
          imFatWorm = false;
          hideWorm = false;
          WAP->ReadyDeactive = false;
          animatedWorm = false;
          GravityFactor = GravityFactor * 0.8;       // 0.6
          SpeedDivider = SpeedDivider * 1.666666;    // 1.666666
//         }
}

void CWorm::FatActivate()
{
         PosY -= 30;
         wormsOriginalMask = ColMask;
          local hit1 = CheckMaskAt(this,smallWormMask,PosX-5,PosY-7, -1);
          local hit2 = CheckMaskAt(this,smallWormMask,PosX+5,PosY-7, -1);
          local hit3 = CheckMaskAt(this,smallWormMask,PosX-3,PosY-3, -1);
          local hit4 = CheckMaskAt(this,smallWormMask,PosX+3,PosY-3, -1);
            if (hit1 != NullObj || hit2 != NullObj || hit3 != NullObj || hit4 != NullObj) {
            GG->land->MakeHole( 5, PosX, PosY-4);
            GG->land->MakeHole( 4, PosX-6, PosY+4);
            GG->land->MakeHole( 4, PosX+6, PosY+4);
            GG->land->MakeHole( 4, PosX-6, PosY);
            GG->land->MakeHole( 4, PosX+6, PosY);
            GG->land->MakeHole( 4, PosX, PosY-1);
            }  
         WAP->Size = 1.6;
//         animColor = ARGB(255, 90, 180, 255);    // lol
         wormsOriginalMaskIndex = MaskIndex;
         MaskIndex = 9; // mine = 2
         ColMask = smallWormMask60;
         ShowMessage("Fat worm mode: ON");            
         imFatWorm = true;
         hideWorm = true;
         animatedWorm = true;
         GravityFactor = GravityFactor * 1.25;
         SpeedDivider = SpeedDivider * 0.6; 
}
