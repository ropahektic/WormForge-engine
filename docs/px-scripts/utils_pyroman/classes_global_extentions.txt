fixed FireX[50];
fixed FireY[50];
CFire *Fire[50];
bool FireCheck[50];
bool FireReading;

void classes_global_extentions::Init()
{
    FireReading = false;
}

override void CFire::AddSprite(fixed z,fixed x,fixed y,int unk,int sprite,fixed frame)
{
    super;
    if (!FireReading) return;
    local check;
    for (local i = 0; i < 50; i++)
    {
      if (!FireCheck[i]) return;
      if (Fire[i] == NullObj && check)
      {  
        Fire[i] = this;
        FireX[i] = x;
        FireY[i] = y;
        check = false;
        FireCheck[i] = false;
        break;
      } else {
        if (Fire[i] == this)
        {
           FireX[i] = x;
           FireY[i] = y;
           FireCheck[i] = false;
//           GG->WriteToChat( 1, ItoA(gframe), false);
           break;
        }
      }
    }
}

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    super;
    if (!FireReading) return;
    if (Type == M_FRAME)
    {
        if (gframe % 1) {}
        else {
           for (local i = 0; i < 50; i++)
           {
              FireCheck[i] = true;
           }
        }
    }
}
