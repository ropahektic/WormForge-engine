void CCamSync::ProcessInput() {
	//This function is running from current playing computer, which seems to be the deciding host.
	GG->queue2->AddPXMessage(msgType, 0, GS->ActCameraX);
	GG->queue2->AddPXMessage(msgType, 1, GS->ActCameraY);
	if (WApp->MK == 1){  // left pressed
		GG->queue2->AddPXMessage(msgType, 2, 0);
	}
	if (WApp->MK == 0){  // right pressed
		GG->queue2->AddPXMessage(msgType, 2, 1);
	}
	if (IsKeyPressed(87)) { // w pressed
		GG->queue2->AddPXMessage(msgType, 2, 2);
	}
	if (IsKeyPressed(65)) { // a pressed
		GG->queue2->AddPXMessage(msgType, 2, 3);
	}
	if (IsKeyPressed(83)) { // s pressed
		GG->queue2->AddPXMessage(msgType, 2, 4);
	}
	if (IsKeyPressed(68)) { // d pressed
		GG->queue2->AddPXMessage(msgType, 2, 5);
	}
	if (IsKeyPressed(70)) { // f pressed
		GG->queue2->AddPXMessage(msgType, 2, 6);
	}
}

void CCamSync::OnMessage(CMessageData* MData)
{
	if (MData->GetParameter(0) == msgType)
	{
		local MParam = MData->GetParameter(1);
		local Mval = MData->GetParameter(2);
		if (MParam == 0) {
			SyncedCX = Mval;
		}
		if (MParam == 1) {
			SyncedCY = Mval;
		}
		if (MParam == 2) {
			if (Mval == 0) {
				MKL = 1;
			}
			if (Mval == 1) {
				MKR = 1;
			}
			if (Mval == 2) {
				MKw = 1;
			}
			if (Mval == 3) {
				MKa = 1;
			}
			if (Mval == 4) {
				MKs = 1;
			}
			if (Mval == 5) {
				MKd = 1;
			}
			if (Mval == 6) {
				MKf = 1;
			}
		}
	}
}

void CCamSync::OnFrame() {
	MKL = 0;
	MKR = 0;
	MKw = 0;
	MKa = 0;
	MKs = 0;
	MKd = 0;
	MKf = 0;
}

override void CTurnGame::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{
	if (Type == M_INPUT)
	{
		camSync->ProcessInput();
	}
	if (Type == M_PXMESSAGE)
	{
		camSync->OnMessage(MData);
	}
	super;
	if (Type == M_FRAME)
	{
		camSync->OnFrame();
	}
};

//Calculate normalized vector for mouse aim from current worm
void CalcAimNormal(int apx, int apy, int wpx, int wpy) {
	float dx = float(apx - wpx);
	float dy = float(apy - wpy);
	float dl =  sqrt(dx * dx + dy * dy);
	if (dl > 0.00001) {
		aimNVX = dx / dl;
		aimNVY = dy / dl;
	}
	aimNVL = dl;
	aimPX = wpx + (aimNVX * 70);
	aimPY = wpy + (aimNVY * 70);
}
