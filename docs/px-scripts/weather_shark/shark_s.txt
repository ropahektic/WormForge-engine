        if(Type == M_FRAME)
	{
            super;
	}
   if (Type == M_DRAWQUEUE)
   {
    local worm = GetCurrentWorm();
    if (worm is CWorm)
    {
      if (worm->PosY >= (GS->LevelSY - 650) && worm->PosY <= GS->LevelSY - 45)
      {
        if (SharkPlayNow)
        {
//           sharkSound->Play(1.0, 1.0, false);
//           ShowMessage("Ouch! Careful");
           LastWormPosX = worm->PosX;
           sharkWayCounter = 0;
           SharkPlayNow = false;
        }
        local spr1 = sharkBodyR->Index;
        local spr2 = sharkBodyL->Index;
        local spr;
        local animationPhase;
        if (sharkWayCounter2 >= 349) {sharkWayCounter2 = 0;} else {animationPhase = float(sharkWayCounter2)/350;}
//        local sharkRot = 0;
//        local ang = (sharkRot / 360 + MATH_HALF_PI) * 6.3;
        if (sharkWayCounter < 500) {LastWormPosX += 0.1; sharkWayCounter += 1; sharkWayCounter2 += 1; spr = sharkBodyR->Index;}
        else if (sharkWayCounter < 1500) {LastWormPosX -= 0.1; sharkWayCounter += 1; sharkWayCounter2 += 1; spr = sharkBodyL->Index;}
        else if (sharkWayCounter < 2000) {LastWormPosX += 0.1; sharkWayCounter += 1; sharkWayCounter2 += 1; spr = sharkBodyR->Index;}
        else {SharkPlayNow = true; spr = sharkBodyL->Index;}
        AddSpriteEx(5, LastWormPosX, GS->LevelSY + 65, spr, animationPhase, 0.0, 1);
  }
  local tempWormSp = worm->SpY;
  if (tempWormSp < 1.0) tempWormSp = 1.0;
  local tempAnimSpeed = worm->SpY;
  if (tempAnimSpeed < 6) tempAnimSpeed = 6;
  if (worm->PosY >= GS->LevelSY - tempWormSp*49.95 && worm->PosY <= GS->LevelSY + 40)
  {
         if (worm->ObjState != WS_IDLE && worm->ObjState != WS_AIMING && worm->ObjState != WS_WALKING && worm->ObjState != WS_SHOOTINGROPE && worm->ObjState != WS_JUMP && worm->ObjState != WS_JUMPPREPARE)
         {
            if (sharkI <= 100)
            {
               local sideCoef;
               local sideCoef2;
               if (worm->SpX >= 1) {spr3 = sharkHeadL->Index; sideCoef = 0.7; sideCoef2 = 300.0;} else {spr3 = sharkHeadR->Index; sideCoef = -0.7; sideCoef2 = -300.0;}
               tempAnimSpeed2 = sharkI*tempAnimSpeed/16;
               AddSpriteEx(5, worm->PosX, GS->LevelSY - tempAnimSpeed2, spr3, sharkI/100.0, sharkI/sideCoef2 - sideCoef, 0.85);
            }
            else if (sharkI <= 200)
            {
/*               if (sharkBloodNow5 == false)

            }
            if (sharkI <= 750) {sharkI += 1;} else {sharkI = 0;}
         }
  } else sharkI = 0;

  if (worm->PosY >= GS->LevelSY + 10)
  {
     local ang;
     ang = float(worm->PosY - GS->LevelSY) / 150.0;

  }
// -----------------------------------------//

    {

        {
          local checkNow;
