 }

//---------------------------------MINI-GIRDER--------------------------------//

override void CMissile::ExplodeAt(fixed x,fixed y)
{

 if(weap->CheckName("Girder Launcher") == true)
 {
           CGObject *mgirderob;
     	   for(int i = 0; i < Env->Objs.Count; i += 1)
	   {
		mgirderob = CGObject(Env->Objs.Objs[i]);
		if(mgirderob == this) continue;
		if(mgirderob == NullObj) continue;
		if(mgirderob->IsMaterial == false) continue;
			if(mgirderob->ClType == OC_Worm || mgirderob->ClType == OC_Mine || mgirderob->ClType == OC_OilDrum || mgirderob->ClType == OC_Crate)
			{
				local trg = CGObject(mgirderob);
                                if(x > trg->PosX-20 && x < trg->PosX+20 && y > trg->PosY-20 && y < trg->PosY+20)
                                {
                                GirderDamage(trg);
                                super;
                                return;
                                }
			}
           }

           if(GetCurrentWorm() != NullObj)
           {
                   if(x > GetCurrentWorm()->PosX-25 && x < GetCurrentWorm()->PosX+25 && y > GetCurrentWorm()->PosY-25 && y < GetCurrentWorm()->PosY+25)
                   {
                   GirderDamage(GetCurrentWorm());
                   super;
                   return;
                   }
           }

           GirderStickToLandscape(x,y);
           mgirderimpactsnd->Play(1.0,0.0,false);
 }

 super;
}

void CMissile::GirderDamage(CGObject* target)
{
                     local damage = 1;
                     if(weap->CheckName("Girder Launcher") == true)
                     {damage = 3;}

                     target->SpX = SpX/4;
                     target->SpY = SpY/4;

                     CMessageData msgdamage;
                     msgdamage.params[0] = 3;
                     msgdamage.params[1] = PosX;
                     msgdamage.params[2] = PosY;
                     msgdamage.params[3] = SpX/2;
                     msgdamage.params[4] = SpY/2;
                     msgdamage.params[5] = damage;
                     msgdamage.params[6] = 0;
                     target->Message(NullObj, M_GUNEXP, 1032, &msgdamage);
}

void CMissile::GirderStickToLandscape(fixed x, fixed y)
{
local bmp = MgirderBitmap;
local top = bmp->top;
int mtp = 0;

      if(SpX == 0.0 && SpY < 0.0){mtp = 0;}
      if(SpX > 0.0 && SpY == 0.0){mtp = 8;}
      if(SpX == 0.0 && SpY > 0.0){mtp = 16;}
      if(SpX < 0.0 && SpY == 0.0){mtp = 24;}

      if(SpX > 0.0 && SpY < 0.0)
      {
      mtp = 3*SpX/sqrt(SpY*SpY);
      if(mtp>7){mtp = 7;}
      if(mtp<0){mtp = 0;}
      }

      if(SpX > 0.0 && SpY > 0.0)
      {
      mtp = 8+3*SpY/SpX;
      if(mtp>15){mtp = 15;}
      }

      if(SpX < 0.0 && SpY > 0.0)
      {
      mtp = 16+3*sqrt(SpX*SpX)/SpY;
      if(mtp>23){mtp = 23;}
      }

      if(SpX < 0.0 && SpY < 0.0)
      {
      mtp = 24+3*sqrt(SpY*SpY)/sqrt(SpX*SpX);
      if(mtp>31){mtp = 31;}
      }

top = mtp*60;
bmp->top = top;

bmp->bottom = bmp->top+60;
GG->land->Draw(bmp, x-30, y-(30+bmp->top));
}
