CSprite* surfer_aiming;
CSprite* surfer_bullet;
                                                                                                                               
void surfer_animation::InitGraphic() {
    CFile* f;                                                                    
    f = GetAttachment("surfer_aiming.png");                             
    surfer_aiming = LoadSprite(f, 1, 0);
    f = GetAttachment("surfer_bullet.png");                             
    surfer_bullet = LoadSprite(f, 10, 0);                                                                 
}                                                                                
                                                                                 
override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) { 
 super;                                                                          
  if (Type == M_FRAME) {                                                         
    if (gframe == 1) {                                                           
     local weapName; local sprite;

     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Surfer";                                               
     sprite = surfer_aiming->Index;                                     
     Params->hand_radius = 8.0;    
     Params->hand_radial_rotation = -0.95;                                                     
     Params->weap_rotation =  0.5;
     Params->weap_radius = 8.0;
     Params->hand_type = 1; 
     Params->scale = 0.8;                                 
                                                  
     local WA = new WormSprite(weapName, sprite, Params);
    }                                                                           
  }                                                                              
}                                                                                

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){
super;                                                                          
  if (Type == M_FRAME){                                                          
     if (weap->CheckName("Surfer")) {                                
         if (surfer_anim) {                                              
            local sprite = surfer_bullet->Index;                      
            WeapSpriteParams* Params = new WeapSpriteParams;
            Params->animate = true;
            Params->anim_speed = 1.5;
            Params->dont_rotate_by_speed = true;
            Params->play_backwards = true;
//            Params->hideWorm = true;
            local wa = new WeapSprite(this, sprite, Params); 
//            wa->type = 5;             
            weap->launch.anim.spriteIndex = 0;
            surfer_anim = false;                                         
         } 
     }                                                                                                                                                                                                                          
  }                                                                              
}                                                                                
override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;                                                                           
  surfer_anim = true;                                                    
}
