require utils, sprite_curtis_458;

CColMask* jolt_COLMASK;
CSprite* joltSprite;

void staff_script::InitGraphic()
{
        CFile *f;
	f = GetAttachment("jolt.png");	joltSprite = LoadSprite(f, 20, 0);
}

void staff_script::Init(){}

void staff_script::FirstFrame()
{
     jolt_COLMASK = new CColMask(10,10,MakeRectMask(20,20));
}


override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata)
{
super;
SType = false;  //Jolt flag
}


override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
super;
if(Weap->CheckName("Staff of Jolting"))
 	{
    CGObject *jolto;
    CMissile *joltm;
	jolto = CGObject(Env->Objs.Objs[Env->Objs.Count-1]);

	if (jolto != NullObj)
	if(jolto->ClType == OC_Missile)
        {
        joltm = jolto;
        joltm->SType = true;
        }
 	}
}

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
   if(SType == true)
   {
              if(Type == M_FRAME)
              {
                      JoltCollision(15.0);
              }
   }
super;
}

void CMissile::JoltCollision(float Speed)
{
	CGObject *joltob;

	for(int i = 0; i < Env->Objs.Count; i += 1)
	{
		joltob = CGObject(Env->Objs.Objs[i]);
		if(joltob == this) continue;
		if(joltob == NullObj) continue;
		if(GetCurrentWorm() != NullObj)
		                    {
		                    if(joltob == GetCurrentWorm()) continue;
		                    }


		if(joltob->ClType == OC_Worm)
		{
			if(jolt_COLMASK->Check(PosX, PosY, joltob->ColMask, joltob->PosX, joltob->PosY))
			{
				float ForceX = joltob->PosX - PosX;
				float ForceY = joltob->PosY - PosY;
				float dist = sqrt(ForceX*ForceX + ForceY*ForceY);

				if(dist > 0)
				{
					ForceX = 0.6*sqrt(Speed) * ForceX / dist;
					ForceY = 0.6*sqrt(Speed) * ForceY / dist;

					CMessageData msg;
					msg.params[0]	= 0;
					msg.fparams[1]	= PosX;
					msg.fparams[2]	= PosY;
					msg.fparams[3]	= ForceX;
					msg.fparams[4]	= ForceY;
					msg.params[5]	= 1;
					msg.params[6]	= 0;

					joltob->Message(this, M_GUNEXP, 1032, &msg);
				}
			}
		}
	}
}
