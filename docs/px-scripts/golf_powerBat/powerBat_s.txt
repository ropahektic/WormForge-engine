 override void CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
 {
     super;
     golfClub = 1;
     clubAngleMin = 0.0;
     clubAngleMax = 1.0;
     golfPower = 1.0;
     golfPowerChanger = 1.0;
     golfShotVolume = 1.0;
     powerStartDecrease = false;
     ReadyToFinishTurn = false;
 }
 
 override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
 {
     if (Weap->CheckName("Power Bat"))
     {
         if (ObjState == WS_JETPACK) 
         {
            PlaySound(120, 0, 0, 1.0);
            return;
         }
         local x = Desc->X;
         local y = Desc->Y;
         local spx = Desc->SpX;
         local spy = Desc->SpY;
         ThrowObject(x, y, spx, spy);
         return;
     }
     super;
 }
 
 void CWorm::ThrowObject(fixed x, fixed y, fixed spx, fixed spy)
 {  
     local delta = 1.2;
     local counter = 0;
     golfShotVolume = (spx+spy)/8;
     if (golfShotVolume < 0.6) golfShotVolume = 0.6;
     else if (golfShotVolume > 4.0) golfShotVolume = 4.0;
     for (local i = 1; i < Env->Objs.Count; i++)
     {
         local hit = CGObject(Env->Objs.Objs[i]);
         if (hit == NullObj) continue;
         if (hit == this) continue;
         if (#WEAPON_GOLFBALL)
         {
            if (hit is CGolfBall)
            {
              if (hit->PosX > x-25 && hit->PosX < x+25 && hit->PosY > y-25 && hit->PosY < y+25)
              {
                  local golf = CGolfBall(hit);
                  if (golf->ballOwner == this)
                  {
                      golf->Wake();
                      golf->ballOwner->golf_teleported = false;
                      if (golfClub < 5) hit->PosY -= 1;
                      hit->SpX += spx * golfPower * golfPowerChanger * delta;
                      hit->SpY += spy * golfPower * 1.35 * golfPowerChanger;
                      if (golfClub == 1) golf->spin = 0.3;
                      else if (golfClub == 2) golf->spin = 0.5;
                      else if (golfClub == 3) golf->spin = 0.65;
                      else if (golfClub == 4) golf->spin = 0.6;
                      else if (golfClub == 5) golf->spin = 0.0;
//                      PlaySound( 97, 0, 0, 1.0);
                      if (golf->zoneType == 3 || golfClub == 4) ClubBallSand->Play( golfShotVolume*2, 1.0, false);
                      else if (golfClub == 1) ClubBallHitDriver->Play( golfShotVolume*1.3, 1.0, false);
                      else if (golfClub == 2) ClubBallHitIron->Play( golfShotVolume*2, 1.0, false);
                      else if (golfClub == 3) ClubBallHitPW->Play( golfShotVolume*2, 1.0, false);
                      else if (golfClub == 5) ClubBallHitPutter->Play( golfShotVolume*1.5, 1.0, false);
//                      ClubBallHit->Play(golfShotVolume/2.5, 1.0, false);
                      counter++;
//                      FinishTurnEx();
                      ReadyToFinishTurn = true;  
                      if (golf->ballOwner != NullObj) 
                      {
                         golf->ballOwner->holePoints++;
                         golf->ballOwner->totalPoints++;
                      }
                  }
              }
            }
         }
     }
     if (counter == 0)
     {
         ClubBallMiss->Play( golfShotVolume*1.5, 1.0, false);
     }
 }
 
 override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
 {
     super;
     if (Type == M_DRAWQUEUE)
     {
         if (hideWorm)
         {
//             AddSprite( 4, PosX, PosY, 0, golfPowerUp->Index, FirePower);
         }
     }
     
     if (Type == M_FRAME)
     {
        if (CurWeapon->CheckName("Power Bat"))
        {
             if (keySync->KeyPressed(32))
             {
                if (FirePower >= 0.97) 
                {
                   powerStartDecrease = true;
                   FirePower = 0.97;
                }
//                else if (FirePower <= 0.02)
                else if (FirePower <= 0.02)
                {
                   powerStartDecrease = false;
                   SetState(WS_AIMING);
                }
                else if (powerStartDecrease == false)
                {
                    local powa = -0.01 + 0.03 * FirePower;
                    if (powa > 0.001) FirePower += powa;
                }
                if (powerStartDecrease)
                {
                   local powa = -0.015 + 0.03 * FirePower;
                   if (powa < 0.026) powa = 0.026;
                   FirePower -= powa;
                }
//                hideWorm = true;
             }
             else 
             {
                powerStartDecrease = false;
//                hideWorm = false;  
             }
             local Team = CTeam(GetObject(21, WormTeam));
	     Team->AllowSelectBounce = false;
             if (keySync->KeyPressedNow(49) || keySync->KeyPressedNow(50) || keySync->KeyPressedNow(51) || keySync->KeyPressedNow(52) || keySync->KeyPressedNow(53))// && ObjState == WS_AIMING)
             {
                 if (keySync->KeyPressedNow(49))
                 {
                     golfClub = 1;
                     ShowMessage("club: Driver");
                 }
                 if (keySync->KeyPressedNow(50))
                 {
                     golfClub = 2;
                     ShowMessage("club: Iron 5");
                 }
                 if (keySync->KeyPressedNow(51))
                 {
                     golfClub = 3;
                     ShowMessage("club: Pitching Wedge");
                 }
                 if (keySync->KeyPressedNow(52))
                 {
                     golfClub = 4;
                     ShowMessage("club: Sand Wedge");
                 }
                 if (keySync->KeyPressedNow(53))
                 {
                     golfClub = 5;
                     ShowMessage("club: Putter");
                 }
             }
             
             if (golfClub == 1) {clubAngleMin = 0.65; clubAngleMax = 0.67; golfPower = 1.0;}       // Driver  // 0.615
             if (golfClub == 2) {clubAngleMin = 0.67; clubAngleMax = 0.69; golfPower = 0.72;}    // Iron 5    // 0.75
             if (golfClub == 3) {clubAngleMin = 0.72; clubAngleMax = 0.81; golfPower = 0.47;}    // Pitching Wedge    // 0.85
             if (golfClub == 4) {clubAngleMin = 0.7; clubAngleMax = 0.85; golfPower = 0.33;}      // Sand Wedge // 0.8
             if (golfClub == 5) {clubAngleMin = 0.5; clubAngleMax = 0.5; golfPower = 0.2;}     // Putter  // 0.516
                 
                 if (FireAngle < clubAngleMin) FireAngle = clubAngleMin;
                 if (FireAngle > clubAngleMax) FireAngle = clubAngleMax;
//                 FireAngle = clubAngle;
        }
     }
     if (Type == M_TURNEND) powerStartDecrease = false;
 }
