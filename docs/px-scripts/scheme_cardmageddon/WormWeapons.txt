override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if(Type == M_FRAME && !isPlacing && teleportrange != 0)
  {
    if(CurWeapon->CheckName("Teleport") || CurWeapon->CheckName("Extra Teleport") ||
	   CurWeapon->CheckName("Bazooka Worm") || CurWeapon->CheckName("Shotgun Worm") ||
	   CurWeapon->CheckName("Haste Worm") || CurWeapon->CheckName("Tanky Worm") ||
	   CurWeapon->CheckName("Rope Worm") || CurWeapon->CheckName("Melee Worm") ||
	   CurWeapon->CheckName("Fat Worm") || CurWeapon->CheckName("Mine Worm") ||
	   CurWeapon->CheckName("Sniper Worm") || CurWeapon->CheckName("Minigun Worm")||
	   CurWeapon->CheckName("Prince")|| CurWeapon->CheckName("Infected Worm") ||
	   CurWeapon->CheckName("Necromancer")|| CurWeapon->CheckName("Flying Zombie") ||
	   CurWeapon->CheckName("Spawn 3 Zombies") || CurWeapon->CheckName("Spawn Zombie"))
    {
      local dx = PosX - GS->landSelectX;  local dy = PosY - GS->landSelectY;                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                
      local dist = sqrt(dx*dx + dy*dy);
   
      if(dist > teleportrange)
      {
        local angle = atan2(dx, dy);
        GS->landSelectX = PosX-teleportrange*sin(angle);  GS->landSelectY = PosY-teleportrange*cos(angle);
      }
    }
  }
  super;
}   
       
override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_FRAME)
    {
        if (GetFreeWormIndex(Root->CurTeam) > 8 || GetFreeWormIndex(Root->CurTeam) < 1)
        {
            lockworms2 = true;
        }    
        
        
    }
    
    if (Type == M_DISABLECONTROL)
    {
        lockworms2 = false;
    }
    
    super;
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
         if(Type == M_TURNBEGIN)
         {
                
                  
         }
         
         if(Type == M_TURNEND)
         {
                 RemoveWormWeapons();
                 RemoveZombieWeapons();
                
                 
         }
         
         if(Type == M_FRAME)
         {
           if(GS->Info.GetWormHealth(WormTeam, WormNumber) == 0 && dead == false)
                 {
                 dead = true;
                                   
                        if(CurW == this && CurW != NullObj && CurW->dead == true)
                        {
                           RemoveWormWeapons(); 
                           RemoveZombieWeapons(); 
                        }
                 } 
         }
super;
}

   }
super;
}
