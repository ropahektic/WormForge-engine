override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if(GetCurrentWorm() != NullObj)
  {
       if(PlayerBought[WormTeam-1] == false && GetCurrentWorm() == this && getweapons == true)
       {
                   if(Type == M_FRAME)
                   {
                    if(cooldown > 0){cooldown = cooldown - 1;}
                    if(PlayerBought[WormTeam-1] == false){Root->Timer = 10000;}
                   }
       
                   if(Type == M_PLEFT)
                   {
                           if(Selection != 0 && cooldown == 0)
                           {
                           cooldown = 15;
                           Selection = Selection - 1;
                           }
                   return;        
                   }
                   
                   if(Type == M_PRIGHT)
                   {
                           if(Selection != 39 && cooldown == 0)
                           {
                           cooldown = 15;
                           Selection = Selection + 1;
                           }
                   return;        
                   }
                   
                   if(Type == M_PUP)
                   {
                           if(Selection > 4 && cooldown == 0)
                           {
                           cooldown = 15;
                           Selection = Selection - 5;
                           }
                   return;        
                   }
                   
                   if(Type == M_PDOWN)
                   {
                           if(Selection < 35 && cooldown == 0)
                           {
                           cooldown = 15;
                           Selection = Selection + 5;
                           }
                   return;        
                   }
                   
                   if(Type == M_ENTER)
                   {
                           if(Cash[WormTeam-1] > 0 && WeaponA[Selection] < 3)
                           {
                           WeaponA[Selection] = WeaponA[Selection] + 1;
                           local wep1 = GS->GetWeaponByName(WeaponN[Selection]);
                           if(wep1 != NullObj){
                           if(GetTeamColor(WormTeam) == 0) { RedWeapons[WeapCounter] = WeaponN[Selection]; }
                           if(GetTeamColor(WormTeam) == 1) { BlueWeapons[WeapCounter] = WeaponN[Selection]; }
                           if(GetTeamColor(WormTeam) == 2) { GreenWeapons[WeapCounter] = WeaponN[Selection]; }
                           if(GetTeamColor(WormTeam) == 3) { YellowWeapons[WeapCounter] = WeaponN[Selection]; }
                           if(GetTeamColor(WormTeam) == 4) { PurpleWeapons[WeapCounter] = WeaponN[Selection]; }
                           if(GetTeamColor(WormTeam) == 5) { CyanWeapons[WeapCounter] = WeaponN[Selection]; }
                           WeapCounter++;}
                           Cash[WormTeam-1] = Cash[WormTeam-1] - 1;
                           }
                   return; 
                   }
                   
                   if(Type == M_SPACE)
                   {
                  
                   PlayerBought[WormTeam-1] = true;
                   
                   if(Cash[WormTeam-1]> 0){
                   for(local y = WeapCounter; y < Credits; y++)
                     {
                           if(GetTeamColor(WormTeam) == 0) { RedWeapons[y] = WeaponN[RandomInt(0, 39)]; }
                           if(GetTeamColor(WormTeam) == 1) { BlueWeapons[y] = WeaponN[RandomInt(0, 39)]; }
                           if(GetTeamColor(WormTeam) == 2) { GreenWeapons[y] = WeaponN[RandomInt(0, 39)]; }
                           if(GetTeamColor(WormTeam) == 3) { YellowWeapons[y] = WeaponN[RandomInt(0, 39)]; }
                           if(GetTeamColor(WormTeam) == 4) { PurpleWeapons[y] = WeaponN[RandomInt(0, 39)]; }
                           if(GetTeamColor(WormTeam) == 5) { CyanWeapons[y] = WeaponN[RandomInt(0, 39)]; }
                   }
                   }
                   for(local t = 0;t<99;t++)
                             {
                             WeaponA[t] = 0;
                             }
                   Root->RetreatTimer = 1;
	           WeapCounter = 0;
                   getweapons = false;
                  
                   local TeamNumber = GetTeamNumber(WormTeam);
                   GS->Info.SetWeaponDelay(TeamNumber, 62, 1, -1); 
                   
                    GiveWeapons();
                    GiveWeapons();
                    GiveWeapons();
                    GiveWeapons();
                    GiveWeapons(); 
    
   
    
                 teamN = 800*WormTeam - 800;
                 wormN = 100*1 - 100;
                 multipl = teamN+wormN; 
                                             
                 HasWeapons[multipl] = "BlowTorch";
                 HasWeapons[multipl+1] = "Girder";
                 HasWeapons[multipl+2] = "Draw 1 Weapon";
                 HasWeapons[multipl+3] = "Teleport";
                 GiveWormWeapons();  
   
    TeleportingCheck[WormTeam] = 1;
    PreturnCheck[WormTeam] = 0;
                   return;        
                   }
                   
                   if(Type == M_DRAWQUEUE)
                   {
                           if(PlayerBought[WormTeam-1] == false)
                           {
                           DrawBuyMenu();
                           }
                   }
       }
  }
super;
}

void CWorm::DrawBuyMenu()
{
 if(DoIKnowHim(this) == true)
 {
  local spr1 = wepanel->Index;
  AddSprite(5,GS->CameraX,GS->CameraY,0,spr1, 0.0);
  
  local x = 0;
  local y = 0;
  local t = 0;
  x = GS->CameraX-90;
  y = GS->CameraY-190;
  if(Selection != 0){t = Selection/5;}else{t = 0;}
  y = y + 16 + 29 * t;
  x = x + 48 + 29 * (Selection - t*5);

  local spr2 = sel->Index;
  AddSprite(4,x,y,0,spr2, 0.0);
 }
