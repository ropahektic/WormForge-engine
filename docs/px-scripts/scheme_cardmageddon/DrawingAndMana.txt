override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
  super;
  
  
   if(Weap->CheckName("Select Worm"))
  {
   
   SelectCheck[WormTeam] = 1;
   }
   
  if(Weap->CheckName("Get Weapons"))
  {  
    RemoveWeapons();
    
     getweapons = true;     
  }
  
 	
	if (Weap->CheckName("Draw 3 Weapons"))
	{    GiveWeapons();
             GiveWeapons();
             GiveWeapons();
        }

        if (Weap->CheckName("Get 2 Mana"))
	{    GS->Info.Stores[WormTeam-1].Count[62+2] = GS->Info.Stores[WormTeam-1].Count[62+2] + 2;
        }
        
        if(Weap->CheckName("MedPack")) GS->Info.SetWormHealth( Root->CurTeam, Root->CurWorm, GS->Info.GetWormHealth( Root->CurTeam, Root->CurWorm)+25);
 
        if (Weap->CheckName("Draw 1 Weapon"))
	{    GiveWeapons();
             GS->Info.SetWormHealth( Root->CurTeam, Root->CurWorm, GS->Info.GetWormHealth( Root->CurTeam, Root->CurWorm)-10);
        }

        if (Weap->CheckName("Get 1 Mana"))
	{    GS->Info.Stores[WormTeam-1].Count[62+2] = GS->Info.Stores[WormTeam-1].Count[62+2] + 1;
        }
        
        if (Weap->CheckName("Extra Teleport"))
	{    GS->Info.Stores[WormTeam-1].Count[62+2] = GS->Info.Stores[WormTeam-1].Count[62+2] - 1;
        }
 
 
        if (Weap->CheckName("Zombie Damage*2"))
	{    Zombie2X = true;   	     
        }
        
        if (Weap->CheckName("+25hp For All"))
        {
          CGObject *eobj; local TeamColor = GetTeamColor(WormTeam);
            for(local i = 0; i < Env->Objs.Count; i++)
              {
                 eobj = CGObject(Env->Objs.Objs[i]);
      
         if(eobj == NullObj) continue;
           if(eobj->IsMaterial == false) continue;                         			
              if(eobj->ClType == OC_Worm)
                   {
                        local worm = CWorm(eobj);
              if(GetTeamColor(worm->WormTeam) == TeamColor)
                    {
                           
                            GS->Info.SetWormHealth( worm->WormTeam, worm->WormNumber, GS->Info.GetWormHealth( worm->WormTeam, worm->WormNumber)+25);
                                   } 
             }
                 }   
    
        }
    
    
   
   RemoveDelays();
   PersonalDelayCheck();
   DelayCheck();
   if(Zombie == true) { AddDelays(); }   
}

	         ShotgunCheck[WormTeam] = 0;

		 if (TeleportingCheck[WormTeam] == 1) {

                GiveWeapons(); 
              
              
                 
              
            }   
             SelectCheck[WormTeam] = 0;  
             
             
            }    
	
            if(CurW != NullObj){CurW->RemoveWormWeapons();}
                 CurW = this;
                 
                  if(Zombie == true)
                  {                   
                   if(Zombie2X == false) { GiveZombieWeapons(); RemoveDelays(); AddDelays(); PersonalDelayCheck();}
                   if(Zombie2X == true) { GiveZombieSuperWeapons(); RemoveDelays(); AddDelays(); PersonalDelayCheck();}

	}

         {      
            for(local x = 0; x < 71; x++)
            {  
                if(CurWeapon->CheckName(WeaponN[x]))                         
                {   
                                 
                    if(CurWeapon->CheckName("Extra Teleport") || CurWeapon->CheckName("Bazooka Worm") || CurWeapon->CheckName("Shotgun Worm") || CurWeapon->CheckName("Haste Worm") || CurWeapon->CheckName("Tanky Worm") || CurWeapon->CheckName("Rope Worm") || CurWeapon->CheckName("Melee Worm") || CurWeapon->CheckName("Fat Worm") || CurWeapon->CheckName("Mine Worm") || CurWeapon->CheckName("Sniper Worm") || CurWeapon->CheckName("Minigun Worm") || CurWeapon->CheckName("Prince") || CurWeapon->CheckName("Infected Worm") || CurWeapon->CheckName("Necromancer")|| CurWeapon->CheckName("Flying Zombie") || CurWeapon->CheckName("Spawn 3 Zombies"))

                        super;
                        return;
                       }
                   else if(CurWeapon->CheckName("Longbow")) 
                   {
                    super;

                    {  GS->Info.Stores[WormTeam-1].Count[62+2] = GS->Info.Stores[WormTeam-1].Count[62+2] - WeaponC[x]; 
                        ShotgunCheck[WormTeam]++; }
                       else  { ShotgunCheck[WormTeam] = 0; }
                    return; 
                   }
                   
                   else
                       super;
                       GS->Info.Stores[WormTeam-1].Count[62+2] = GS->Info.Stores[WormTeam-1].Count[62+2] - WeaponC[x];
                       RemoveDelays();
                       PersonalDelayCheck(); 
                       DelayCheck();  
 
                    return;                                                     
                } 
                
            }    
         } 
       super;

	
}

void CWorm::RemoveWeapons()
{
 local TeamNumber = GetTeamNumber(WormTeam);
  for(local i=1; i<68; i++)
  {
    local weap = GS->GetWeaponByIndex(1, i);
    if(weap == NullObj) continue;
    if(weap->CheckName("Skip Go") || weap->CheckName("Surrender") || weap->CheckName("Teleport")) continue;

    GS->Info.SetWeaponEx(TeamNumber, i, weap->GetPageIndex(), 0);
  }
  
	
}



void CWorm::DelayCheck()
{
 local TeamNumber = GetTeamNumber(WormTeam);
  for(local i=1; i<68; i++)
  {
    local weap = GS->GetWeaponByIndex(1, i);
    if(weap == NullObj) continue;
    if(weap->CheckName("Skip Go") || weap->CheckName("Surrender") || weap->CheckName("Freeze")) continue;
   
      for(local j=0; j<71; j++)
  {   
        
      if(weap->CheckName(WeaponN[j]))
      {
       if (GS->Info.Stores[WormTeam-1].Count[62+2] < WeaponC[j])
        {          
        if (GS->Info.Stores[WormTeam-1].Count[weap->GetWeaponIndex()+2] !=0) 
        {
        GS->Info.SetWeaponDelay(TeamNumber, i, weap->GetPageIndex(), -1);
        }
      }
                            
      }  
  }  
 }
	
}

{
    local TeamNumber = GetTeamNumber(WormTeam);

      for(local j=0; j<71; j++)

  }  
 }
