require p_sprite_builder;

CSprite* SQBA;

void SQBS::InitGraphic()               
{   CFile* f;                          
    f = GetAttachment("beeclust.png");
    SQBA = LoadSprite(f, 0, 0);       
}

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) { 
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {  
     local weapName; local sprite;
     
     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Super Queen Bee";
     sprite = SQBA->Index;        
     Params->hand_radius = 25.5;                                                 
     Params->hand_rotation = 1.0; 
     Params->hand_radial_rotation = 1.05;
     Params->weap_radial_rotation = 9.0;     
     Params->weap_rotation = 0.1;
     Params->weap_radius = 16.0;
     Params->hand_type = 2;     
     Params->hand_scale = 0.8;  
//     Params->zplane = 10.0;   
//     Params->scale = 1.0;     

     local WA = new WormSprite(weapName, sprite, Params);
    }
  }  
}
