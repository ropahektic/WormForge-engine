//---------------------------------CROSSBOW(s)--------------------------------//

override void CMissile::ExplodeAt(fixed x,fixed y)
{

 if(weap->CheckName("Crossbow") == true)
 {
           CGObject *boltoba;
     	   for(int i = 0; i < Env->Objs.Count; i += 1)
	   {
		boltoba = CGObject(Env->Objs.Objs[i]);
		if(boltoba == this) continue;
		if(boltoba == NullObj) continue;
		if(boltoba->IsMaterial == false) continue;
			if(boltoba->ClType == OC_Worm || boltoba->ClType == OC_Mine || boltoba->ClType == OC_OilDrum || boltoba->ClType == OC_Crate)
			{
				local trg = CGObject(boltoba);
                                if(x > trg->PosX-20 && x < trg->PosX+20 && y > trg->PosY-20 && y < trg->PosY+20)
                                {
                                BoltaDamage(trg);
                                super;
                                return;
                                }
			}
           }

           if(GetCurrentWorm() != NullObj)
           {
                   if(x > GetCurrentWorm()->PosX-25 && x < GetCurrentWorm()->PosX+25 && y > GetCurrentWorm()->PosY-25 && y < GetCurrentWorm()->PosY+25)
                   {
                   BoltaDamage(GetCurrentWorm());
                   super;
                   return;
                   }
           }

           BoltaStickToLandscape(x,y);
           boltimpactsnd->Play(1.0,0.0,false);
 }

 super;
}

void CMissile::BoltaDamage(CGObject* target)
{
                     local damage = 1;
                     if(weap->CheckName("Crossbow") == true)
                     {damage = 30;}

                     target->SpX = SpX/4;
                     target->SpY = SpY/4;

                     CMessageData msgdamage;
                     msgdamage.params[0] = 3;
                     msgdamage.params[1] = PosX;
                     msgdamage.params[2] = PosY;
                     msgdamage.params[3] = SpX/2;
                     msgdamage.params[4] = SpY/2;
                     msgdamage.params[5] = damage;
                     msgdamage.params[6] = 0;
                     target->Message(NullObj, M_GUNEXP, 1032, &msgdamage);
}

void CMissile::BoltaStickToLandscape(fixed x, fixed y)
{
local bmp = BoltBitmap;
local top = bmp->top;
int mtp = 0;

      if(SpX == 0.0 && SpY < 0.0){mtp = 0;}
      if(SpX > 0.0 && SpY == 0.0){mtp = 8;}
      if(SpX == 0.0 && SpY > 0.0){mtp = 16;}
      if(SpX < 0.0 && SpY == 0.0){mtp = 24;}

      if(SpX > 0.0 && SpY < 0.0)
      {
      mtp = 3*SpX/sqrt(SpY*SpY);
      if(mtp>7){mtp = 7;}
      if(mtp<0){mtp = 0;}
      }

      if(SpX > 0.0 && SpY > 0.0)
      {
      mtp = 8+3*SpY/SpX;
      if(mtp>15){mtp = 15;}
      }

      if(SpX < 0.0 && SpY > 0.0)
      {
      mtp = 16+3*sqrt(SpX*SpX)/SpY;
      if(mtp>23){mtp = 23;}
      }

      if(SpX < 0.0 && SpY < 0.0)
      {
      mtp = 24+3*sqrt(SpY*SpY)/sqrt(SpX*SpX);
      if(mtp>31){mtp = 31;}
      }

top = mtp*60;
bmp->top = top;

bmp->bottom = bmp->top+60;
GG->land->Draw(bmp, x-30, y-(30+bmp->top));
}
