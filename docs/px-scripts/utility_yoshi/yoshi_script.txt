void yoshi_script::Init()
{
    cooldown = 0;

	keySync->AddKey(38); // up arrow key
	
	YoshiJump1 = new CSoundFile(GetAttachment("jump1.wav"));
    YoshiJump2 = new CSoundFile(GetAttachment("jump3.wav"));
    YoshiJump3 = new CSoundFile(GetAttachment("jump6.wav"));
    YoshiJump4 = new CSoundFile(GetAttachment("throw.wav"));
    
    YoshiFly1 = new CSoundFile(GetAttachment("flutter.wav"));
    YoshiFly2 = new CSoundFile(GetAttachment("yoshi2.wav"));
    YoshiFly3 = new CSoundFile(GetAttachment("yoshi5.wav"));
}

override CWorm::CWorm(CObject* Parent,int aTeam,int aIndex,CWormParams* params)
{
YoFlyingMode = false;
YoFlyAnimationPhase = 0.0;
YoFlyAnimSpeed = 0.01;
super;
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
	if(Type == M_FRAME)
    	{
		if(ObjState != WS_PARACHUTE && YoFlyingMode)
	    	{
			GravityFactor = 1;
            YoFlyingMode = false;
            }

		if (YoFlyingMode)
        	{
			if (cooldown > 0){cooldown = cooldown - 1;}
			if (cooldown > 30)
				    {
				    SpY = SpY - 0.5; SpX = SpX + (0.5*TurnSide);
					}
			}
        }

	if (ObjState == WS_PARACHUTE)
    	{
        if (YoFlyingMode)
        	{
        	if(keySync->KeyPressedNow(38) && cooldown == 0)
				{
				local i = RandomInt(1, 6);
               	if      (i == 1) YoshiFly1->Play(1.0, 0.0, false);
               	else if (i == 2) YoshiFly2->Play(1.0, 0.0, false);
               	else if (i == 3) YoshiFly3->Play(1.0, 0.0, false);
               	else if (i == 4 || i == 5 || i == 6){}

           		cooldown = 60;
				}
        	else if (PressedDown) {SpY = 1;}
        	if (cooldown < 30)
        		{
            	if (PressedRight) {TurnSide = 1; SpX = SpX + 0.1;}
            	else if (PressedLeft) {TurnSide = -1; SpX = SpX - 0.1;}
            	}
            }  
        }
                         
        if(YoFlyingMode == false && YoFlyAnimationPhase != 0.0) {YoFlyAnimationPhase = 0.0;}
        if(YoFlyingMode == true && YoFlyAnimationPhase < 1.0){ YoFlyAnimationPhase = YoFlyAnimationPhase + YoFlyAnimSpeed;}
        else {YoFlyAnimationPhase = 0.0;}

        if(Type == M_DRAWQUEUE)
        	{
            if(YoFlyingMode == true)
            	{
                local spr1 = YoflySprite->Index;
                if(TurnSide == -1){AddSprite(5,PosX,PosY,0,spr1,YoFlyAnimationPhase);}
                else{AddSprite(5,PosX,PosY,0,spr1+262144,YoFlyAnimationPhase );}
                }
         	}
super;
}

override void CWorm::FireFinal(CWeapon* Weap, CShootDesc* Desc)
{

if(Weap->CheckName("Yoshi"))
	{
	FlyingMode = false;
    YoFlyingMode = true;
    GravityFactor = 0.3;
     
    local i = RandomInt(1, 4);
    if      (i == 1) YoshiJump1->Play(1.0, 0.0, false);
    else if (i == 2) YoshiJump2->Play(1.0, 0.0, false);
    else if (i == 3) YoshiJump3->Play(1.0, 0.0, false);
    else if (i == 4) YoshiJump4->Play(1.0, 0.0, false);
	}
super; 
}

override void CWorm::RenderWorm()
{
         if(YoFlyingMode == false)
         {
             super;
         }
}

override void CSpriteAnim::SetAnimEx(int sindex, fixed frame)   //mostly taken from the end of hang-glider script
{
	if(GS->Tick >= 1)
	{
		CWorm *w = GetCurrentWorm();
		if(w != NullObj && w->Anim == this)
		{
			if(sindex == 289)
			{
			   	if(w->YoFlyingMode == true)
			   	{
				super(YoflybackSprite->Index, frame);
				return;
			   	}
			}
		}
	}
	
	super;
}
