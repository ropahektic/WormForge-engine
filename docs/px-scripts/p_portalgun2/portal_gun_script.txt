void CMissile::GetHitAngle(int posx, int posy)
{
   local worked = false;
   x = posx;
   y = posy;
   local m_koef = 10;           // move koefficient: we actually placing portal a bit away from the wall, to avoid conflict collision
   
////////////   let`s calculate which shape and angle has landscape where missile exploded  ////////////
//  note: rough calculation. reading only 45 degree range:  0, 45, 90, 135, 180, 225, 270, 315, 360  //
        
			fixed SpX2 = SpX;
		        fixed SpY2 = SpY;
                                        
                TraceMaskEx(this, ColMask, PosX, PosY, PosX + SpX2, PosY + SpY2, NoWormsColGroup, Res);
                if(Res->Hit) 
		{        
                        portal_angle = atan2(Res->NormalX, Res->NormalY);
//                        portal_angle_deg = portal_angle/MATH_PI*360-180;                
//                        if (Res->NormalX <= 0.25 && Res->NormalX >= -0.25 && Res->NormalY >= 0.95) // Top
                        if (Res->NormalY >= 0.95) // Top
                        {
                            portal_angle_type = 1;
                            x_koef = 0;
                            y_koef = m_koef;
                            worked = true;
                        }
//                        else if (Res->NormalX < -0.25 && Res->NormalX > -0.850 && Res->NormalY > 0.20 && Res->NormalY < 0.97) // Top Right
                        else if (Res->NormalX < 0 && Res->NormalY > 0.25 && Res->NormalY < 0.97) // Top Right
                        {
                            portal_angle_type = 2;
                            x_koef = -m_koef;
                            y_koef = m_koef;
                            worked = true;
                        }
//                        else if (Res->NormalX <= -0.85 && Res->NormalY >= -0.50 && Res->NormalY <= 0.60) // Right
                        else if (Res->NormalX <= -0.90) // Right
                        {
                            portal_angle_type = 3;
                            x_koef = -m_koef;
                            y_koef = 0;
                            worked = true;
                        } 
//                        else if (Res->NormalX > -0.85 && Res->NormalX < -0.25 && Res->NormalY > -0.97 && Res->NormalY < -0.35) // Bottom Right
                        else if (Res->NormalX < 0 && Res->NormalY < -0.20 && Res->NormalY > -0.97) // Bottom Right
                        {
                            portal_angle_type = 4;
                            x_koef = -m_koef;
                            y_koef = -m_koef;
                            worked = true;
                        }
//                        else if (Res->NormalX <= 0.25 && Res->NormalX >= -0.25 && Res->NormalY <= -0.95) // Bottom
                        else if (Res->NormalY <= -0.97) // Bottom
                        {
                            portal_angle_type = 5;
                            x_koef = 0;
                            y_koef = -m_koef;
                            worked = true;
                        }
//                        else if (Res->NormalX < 0.85 && Res->NormalX > 0.25 && Res->NormalY < -0.35 && Res->NormalY > -0.97) // Bottom Left
                        else if (Res->NormalX > 0 && Res->NormalY < -0.20 && Res->NormalY > -0.97) // Bottom Left
                        {
                            portal_angle_type = 6;
                            x_koef = m_koef;
                            y_koef = -m_koef;
                            worked = true;
                        }  
//                        else if (Res->NormalX >= 0.85 && Res->NormalY >= -0.60 && Res->NormalY <= 0.60) // Left
                        else if (Res->NormalX >= 0.97) // Left
                        {
                            portal_angle_type = 7;
                            x_koef = m_koef;
                            y_koef = 0;
                            worked = true;
                        }
//                        else if (Res->NormalX > 0.25 && Res->NormalX < 0.85 && Res->NormalY > 0.20 && Res->NormalY < 0.97) // Top Left
                        else if (Res->NormalX > 0 && Res->NormalY > 0.25 && Res->NormalY < 0.97) // Top Left                        
                        {
                            portal_angle_type = 8;
                            x_koef = m_koef;
                            y_koef = m_koef;
                            worked = true;
                        }              
		} 
}


void CMissile::CheckWall(int x, int y)
{
  if (open_portal)
  {
     //  Check collision wilth landscape in 7 pixels (becouse missile`s collision mask center has also some radius) around explosion point.
     // i believe missile`s collision mask has 10pxls radius, so 5 - is a half. And we must check collision minimun un 6 pixels from col mask center.
     // 6-7 will work best for basic usage
     // But setting it with larger number, such as 15-30 - it will avoid portals to open on very tight parts of landscape.
     // Remeber the Portal game. if part is too small - its mot gonna open there.
     // So don`t be surprised if portal will not open on some spikey or tight parts of the map.
     local rad = 15;
     int hitp1_x;           int hitp1_y;            
     int hitp2_x;           int hitp2_y;
     int hitp3_x;           int hitp3_y;
     int hitp4_x;           int hitp4_y;
		
     
     CGObject *hitp1 = TraceLine(this, x, y, x+rad, y+rad, 2, &hitp1_x, &hitp1_y);
     CGObject *hitp2 = TraceLine(this, x, y, x-rad, y-rad, 2, &hitp2_x, &hitp2_y);
     CGObject *hitp3 = TraceLine(this, x, y, x+rad, y-rad, 2, &hitp3_x, &hitp3_y);
     CGObject *hitp4 = TraceLine(this, x, y, x-rad, y+rad, 2, &hitp4_x, &hitp4_y);
     
     if (hitp1 != NullObj || hitp2 != NullObj || hitp3 != NullObj || hitp4 != NullObj) MakePortal(portal_bullet, x, y);
     else portal_gun_failhit->Play( 1.0, 1.0, false);
  }
  if (open_portal_there)
  {
    for (i = 1; i < Root->Childs.Count; i++)
    {
        local wall = CPortal_Ignore_Wall(Root->Childs.Objs[i]);
        if (wall == NullObj) continue;
        if (#PORTAL_WALLS)
        {
          if (wall is CPortal_Ignore_Wall)
          {
            if (x > wall->pr_x1 && x < wall->pr_x2 && y > wall->pr_y1 && y < wall->pr_y2)
            {
              if (portal_angle_type > 0)                  // to don`t open portal in bad places with "Ignore Wall" mode
              {
                MakePortal(portal_bullet, x, y);
                return;
              } else {
                open_portal = false;
                return;
              }
            }
          }
        }
    }
  } else
  {
    for (i = 1; i < Root->Childs.Count; i++)
    {
        local wall = CPortal_Ignore_Wall(Root->Childs.Objs[i]);
        if (wall == NullObj) continue;
        if (#PORTAL_WALLS)
        {
          if (wall is CPortal_Ignore_Wall)
          {
            if (x > wall->pr_x1 && x < wall->pr_x2 && y > wall->pr_y1 && y < wall->pr_y2)
            {
                open_portal = false;
                return;
            }
          }
        }
    }
  }
}

void CMissile::TakeOffPortalWeap()
{
     local w = GetCurrentWorm();
     local weaps = GetWeaponByName("Portal Gun");
//     if (w != NullObj) GS->Info.AddWeaponEx(w->WormTeam, weaps->GetWeaponIndex(), weaps->GetPageIndex(), -1);
     if (w != NullObj) {  
                 if (w->portal_firstShot)
                 {
                    w->portal_firstShot = false;
                 }
                 else 
                 {
                    GS->Info.AddWeaponEx(w->WormTeam, weaps->GetWeaponIndex(), weaps->GetPageIndex(), -1);
                    if (w->CurWeapon->CheckName("Portal Gun")) w->SetState(WS_IDLE);
                    w->portal_firstShot = true;
                 }
     }
}

override void CMissile::ExplodeAt(fixed x,fixed y)
{
//    local ang_rad = (-135 + 45 * portal_angle_type)*MATH_PI/180;
    local ang_rad = MATH_HALF_PI - portal_angle;
    if (i_am_portal) {
       DoSpark(7); 
       DoFlash(1, 3);
    }    
    if (portal_bullet == this)
    {
//    if (weap->CheckName("Portal Gun"))     // no longer need this
//    {
     if (PosX < -5 || PosX > GS->LevelSX || PosY < -5 || PosY > Env->Water + 150)
     {
         open_portal = false;
     }
      if (y < Env->Water)
      {
//        local type_reader = this;
//        if (portal_angle_type == 0) portal_angle_type = 2;
        GetHitAngle(x, y);
        
        DoSpark(10);

///////////////////////////////////////////////////////////////////        
        if (PortalWT == 1){                          // red
        portal_team_Params->color = ARGB(85, 210, 20, 0);}
        else if (PortalWT == 2){                     // blue
        portal_team_Params->color = ARGB(85, 0, 20, 220);}     
        else if (PortalWT == 3){                     // green
        portal_team_Params->color = ARGB(85, 0, 220, 0);}
        else if (PortalWT == 4){                     // yellow
        portal_team_Params->color = ARGB(85, 250, 220, 0);}
        else if (PortalWT == 5){                     // purple
        portal_team_Params->color = ARGB(85, 220, 0, 220);}
        else if (PortalWT == 6){                     // cyan
        portal_team_Params->color = ARGB(85, 0, 230, 230);}
///////////////////////////////////////////////////////////////////
        portal_team_Params->blend_type = 1;

        if (ignore_wall_mode)                                                    // maded for MapEditor object, "Ignored walls"
        {
           open_portal = true;
           if (open_portal_there) open_portal = false;
           CheckWall(x, y);                                                  
//           MakePortal(portal_bullet, x, y);
        }
        if (open_portal){                     
            CheckWall(x, y);                                                     // No more portals in the air! only on landscape.
//            MakePortal(portal_bullet, x, y);                                     // make portal
        }
      }
    }
    super;
}

void CMissile::DoFlash(int k, int lt)        // Moar effects! :D   love this one so much :D
{
    for(int i=0 ; i<k ; i += 1)
	{
		p = new PxParticle(plazmaballglowSprite->Index, PosX, PosY);
		
                p->SetLifeTime(lt);
		p->GravityFactor(0);
//		p->SetRandomVelocity(1.5, 0.8);
		p->SetAlpha(150);
		s = RandomFloat(2.0,3.2);
		p->SetStartSize(s, s);
		p->SetEndSize(0.11, 0.11);
		p->SetBlendMode(1);
		
		s = RandomInt(0, 50);
		p->SetStartColor(255,255 - s,255);
		p->SetEndColor(200,225 - s,125);
		
		p->SetMotionRandomness(0.2);
	}
}

void CMissile::DoSpark(int k)        // yyes, do them particles! :D
{
    for(int i=0 ; i<k ; i += 1)
	{
		p = new PxParticle(portal_glow_sprite->Index, PosX, PosY);

void CWorm::ShotPortal(int x, int y, float fireAngle, float firePower)
{
//      GS->Info.AddWeapon(w->WormTeam, w->CurWeapon->GetWeaponIndex(), 1);
      local worm = GetCurrentWorm();
      if (fireAngle < 1.0) portal_type_sending = 1;
      else if (fireAngle < 0.66) portal_type_sending = 2;
      else if (fireAngle < 0.33) portal_type_sending = 3;
      
      if (keySync->KeyPressed(18) || keySync->KeyPressed(17))
      {
         ctrl_alt_pressed = true;
      } else {
         ctrl_alt_pressed = false;
      }
         local ang = (1 - fireAngle) * 3.14159;
         if (TurnSide < 0) ang = 6.28318 - ang;
         
         float PowerInit = firePower * 35.0;          // bullet should be faster then worm`s max speed (32)
         
         local sx = sin(ang) * PowerInit;
         local sy = - cos(ang) * PowerInit;
                
         local ang = (1 - fireAngle) * 3.14159;
         if (TurnSide < 0) ang = 6.28318 - ang;
         
         local ldata = GetWeaponByName("Portal Gun");
         local team = worm->WormTeam;       
         CShootDesc SDesc;
         zero(&SDesc);	
         SDesc.SpX = sx;
         SDesc.SpY = sy;
         SDesc.X = x + sin(ang);      // 15 is a worm`s rad
         SDesc.Y = y - cos(ang);
         SDesc.Team = team;
		
         portal_bullet = new CMissile(Root->GetObject(25, 0), &ldata->launch, &SDesc);
         
         portal_num2 = MakeNumberString("portal_", WormTeam, "");
         portal_gun_shoot->Play( 1.0, 1.0, false);
         read_portal_name = true;   
}

{
        if (Weap->CheckName("Portal Gun"))
        {       
		ShotPortal(PosX, PosY, FireAngle, FirePower);
		
                local f_a = FireAngle;          // FireAngle brokes often when adding weapon (somehow), so lets better remember it.
                 GS->Info.AddWeaponEx(WormTeam, Weap->GetWeaponIndex(), Weap->GetPageIndex(), 1);
                 last_WormTeam = WormTeam;
                 if (FireAngle > 0.99 || FireAngle < 0.01)
                 {
                     FireAngle = f_a;
                 }
	}
	
	else super;
}
