require p_sprite_builder;

CSprite* Mummy_Scarabs_Aiming;                     

void Mummy_Scarabs_SpriteBuilder::InitGraphic()      
{   CFile* f;                                        
    f = GetAttachment("Scarab Vase.png");            
    Mummy_Scarabs_Aiming = LoadSprite(f, 1, 0);      
}                                                    
                                                     
override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData) { 
 super;                                              
  if (Type == M_FRAME) {                             
    if (gframe == 1) {                               
     local weapName; local sprite;

     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Mummy Scarabs";                        
     sprite = Mummy_Scarabs_Aiming->Index;              
     Params->hand_radius = 25.0;                                                 
     Params->hand_rotation = 1.0;                       
     Params->hand_radial_rotation = 1.0;
     Params->weap_radial_rotation = 0.9;     
     Params->weap_rotation = -1.0;
     Params->weap_radius = 12.0;
     Params->hand_type = 2;     
     Params->hand_scale = 0.8;  
//     Params->zplane = 10.0;   
//     Params->scale = 1.0;     

     local WA = new WormSprite(weapName, sprite, Params);
    }                                                                  
  }                                                                    
}                                                                      

override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData){    
super;
  if (Type == M_FRAME){
     if (weap->CheckName("Mummy Scarabs")) {
         if (Mummy_Scarabs_anim) {          
              local sprite = Mummy_Scarabs_Aiming->Index; 
              WeapSpriteParams* Params = new WeapSpriteParams;
              new WeapSprite(this, sprite, Params);              
              weap->launch.anim.spriteIndex = 0;
              Mummy_Scarabs_anim = false;       
         }                                                                     
     }                                                                                                                                                     
  }                                             
}                                               
override CMissile::CMissile(CObject* parent,CWeaponLaunch* ldata,CShootDesc* sdata){
super;                                          
  Mummy_Scarabs_anim = true;                    
}
