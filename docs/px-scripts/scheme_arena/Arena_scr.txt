//vars
CEnemySpawn : CObject;
CBasePoint : CObject;

CSprite* BaseSprite;

int SpawnPosX;
int SpawnPosY;
int GlobalWave;
CWorm* CurrentChallenger;
bool DestroyEnemies;    
int MobNum;
CBasePoint* Base;


//var init
void Arena_scr::InitGraphic()
{
    local spr1       = GetAttachment("base.png");             
    BaseSprite     = LoadSprite(spr1, 1, 0); 
} 

void Arena_scr::Init()
{
GlobalWave = 0;  //Global progress for all objects
CurrentChallenger = NullObj;
SpawnPosX = 0;       //spawn pars
SpawnPosY = 0;
DestroyEnemies = false; //^
MobNum = 0;
}



//scr

void ArenaSpawnPoint(CEditorObject* obj)
{
  x = obj->PosX;
  y = obj->PosY;

  for (j = 1; j < 7; j++) //team
  {
    for (i = 1; i < 9; i++)     //worm
    {
        local Worm = CWorm(Root->GetObject(2, 16 * j + i));
        if (Worm != NullObj)
        {
          Worm->PosX = x;
          Worm->PosY = y;
          if(SpawnPosX == 0 && SpawnPosY == 0)
          {
          SpawnPosX = x;
          SpawnPosY = y;
          }
          
          x += 10;
        };
    };
  };
}; 



CEnemySpawn::CEnemySpawn(CEditorObject* aobj)
{
  super(Root, GS);
  PosX = aobj->PosX;
  PosY = aobj->PosY;
  CurrentWave = 0;
  
  if (StrCmp(aobj->GetString("Direction"), "Left"))
  {
    DirType = 1;      //def
  };
  if (StrCmp(aobj->GetString("Direction"), "Up"))
  {
    DirType = 2;
  };
  if (StrCmp(aobj->GetString("Direction"), "Right"))
  {
    DirType = 3;
  };
  if (StrCmp(aobj->GetString("Direction"), "Down"))
  {
    DirType = 4;
  }; 
  
  if (StrCmp(aobj->GetString("Boss"), "No"))
  {
    BossType = false;      //def
  };
  if (StrCmp(aobj->GetString("Boss"), "Yes"))
  {
    BossType = true;
  };
   
};



CBasePoint::CBasePoint(CEditorObject* aobj)
{
  super(Root, GS);
  PosX = aobj->PosX;
  PosY = aobj->PosY;
};

void SpawnES(CEditorObject* obj)
{
  new CEnemySpawn(obj);
};

void SpawnBase(CEditorObject* obj)
{
  Base = new CBasePoint(obj);
};


///////////MESSAGES AND STUFF///////////////////////////////////////////////////
void CBasePoint::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
   local spr1 = BaseSprite->Index;
 if(this == Base)
 {
         if(Type == M_DRAWQUEUE)
         {
         AddSprite(5,PosX,PosY,0,spr1,0.0);
         }
 }
}

////////////////////////////////////////////////WAVE MANAGEMENT/////////////////
void CEnemySpawn::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
 if(Type == M_FRAME)
 {
  if(GetCurrentWorm() != NullObj)
  {
      if(CurrentChallenger != NullObj) //Wave spawning
      {
           if(CurrentWave < GlobalWave)
           {
           CurrentWave = GlobalWave;
           SpawnEnemys(PosX, PosY, CurrentWave, DirType, BossType);
           }
           if(CurrentWave > GlobalWave)
           {
           CurrentWave = GlobalWave;
           }
      }
  }
 }
 
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
  if(Weap->CheckName("Fight") && MobNum == 0)
  {
  GlobalWave = GlobalWave + 1;
  ShowMessage(MakeNumberString("Wave: ",GlobalWave,"!"));
  }
  super;
}   

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
  if(Type == M_TURNEND)
  {
  if(MobNum == 0)ShowMessage("Challenger skipped turn!");
  if(MobNum != 0)ShowMessage("Challenger got hit!");
  if(MobNum != 0)GlobalWave = GlobalWave - 1;
  if(MobNum != 0)DestroyEnemies = true;
  PosX = SpawnPosX;
  PosY = SpawnPosY;
  }
  super;
}
