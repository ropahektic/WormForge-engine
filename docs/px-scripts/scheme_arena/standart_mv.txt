void CAttacker::Move(CGObject* obj1, CBasePoint* obj2)
{
 if(obj1 != NullObj)
 {
            if(PosX == obj1->PosX){MovingX = false; SpX = 0;}
            if(PosX != obj1->PosX)
                    {
                    MovingX = true;
                            if(PosX > obj1->PosX){SpX = -MoveSpeed;}
                            if(PosX < obj1->PosX){SpX = MoveSpeed;}
                    }
                                    
            if(PosY == obj1->PosY){MovingY = false; SpY = 0;}
            if(PosY != obj1->PosY)
                    {
                    MovingY = true;
                            if(PosY > obj1->PosY){SpY = -MoveSpeed;}
                            if(PosY < obj1->PosY){SpY = MoveSpeed;}
                    }
 }
 
 if(obj2 != NullObj)
 {
            if(PosX == obj2->PosX){MovingX = false; SpX = 0;}
            if(PosX != obj2->PosX)
                    {
                    MovingX = true;
                            if(PosX > obj2->PosX){SpX = -MoveSpeed;}
                            if(PosX < obj2->PosX){SpX = MoveSpeed;}
                    }
                                    
            if(PosY == obj2->PosY){MovingY = false; SpY = 0;}
            if(PosY != obj2->PosY)
                    {
                    MovingY = true;
                            if(PosY > obj2->PosY){SpY = -MoveSpeed;}
                            if(PosY < obj2->PosY){SpY = MoveSpeed;}
                    }
 }              
}

void CAttacker::Explode(CGObject* obj)
{
           DoExplosion(obj->PosX,obj->PosY, 50, 25, 0, 0);
           Health = 0; 
}






void CAttacker::StandartMovement()
{
   if(DistBetweenObjs2(Base, this) > 250 && DistBetweenObjs(CurrentChallenger, this) > 250)//resetting to hovering state
           {
           MovingX = false; MovingY = false;
           } 
////////////////////////////////////BASE PATHFINDING////////////////////////////
           if(DistBetweenObjs2(Base, this) < 250 && DistBetweenObjs(CurrentChallenger, this) > 250)    //attack the base!
           {
           Move(NullObj, Base);              
           }
           
           if(DistBetweenObjs2(Base, this) < 25)    //base in proximity zone!
           {
           Explode(CurrentChallenger);
           }
////////////////////////////////////BASE PATHFINDING END////////////////////////         
                               
////////////////////////////////////WORM PATHFINDING////////////////////////////  
           if(DistBetweenObjs(CurrentChallenger, this) < 25)    //worm in proximity zone!
           {
           Explode(this);
           }   
   
           if(DistBetweenObjs(CurrentChallenger, this) < 250)      //most awesomiest ai i have ever created :PP 
           {
           Move(CurrentChallenger, NullObj);              
           } 
           
           if (MovingX == false && MovingY == false)//Hovering state
           {
           if(FloatDirection == 1){SpX = -1;}
           if(FloatDirection == 2){SpY = -1;}
           if(FloatDirection == 3){SpX = 1;}
           if(FloatDirection == 4){SpY = 1;} 
           }
}
