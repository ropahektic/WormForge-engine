void CAttacker::TakeExplosionDamage(fixed x, fixed y, int dmg)
{
	if(dmg == 0) return;
	
	float fdmg = dmg;
	float dx = PosX - x;
	float dy = PosY - y;
	
	float distsqr = dx*dx + dy*dy;
	float fdmgsqr = fdmg*fdmg*4.0;
	
	fdmg = (1.0-distsqr/fdmgsqr) * fdmg;
	dmg = fdmg;
	
	if(dmg > 0) TakeDamage(dmg);
}


void CAttacker::TakeDamage(int dmg)
{	
	Health -= dmg;
}


int DistBetweenObjs2(CBasePoint* ObA, CGObject* ObB)
{
  int dx = ObA->PosX - ObB->PosX;
  int dy = ObA->PosY - ObB->PosY;

  int r = sqrt(dx * dx + dy * dy);
  
  return r;
};
