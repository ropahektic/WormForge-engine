void CWorm::Death()
{
  if(PosY > GS->LevelSY)
  {
  CanBeZombie = false;
  Zombie = false;
  Act = false;
  }
                 
  if(GS->Info.GetWormHealth(WormTeam,WormNumber) <= 0 && CanBeZombie == true)
  {
  Zombie = true;
  CanBeZombie = false;
  GS->Info.SetWormHealth( WormTeam, WormNumber, 100);
  }
}

void CWorm::MakeLeap(CWorm* Leaper, CWorm* Target)     //how to skillfully apply aiming script to leaping worm:
{
  if(Leaper->PosX < Target->PosX){Leaper->TurnSide = 1;}else{Leaper->TurnSide = -1;}
  CShootDesc sdesc;
  zero(&sdesc);

  local l = DistBetweenObjs(Target, Leaper);
  local timeForShoot = l / 10 + 1;         
  local x = Target->PosX - PosX;
  local rx = x / timeForShoot;
  local y = Target->PosY - PosY;
  local ry = SolveBallistic(y, timeForShoot, Env->Gravity * Target->GravityFactor);
  
  local angToShoot = 0.0;
  local shootdir = 0;
  if (rx > 0)
  {
    angToShoot = atan2(rx, 0 - ry);
    shootdir = 1;
  } else {
    angToShoot = atan2(0 - rx, 0 - ry);
    shootdir = -1;  
  };
  
  local shootPower = sqrt(rx * rx + ry * ry);
  local ForceVar = GS->GetRandom() % 64;
  local BaseForce = shootPower - 0.1 + ForceVar / 96.0;
  
  sdesc.X = PosX + sin(angToShoot) * shootdir * 14;      //
  sdesc.Y = PosY - cos(angToShoot) * 14;                 //
  local ShootSX = sin(angToShoot) * shootdir;
  local ShootSY = 0.0 - cos(angToShoot);
  
  SpX = ShootSX * BaseForce;          //
  SpY = ShootSY * BaseForce;          //
}
