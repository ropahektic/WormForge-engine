//Flag
CGolfNewSpriterHole::CGolfNewSpriterHole(CEditorObject* aobj)
{
  super(Root, GS);
  //scale = 0.33;
  scale = 1;
  x = aobj->PosX + 20 *scale;
  //y = aobj->PosY - 68 *scale;
  //y = aobj->PosY - 64 *scale;
  y = aobj->PosY - 58 *scale;
  z = 16.0 + aobj->GetInt("Z_Layer")/100;
  rot = 0.0;
  hx1 = x-25;
  hx2 = x-5;
  hy1 = y+46;
  hy2 = y+58;
  par = aobj->GetInt("Par");
  hole = aobj->GetInt("Hole_Number");
  flagColor = aobj->GetString("Color");

  lastHole = false;
  lastHole = aobj->GetBool("LastHole");
  
  if(lastHole == false){
    if(hole == 1)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag1_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag1_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag1_blue->Index;               
    }else if(hole == 2)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag2_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag2_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag2_blue->Index;               
    }else if(hole == 3)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag3_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag3_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag3_blue->Index;               
    }else if(hole == 4)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag4_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag4_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag4_blue->Index;               
    }else if(hole == 5)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag5_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag5_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag5_blue->Index;               
    }else if(hole == 6)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag6_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag6_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag6_blue->Index;               
    }else if(hole == 7)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag7_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag7_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag7_blue->Index;               
    }else if(hole == 8)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag8_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag8_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag8_blue->Index;               
    }else if(hole == 9)
    {
      if (StrCmp(aobj->GetString("Color"), "Red")) sprite = golfFlag9_red->Index;
      if (StrCmp(aobj->GetString("Color"), "Yellow")) sprite = golfFlag9_grey->Index;
      if (StrCmp(aobj->GetString("Color"), "Blue")) sprite = golfFlag9_blue->Index;               
    }
  

    }else{ sprite = golfFlag_final->Index; }
   
  flagAnimTimer = 0.0;
}

void CGolfNewSpriterHole::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_DRAWQUEUE)
    {
        AddSpriteEx( z, x, y, sprite, flagAnimTimer, rot, scale);
//        AddSpriteEx( z, (hx1+hx2)/2, (hy1+hy2)/2, 611, 0.0, rot, scale);
    }
    if (Type == M_FRAME)
    {  
        if (flagAnimTimer < 1.0) {
                  flagAnimTimer += 0.06;
               }
        else flagAnimTimer = 0.0;

    }
}

//Start Sign
CGolfNewHoleStart::CGolfNewHoleStart(CEditorObject* aobj)
{
  super(Root, GS);
  hole = aobj->GetInt("HoleNumber");
  x = aobj->PosX;
  y = aobj->PosY;
  x_sign = aobj->PosX - 90;
  y_sign = aobj->PosY + 6;
  z_sign = 1.0/100;
  restart = aobj->GetBool("Restart_Counter");
  if (StrCmp(aobj->GetString("Direction"), "LEFT_to_RIGHT")) offset = -13;
  if (StrCmp(aobj->GetString("Direction"), "RIGHT_to_LEFT"))
    { 
    offset = 13;
    x_sign = aobj->PosX + 90;
    }
  
  if (aobj->GetInt("HoleNumber") == 1) startSignsprite = golfStartSign1->Index;
  if (aobj->GetInt("HoleNumber") == 2) startSignsprite = golfStartSign2->Index;
  if (aobj->GetInt("HoleNumber") == 3) startSignsprite = golfStartSign3->Index;
  if (aobj->GetInt("HoleNumber") == 4) startSignsprite = golfStartSign4->Index;
  if (aobj->GetInt("HoleNumber") == 5) startSignsprite = golfStartSign5->Index;
  if (aobj->GetInt("HoleNumber") == 6) startSignsprite = golfStartSign6->Index;
  if (aobj->GetInt("HoleNumber") == 7) startSignsprite = golfStartSign7->Index;
  if (aobj->GetInt("HoleNumber") == 8) startSignsprite = golfStartSign8->Index;
  if (aobj->GetInt("HoleNumber") == 9) startSignsprite = golfStartSign9->Index;

}

void CGolfNewHoleStart::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
    if (Type == M_DRAWQUEUE)
    {     
        AddSpriteEx( z_sign, x_sign, y_sign, startSignsprite, 0.0, 0.00, 1);
    }
}



void SpawnHoleFlag(CEditorObject* obj)
{
  zone = new CGolfNewSpriterHole(obj);
}

void SpawnNextHole(CEditorObject* obj)
{
  zone = new CGolfNewHoleStart(obj);
