require utils;

bool deb_speedup_state;

override void CTurnGame::Message(CObject* sender, EMType type, int size, CMessageData* data)
{
    if (type == M_TURNEND)
    {
        // Reset the game speed
        if (deb_speedup_state)
        {
            deb_speedup_SetSpeedup(false);
        }
    }
    super;
}

override void CWorm::FireFinal(CWeapon* Weap, CShootDesc* Desc)
{
    if (Weap->CheckName("Speedup"))
    {
        deb_speedup_SwitchSpeedup();
    }
    super; 
}

void deb_speedup_SwitchSpeedup()
{
    deb_speedup_SetSpeedup(!deb_speedup_state);
}

void deb_speedup_SetSpeedup(bool enable)
{
    deb_speedup_state = enable;
    if (enable)
    {
        deb_gamespeed_SetGameSpeed(deb_gamespeed_GetGameSpeed() * 1.7);
    }
    else
    {
        deb_gamespeed_SetGameSpeed(deb_gamespeed_GetGameSpeed() / 1.7);   
    }
}
