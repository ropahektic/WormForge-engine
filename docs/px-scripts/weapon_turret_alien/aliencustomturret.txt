void CCustomAlienTurret::ChooseWeapon2(int i)
{
    i = weapI;
    if (CheckMaskAt(this, FourCheckMask, ShootX, ShootY, -1) != NullObj) return false;
    CMineParams MParams;
    CShootDesc SDesc;
    zero(&MParams);
    zero(&SDesc);
    
    local ForceVar = GS->GetRandom() % 64;
    local BaseForce = shootPower - 0.1 + ForceVar / 96.0;
    
    SDesc.Team = turretTeam;
    SDesc.X = ShootX;
    SDesc.Y = ShootY;
    local worm = GetCurrentWorm();
    SDesc.Worm = worm->WormNumber;
    SDesc.AddX = worm->PosX;
    SDesc.AddY = worm->PosY;
    
    
    if (turretTeam < 0) SDesc.Team = 0;

    SDesc.SpX = ShootSX * BaseForce;
    SDesc.SpY = ShootSY * BaseForce;
    SDesc.Delay = ExplTimer;
    
    MParams.Prefuse = 0;
    MParams.Flags = 60;
    MParams.Bias = 0;
    MParams.Damage = 50;
    MParams.BlastPower = 100;
    MParams.Fuse = ExplTimer; //20 frames
    MParams.Radius = 48;
    
    if (i == 20) {CFireDesc FDesc; FDesc.X = ShootX; FDesc.Y = ShootY;  FDesc.SpX = ShootSX * BaseForce * 4; FDesc.SpY = ShootSY * BaseForce * 4; FDesc.Count = 0; FDesc.Duration = 1500; FDesc.UpPower = 1; FDesc.Damage = FNum-1; FDesc.Unk = 0; FDesc.UnkB = admg/5; FDesc.UnkD = 0; FDesc.OwnerTeam = turretTeam;
           random = new CFire(Root->GetObject(25, 0), &FDesc, 0); PlaySound(22, 10, RandomFloat(1.0, 1.0+BaseForce/30.0), 0.9); return;}
}

override void CWorm::FireFinal(CWeapon* Weap, CShootDesc* Desc)
{
   if (Weap->CheckName("Alien Turret"))
   {
     local sg = new CCustomAlienTurret(GetObject(25, 0), PosX + 15 * TurnSide, PosY - 50);
     sg->turretTeam = WormTeam;
     sg->ammoPerTurn = 1000;
     sg->weaptype = 20;
     sg->sprNum = 0;
   } else {
     super;
   };
};

void CCustomAlienTurret(CEditorObject* eobj)
{
  sprNumSet2 = true;
  weaptypeSet2 = true;
  if (sprNumSet2)        //less lags at start plz
  {
      if (StrCmp(eobj->GetString("WeapType"), "magicbull")) {sprNumW2 = 134; sprNumSet2 = false;}
      if (StrCmp(eobj->GetString("WeapType"), "random")) {sprNumW2 = 102; sprNumSet2 = false;} //599
  } 
  local sg = new CCustomAlienTurret(Root->GetObject(25, 0), eobj->PosX, eobj->PosY);
  sg->turretTeam = -1;
  sg->clearMines = eobj->GetBool("ClearMines");   
  sg->ammoPerTurn = eobj->GetInt("AmmoPerTurn");
  if (weaptypeSet2)        //less lags at start plz
  {
     if (StrCmp(eobj->GetString("WeapType"), "magicbull")) {sg->weaptype = 20; weaptypeSet2 = false;}
     if (StrCmp(eobj->GetString("WeapType"), "random")) {sg->weaptype = 21; weaptypeSet2 = false;}
  } 
  sg->FNum = eobj->GetInt("FireNum");          
  sg->admg = eobj->GetInt("Damage");
  sg->ExplTimer = eobj->GetInt("Delay");
  sg->health = eobj->GetInt("Health");
  sg->minimalRadius = eobj->GetInt("MinRadius");
  sg->sensorRadius = eobj->GetInt("SensorRadius");
  sg->lostRadius = sg->sensorRadius + 100;
  sg->ExpDamage = eobj->GetInt("ExplDamage");
  sg->maxHealth = sg->health;
  sg->sprNum2 = sprNumW2; 
  aGrav = eobj->GetInt("ArrowsGravity") / 100.0;
  if (eobj->GetBool("Invincible"))
  {
     sg->breakable = false;
  }
  if (eobj->GetBool("Mounted"))
  {
    sg->SpeedDivider = 32767;
    for (local i = 0; i < 100; i++)
    {
      if (Env->CheckCollision(sg, sg->PosX, sg->PosY + 1) != NullObj) break;
      sg->PosY += 1;
    }
  }
};
