override void CMissile::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
if(Cbumpr == true)
{
if(Type == M_FRAME)
{
BumprCol(15.0);
}
}
super;
}

void CMissile::BumprCol(float Speed)
{
CGObject *bskob;

for(int i = 0; i < Env->Objs.Count; i += 1)
{
bskob = CGObject(Env->Objs.Objs[i]);
if(bskob == this) continue;
if(bskob == NullObj) continue;
if(GetCurrentWorm() != NullObj)
{
if(bskob == GetCurrentWorm()) continue;
}
if(bskob->ClType == OC_Worm 
|| bskob->ClType == OC_Mine 
|| bskob->ClType == EObjectClass(71)
|| bskob->ClType == EObjectClass(73)
|| bskob->ClType == EObjectClass(74))
{
if(col_bmp->Check(PosX, PosY, bskob->ColMask, bskob->PosX, bskob->PosY))
{
float ForceX = bskob->PosX - PosX;
float ForceY = bskob->PosY - PosY;
float dist = sqrt(ForceX*ForceX + ForceY*ForceY);

if(dist > 0)
{
ForceX = 0.6*sqrt(Speed) * ForceX / dist;
ForceY = 0.6*sqrt(Speed) * ForceY / dist;

PlaySound(100,1.0,1.0,0.8);

CMessageData msg;
msg.params[0]	= 0;
msg.fparams[1]	= PosX;
msg.fparams[2]	= PosY;
msg.fparams[3]	= ForceX;
msg.fparams[4]	= ForceY;
msg.params[5]	= 0;
msg.params[6]	= 0;

bskob->Message(this, M_GUNEXP, 1032, &msg);
}
}
}
}
}
