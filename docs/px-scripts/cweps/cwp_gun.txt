//---------------------------GUN-LAUNCHER---------------------------------------
//    NINJA STAR                                       RIPPER
//    FIRE PELLET        REQUIRED IN PXS FOR:          PILLAGER
//    TORCH FLAME                                      MINI-TORCH
//
//    PILLAGER REQUIRES RUBBERWORM FLAME LIMIT BE INCREASED TO AT LEAST 230
//------------------------------------------------------------------------------

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
//------RIPPER------------------------------------------------------------------
if(Weap->CheckName("Ripper"))
{
Desc->SpX = (TurnSide * (0.5 - sqrt((0.5 - FireAngle)*(0.5 - FireAngle)))) * 45;
Desc->SpY = 45*(0.5 - FireAngle);
super(GetWeaponByName("Ninja Star"), Desc);
return;
}
//------PILLAGER----------------------------------------------------------------
if(Weap->CheckName("Pillager"))
{
Desc->SpX = (TurnSide * (0.5 - sqrt((0.5 - FireAngle)*(0.5 - FireAngle)))) * 42;
Desc->SpY = 42*(0.5 - FireAngle);
super(GetWeaponByName("Fire Pellet"), Desc);
return;
}
//------MINI-TORCH--------------------------------------------------------------
if(Weap->CheckName("Mini-Torch"))
{
Desc->SpX = (TurnSide * (0.5 - sqrt((0.5 - FireAngle)*(0.5 - FireAngle)))) * 20;
Desc->SpY = 20*(0.5 - FireAngle);
super(GetWeaponByName("Torch Flame"), Desc);
return;
}

super;
}
