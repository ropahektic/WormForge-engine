override void CWorm::Message( CObject *sender, EMType Type, int mSize, CMessageData *mData )
{
   if(Type == M_TURNEND)
     {
          local cw = GetCurrentWorm();  
          
          if(cw->ObjState == WS_FLYING)
          {
             local fm = RandomInt(1,5);             
             if (fm == 1) bruh->Play(1.0, 0.0, false);
             else if (fm == 2) cartoonhead->Play(1.00, 0.0, false);
             else if (fm == 3) cartoonslip->Play(1.0, 0.0, false);
             else if (fm == 4) clownh->Play(1.0, 0.0, false);
             else if (fm == 5) jumpingrope->Play(1.0, 0.0, false);
                  
          }
          else if(cw->ObjState == WS_FLYING2)
          {
             local fm = RandomInt(1,4);
             if (fm == 1) carnivalm->Play(1.0, 0.0, false);
             else if (fm == 2) claughing->Play(1.0, 0.0, false);
             else if (fm == 3) sadtrombone->Play(1.0, 0.0, false);
             else if (fm == 4) sadtuba->Play(1.0, 0.0, false);     
          }
          
          else if(cw->ObjState == WS_POWERFLY)
          {
             local fm = RandomInt(1,3);
             if      (fm == 1) ibelieveicanfly->Play(1.0, 0.0, false);
             else if (fm == 2) gyell->Play(1.0, 0.0, false);
             else if (fm == 3) tomscream->Play(1.0, 0.0, false);                  
          }
          
          else if(cw->ObjState == WS_SINKING)
          {
             local fm = RandomInt(1,8);
             if      (fm == 1) titanic->Play(1.0, 0.0, false);
             else if (fm == 2) timetosaygoodbye->Play(1.0, 0.0, false);
             else if (fm == 3) tragicviolin->Play(1.0, 0.0, false);
             else if (fm == 4) onlyt->Play(1.0, 0.0, false);             
             else if (fm == 5) whatwonderfulworld->Play(1.0, 0.0, false);
             else if (fm == 6) dontworry->Play(1.0, 0.0, false);
             else if (fm == 7) evilmorty->Play(1.0, 0.0, false);
			 else if (fm == 8) aaahh->Play(1.0, 0.0, false);
          }
     }
     
   if(Type == M_SKIPTURN)
     {
             local fm = RandomInt(1,3);
             if      (fm == 1) beep->Play(1.0, 0.0, false);
             else if (fm == 2) whistle->Play(1.0, 0.0, false);
             else if (fm == 3) windowserror->Play(1.0, 0.0, false);  
     }
     
   if(Type == M_SURRENDER)
     {
             local fm = RandomInt(1,2);
             if      (fm == 1) hellodarkness->Play(1.0, 0.0, false);
             else if (fm == 2) mariogo->Play(1.0, 0.0, false);  
     }  
  
super;
}
