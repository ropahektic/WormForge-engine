require p_sprite_builder;
CSprite* AlienEnergyGun_aiming;                                                                                                                                  //

void AlienEnergyGun_SpriteBuilder::InitGraphic()                                  
{   CFile* f;
    f = GetAttachment("Energy Gun.png");                            
    AlienEnergyGun_aiming = LoadSprite(f, 1, 0);                               
}

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
 super;
  if (Type == M_FRAME) {
    if (gframe == 1) {
     local weapName; local sprite;

     CWormAnimParams* Params = new CWormAnimParams;
     weapName = "Alien Energy Gun";                                             
     sprite = AlienEnergyGun_aiming->Index;                                     
     Params->hand_radius = 18.0;                                                 
     Params->hand_rotation = 5.0;                                               
     Params->hand_radial_rotation = 1.0;
     Params->weap_radial_rotation = 0.9;     
     Params->weap_rotation = -1.0;
     Params->weap_radius = -10.0;
     Params->hand_type = 1;     
     Params->hand_scale = 0.0;  
//     Params->zplane = 10.0;   
//     Params->scale = 1.0;     

     local WA = new WormSprite(weapName, sprite, Params);
    }
  }
}
