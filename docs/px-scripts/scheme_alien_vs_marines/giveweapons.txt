int alienframes;

override void CTurnGame::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
         if(Type == M_FRAME)
         {
                 if(alienframes < 15)
                 {
                 alienframes = alienframes+1;
                 }
                 if(alienframes == 10){GiveWeapons();}
         }
super;
}
