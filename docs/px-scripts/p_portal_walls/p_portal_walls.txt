void p_portal_walls::FirstFrame()
{
    portal_loop2_Params = new CAnimationParams;
        
        portal_loop2_Params->speed = 35;
        portal_loop2_Params->ZPlane = 9;
        portal_loop2_Params->scale = 0.8;
        portal_loop2_Params->turns = 500;
        portal_loop2_Params->loop = true;
        portal_loop2_Params->alpha = 45;
        portal_loop2_Params->blend_type = 3;  
}

CPortalOnMap::CPortalOnMap(CEditorObject* aobj)
{
   super(Root, GS);
   x_koef = 0;
   y_koef = 0;
   x = int(aobj->PosX);
   y = int(aobj->PosY);
   if (StrCmp(aobj->GetString("Type"), "Orange")) PType = 1;
   if (StrCmp(aobj->GetString("Type"), "Blue")) PType = 2;
   LinkedTo = StrSave(aobj->GetString("LinkedTo"));
   if (StrCmp(aobj->GetString("AngleType"), "Top")) {AngType = 1; x_koef = 0; y_koef = 10;}
   if (StrCmp(aobj->GetString("AngleType"), "TopRight")) {AngType = 2; x_koef = -10; y_koef = 10;}
   if (StrCmp(aobj->GetString("AngleType"), "Right")) {AngType = 3; x_koef = -10; y_koef = 0;}
   if (StrCmp(aobj->GetString("AngleType"), "BottomRight")) {AngType = 4; x_koef = -10; y_koef = -10;}
   if (StrCmp(aobj->GetString("AngleType"), "Bottom")) {AngType = 5; x_koef = 0; y_koef = -10;}
   if (StrCmp(aobj->GetString("AngleType"), "BottomLeft")) {AngType = 6; x_koef = 10; y_koef = -10;}
   if (StrCmp(aobj->GetString("AngleType"), "Left")) {AngType = 7; x_koef = 10; y_koef = 0;}      
   if (StrCmp(aobj->GetString("AngleType"), "TopLeft")) {AngType = 8; x_koef = 10; y_koef = 10;}
   ExitOnly = aobj->GetBool("ExitOnly");
   
   t_name = aobj->GetName();
   port = CPortal(NullObj);
}

void CPortalOnMap::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
   if (Type == M_FRAME)
   {
      if (port == NullObj) 
      {
          port = new CPortal(PType, x + x_koef, y + y_koef, 40, 50, 0, 0, ExitOnly, false, LinkedTo, t_name, true, AngType);
          if (PType == 1) {
              portal_loop2_Params->sprite = portal_orange2->Index;
          } else {
              portal_loop2_Params->sprite = portal_blue2->Index;
          }
          portal_loop2_Params->rot = 2*(-45+45*(AngType));
          local anim = new CAnimation(Root->GetObject(25, 0), x, y, NullObj, portal_loop2_Params);
      }
   }
}


CPortal_Ignore_Wall::CPortal_Ignore_Wall(CEditorObject* aobj)
{
  super(Root, GS);

  pr_x2 = aobj->PosX + aobj->GetInt("Size");
  pr_y1 = aobj->PosY;
  pr_y2 = aobj->PosY + aobj->GetInt2("Size");
  timer = 0;

//  ignore_active = false;

CPortal_Ignore_WallControl::CPortal_Ignore_WallControl(CEditorObject* aobj)
{
  super(Root, GS);

}
