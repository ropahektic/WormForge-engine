require utils, utility_laser_pointer;

CSoundFile* snipfiring;

void LaserSight::Init()
{
	snipfiring       = new CSoundFile(GetAttachment("Sniper Firing.wav"));
}

override void CWorm::Message(CObject* sender,EMType Type,int MSize,CMessageData* MData)
{
super;
 if(GetCurrentWorm() != NullObj)
 {
  if(GetCurrentWorm() == this && ObjState == WS_AIMING)
  {
    if(CurWeapon->CheckName("Sniper Rifle") || CurWeapon->CheckName("Sniper Rifle "))
    {
      if(Type == M_DRAWQUEUE)
      {  
      local LaserLen = 15000;
      local ang = (1 - FireAngle) * 3.14159;
      if (TurnSide < 0) 
      {
         ang = 6.28318 - ang;
      };
      MakeLaserRay(this, PosX + sin(ang) * 12, PosY - cos(ang) * 12, 2, ang, LaserLen);
      }
    }
  }
 }
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
  if(Weap->CheckName("Sniper Rifle") || Weap->CheckName("Sniper Rifle "))
  {
  vol = CalculateSoundVolume(PosX,PosY);
  pan = CalculateSoundPan(PosX,PosY);
  snipfiring->Play(vol, pan, false);
  }
  super;
}
