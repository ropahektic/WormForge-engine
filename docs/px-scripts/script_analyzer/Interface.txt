void Interface::FirstFrame()
{
 new CInterfaceHandler(Root);
 TurnOn = false;

}

 

void CInterfaceHandler::Toggler()
{
 CWorm *w = GetCurrentWorm();
 
if(w != NullObj)
 {
 if (IsKeyPressed(79))     // O
    {     
         TurnOn = true;    
    }
 if (IsKeyPressed(80))     // P
    {     
         TurnOn = false; 
    }   
 if (IsKeyPressed(219))     // [
    {     
         CamLock = true; 
    } 
 if (IsKeyPressed(221))     // ]
    {     
         CamLock = false; 
    }
 } 
}
