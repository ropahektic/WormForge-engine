// PxKirovAirshipBomb

PxKirovAirshipBomb : CMine;

PxKirovAirshipBomb::PxKirovAirshipBomb(CObject* Parent, CShootDesc* Desc)
{
	CMineParams MParams;
	zero(&MParams);
	MParams.Prefuse = 0;
	MParams.Radius = 0;
	MParams.Fuse = 1;
	MParams.Flags = 0;
	MParams.Bias = 0;
	MParams.Damage = 0;
	MParams.BlastPower = 0;
	
	super(Parent, &MParams, Desc, false, 0);
	
	// Parameters
	
	SpeedMultiplier = 0.98;
	SpritePara = kirovBombSprite->Index;
	SpriteFall = 250;
	
	ExplosionDamage = 70;
	ExplosionBlast = 115;
	
	AnimSpeed = 0.03;
	
	// internal - do not change
	ClType = OC_Missile;
	AnimCycle = 0.0;
	AnimDir = 1;
	AnimMin = 0.0;
	AnimMax = 1.0;
	
	Parachuting = true;
	Dead = false;
	
	Sound = 85;
	ShouldPlaySound = true;
	
	if(Desc->SpX > 0)
		Direction = 1;
	else
		Direction = -1;
}

void PxKirovAirshipBomb::DropFromAirship()
{
	if(Parachuting == false) return;
	
	Parachuting = false;
	SpeedMultiplier = 1.0;
	AnimCycle = 1.0;
	AnimDir = -1;
	AnimMin = 0.7;
	AnimMax = 1.0;
}

void PxKirovAirshipBomb::Explode()
{
	if(Dead) return;
	
	SpX = 0.0;
	SpY = 0.0;
	GravityFactor = 0.0;
	IsMaterial = false;
	Stopped = 1;
	
	Dead = true;
	
	DoExplosion(PosX, PosY, ExplosionBlast, ExplosionDamage, 0, 0);
}

void PxKirovAirshipBomb::Collide(CGObject* Obj, int type)
{
	Explode();
}

void PxKirovAirshipBomb::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{
	// Drawing operations are done here
	if(Type == M_DRAWQUEUE)
	{
		int spr;
		
		if(Parachuting)	spr = SpritePara;
		else			spr = SpriteFall;
		
		if(Direction > 0) spr += 262144;
		
		AddSprite(10.02, PosX, PosY, 0, spr, AnimCycle);
		
		return;
	}
	
	super;
	
	// This is run every frame
	if(Type == M_FRAME)
	{
		if(ShouldPlaySound)
		{
			ShouldPlaySound = false;
			PlayLocalSound(Sound, 5, 1.0, 1.0);
		}
		
		AnimCycle += AnimSpeed * AnimDir;
		if(AnimCycle > AnimMax)
		{
			AnimCycle = AnimMax;
			AnimDir = -AnimDir;
		}
		else if(AnimCycle < AnimMin)
		{
			AnimCycle = AnimMin;
			AnimDir = -AnimDir;
		}
		
		SpX = SpX * SpeedMultiplier;
		SpY = SpY * SpeedMultiplier;
		
		if(Dead)
		{
			Free(true);
		}
	}
}

// PxKirovAirship

PxKirovAirship : CMine;

PxKirovAirship::PxKirovAirship(CObject* Parent, CShootDesc* Desc)
{
	CMineParams MParams;
	zero(&MParams);
	
	MParams.Prefuse = 0;
	MParams.Flags = 0;
	MParams.Bias = 0;
	MParams.Radius = 1;
	MParams.Fuse = 10;
	MParams.Damage = 48;
	MParams.BlastPower = 100;
	
	super(Parent, &MParams, Desc, false, 0);
	
	SpeedDivider = 1000.0;
	GravityFactor = 0.0;
	
	ClType = OC_AirStrike;
	
	Direction = Desc->SpX;
	Team = Desc->Team;
	Worm = Desc->Worm;
	
	int teamcolor = GetTeamColor(Team);
	
	Sprite = kirovAirshipSprite->Index;
	CrosshairSprite = 2 + teamcolor;
	TextFont = 9 + teamcolor;
	Sound = 52;
	ShouldPlaySound = true;
	
	
	//////////////////////////////////
	
	NumBombs = 16;
	KirovAirshipDisplayText = MakeNumberString("x", NumBombs, "");
	FramesBetweenBombs = 10;
	PlaneMaxSpeed = float(6.0);
	PlaneMinSpeed = float(2.0);
	PlaneSpeed = PlaneMaxSpeed;
	
	PlaneAccelInterval = float(256.0);
	
	PlaneAppearXOffset = 1024;
	PlaneAppearYOffset = -128;
	
	PosX = 0.0;
	PosY = 0.0;
	
	if(Direction >= 0)
	{
		// Left to right
		Direction = 1;
	}
	else
	{
		// Right to left
		Direction = -1;
	}
	
	// Position the airplane on the horizontal axis
	PosY = PlaneAppearYOffset;
	
	if(Direction == 1)
	{
		//PosArriveX = 0 - PlaneAppearXOffset;
		PosStartX = Desc->AddX - 256;
		PosEndX = GS->LevelSX;
		PosArriveX = PosStartX - 512 - PlaneAppearXOffset;
		PosLeaveX = PosEndX + PlaneAppearXOffset;
	}
	else
	{
		//PosArriveX = GS->LevelSX + PlaneAppearXOffset;
		PosStartX = Desc->AddX + 256;
		PosEndX = 0;
		PosArriveX = PosStartX + 512 + PlaneAppearXOffset;
		PosLeaveX = PosEndX - PlaneAppearXOffset;
	}
	
	PosX = PosArriveX;

	Phase = 0;
	// 0: Arriving (max speed)
	// 1: Slowing down (max to min speed)
	// 2: Starting (min speed)
	// 3: Speeding up (min to max speed)
	// 4: Leaving (max speed)
	// 5: Dead
	
	FrameCounter = 0;
	AnimCycle = 0.0;
	AnimSpeed = 0.08;
	
	NextDropBomb = 0;
	
	CrosshairY = 0.0;
	CrosshairAnimCycle = 0.0;
}

void PxKirovAirship::DropBomb()
{
	if(Phase != 2) return;
	if(NumBombs <= 0) return;
	if(NextDropBomb > 0) return;
	
	NumBombs -= 1;
	KirovAirshipDisplayText = MakeNumberString("x", NumBombs, "");
	NextDropBomb = FramesBetweenBombs;
	
	CShootDesc SDesc;
	zero(&SDesc);
	
	SDesc.Team = Team;
	SDesc.X = PosX;
	SDesc.Y = PosY + 20;

	SDesc.SpX = Direction * 0.01;
	SDesc.SpY = 0.0;
	
	new PxKirovAirshipBomb(GetObject(25, 0), &SDesc);
	
	if(NumBombs == 0)
	{
		if((PosX + 64) * Direction < PosEndX * Direction)
		{
			PosEndX = PosX + 64 * Direction;
		}
	}
}

void PxKirovAirship::RenderPlane()
{
	if(Direction > 0)
	{
		AddSprite(10, PosX, PosY, 0, 262144 + Sprite, AnimCycle);
	}
	else
	{
		AddSprite(10, PosX, PosY, 0, Sprite, AnimCycle);
	}
	
	if(CrosshairY > 0)
	{
		AddSprite(10.01, PosX, PosY + CrosshairY, 0, CrosshairSprite, CrosshairAnimCycle);
	}
	
	int sx;
	int sy;
	
	int col = RGB(0, 0, 0);
	KirovAirshipDisplayBitmap->Rect(0, 0, 58, 17, col);
	
	GS->Disp->DrawText(TextFont , KirovAirshipDisplayBitmap, 27, 0, KirovAirshipDisplayText, &sx, &sy);
	
	col = RGB(157,157,157);
	KirovAirshipDisplayBitmap->HLine(1, 57, 0, col);
	KirovAirshipDisplayBitmap->HLine(1, 57, 16, col);
	KirovAirshipDisplayBitmap->VLine(0, 1, 16, col);
	KirovAirshipDisplayBitmap->VLine(57, 1, 16, col);
	KirovAirshipDisplayBitmap->SetPixel(1, 1, col);
	KirovAirshipDisplayBitmap->SetPixel(56, 1, col);
	KirovAirshipDisplayBitmap->SetPixel(56, 15, col);
	KirovAirshipDisplayBitmap->SetPixel(1, 15, col);
	
	AddBitmap(1.01, PosX, PosY - 40, KirovAirshipDisplayBitmap, 58, 17, 0);
	AddSpriteEx(1, int(PosX - 15), int(PosY - 40), 250, 0.0, 0.0, 0.6);
}

void PxKirovAirship::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{     
        // Drawing operations are done here
	if(Type == M_DRAWQUEUE)
	{
		RenderPlane();
		return;
	}
	
	super;
	
	// This is run every frame
	if(Type == M_FRAME)
	{
		float r;
		
		//SetActiveLevel(10);
		
		AnimCycle += AnimSpeed;
		while(AnimCycle > 1.0) AnimCycle -= 1.0;
		
		if(NextDropBomb > 0) NextDropBomb -= 1;
		
		if(ShouldPlaySound)
		{
			ShouldPlaySound = false;
			PlaySound(Sound, 8, 1, 0);
		}
		
		if(Root->RetreatTimer > 0) Root->RetreatTimer = 1000;
		
		if(Phase == 0)
		{
			PlaneSpeed = PlaneMaxSpeed;
			CrosshairY = 0.0;
			CrosshairAnimCycle = 1.0;
			
			if(PosX * Direction > (PosStartX * Direction) - PlaneAccelInterval)
				Phase = 1;
		}
		else if(Phase == 1)
		{
			r = ((PosStartX - PosX) * Direction) / PlaneAccelInterval;
			PlaneSpeed = Lerp(r, PlaneMinSpeed, PlaneMaxSpeed);
			CrosshairY = Lerp(r, 64.0, 0.0);
			CrosshairAnimCycle = r;
			
			if(PosX * Direction > PosStartX * Direction)
				Phase = 2;
		}
		else if(Phase == 2)
		{
			PlaneSpeed = PlaneMinSpeed;
			CrosshairY = 64.0;
			CrosshairAnimCycle = 0.0;
			
			if(PosX * Direction > PosEndX * Direction)
				Phase = 3;
		}
		else if(Phase == 3)
		{
			r = ((PosX - PosEndX) * Direction) / PlaneAccelInterval;
			PlaneSpeed = Lerp(r, PlaneMinSpeed, PlaneMaxSpeed);
			CrosshairY = Lerp(r, 64.0, 0.0);
			CrosshairAnimCycle = r;
			
			if(PosX * Direction > (PosEndX * Direction) + PlaneAccelInterval)
			{
				Phase = 4;
				
				for(local i = 1; i < Env->Objs.Count; i+=1)
				{
					local obj = Env->Objs.Objs[i];
					
					if(obj == NullObj) continue;
					if(obj is PxKirovAirshipBomb)
					{
						local cow = PxKirovAirshipBomb(obj);
						cow->DropFromAirship();
					}
				}
			}
		}
		else if(Phase == 4)
		{
			PlaneSpeed = PlaneMaxSpeed;
			CrosshairY = 0.0;
			CrosshairAnimCycle = 1.0;
			
			if(PosX * Direction > PosLeaveX * Direction)
				Phase = 5;
		}
		
		//if(Phase < 3)
		//{
		//	GS->CameraX = PosX;
		//	GS->CameraY = PosY + (GS->ResY / 2) - 64;
		//}
		
		SpX = 0;
		SpY = 0;
		PosX += PlaneSpeed * Direction;
		
		if(Phase == 5)
		{
			Free(true);
		}
	}
}

override void CWorm::FireFinal(CWeapon* Weap, CShootDesc* Desc)
{  
        if(Weap->CheckName("Kirov Airship"))
	{              
                new PxKirovAirship(GetObject(25, 0), Desc);
		//SetState(WS_CONTROL_WEAPON);
	}
	else
	{
		super;
	}
}

override void CWorm::Message(CObject* sender, EMType Type, int MSize, CMessageData* MData)
{
	if(Type == M_SPACE)
	{
		for(local i = 1; i < Env->Objs.Count; i+=1)
		{
			local obj = Env->Objs.Objs[i];
			
			if(obj == NullObj) continue;
			if(obj is PxKirovAirship)
			{
				local bb = PxKirovAirship(obj);
				if(bb->Team == WormTeam)
				{
					bb->DropBomb();
				}
			}
		}
	}
	
	super;
}
