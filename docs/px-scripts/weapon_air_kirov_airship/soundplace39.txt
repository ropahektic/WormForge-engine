CSoundFile* hma;
CSoundFile* hmb;
CSoundFile* USSRanthem;

void soundplace39::Init()
{
	hma           = new CSoundFile(GetAttachment("Hell March A.wav"));
	hmb           = new CSoundFile(GetAttachment("Hell March B.wav"));
	USSRanthem    = new CSoundFile(GetAttachment("USSR National Anthem.wav"));
        
        KASoundLimiter = 0;
}

override void CWorm::FireFinal(CWeapon* Weap,CShootDesc* Desc)
{
  if(Weap->CheckName("Kirov Airship"))
  {  
             local i = RandomInt(1,3);
	     	 if      (i == 1) hma->Play(1.2, 0.0, false);
             else if (i == 2) hmb->Play(1.2, 0.0, false);
             else if (i == 3) USSRanthem->Play(1.5, 0.0, false);
  }
  super;
}
